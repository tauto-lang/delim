113fn one_of_text(x: Text, l: List(Text)): Bool ! {div, overflow} {
114  match l {
115    Nil => false
116    Cons(y, rest) => x == y || one_of_text(x, rest)
117  }
118}

120fn byte_of(t: Text, i: u64): u64 {
121  match Text::byte(t, i) {
122    Some(b) => U64::of_u8(b)
123    None => 0
124  }
125}
126
127fn letter_or_digit(b: u64): Bool {
128  (b >= 48 && b <= 57) || (b >= 65 && b <= 90) || (b >= 97 && b <= 122)
129}
130
131// A byte of an SVG value: a letter, a digit, space, or one of `.,-+#%()_`. There is no `:`, so
132// no scheme, no `;`, `"`, `'`, `<`, `>` or `&`, so nothing that ends the value or starts markup.
133fn svg_value_byte(b: u64): Bool {
134  letter_or_digit(b) || b == 32 || b == 46 || b == 44 || b == 45 || b == 43 || b == 35 || b == 37 || b == 40 || b == 41 || b == 95
135}
136
137fn lower(b: u64): u64 ! {overflow} { if b >= 65 && b <= 90 { b + 32 } else { b } }
138
139// Whether `url(` starts at `i` (in any case) and is not followed by `#`: a reference to
140// anything but a fragment of the page.
141fn far_url_at(v: Text, i: u64): Bool ! {div, overflow} {
142  lower(
143    byte_of(v, i)
144  ) == 117 && lower(
145    byte_of(v, i + 1)
146  ) == 114 && lower(byte_of(v, i + 2)) == 108 && byte_of(v, i + 3) == 40 && byte_of(v, i + 4) != 35
147}
148
149fn svg_value_from(v: Text, i: u64): Bool ! {div, overflow} {
150  if i >= Text::len(v) { true } else {
151    svg_value_byte(byte_of(v, i)) && not(far_url_at(v, i)) && svg_value_from(v, i + 1)
152  }
153}

155// Whether the DOM renderer creates the SVG element `tag`.
156pub fn svg_tag_ok(tag: Text): Bool ! {div, overflow} { one_of_text(tag, svg_tags()) }
157
158// Whether the DOM renderer writes `name="value"` on an SVG element: the name from the list,
159// and the value of the bytes above, with `url(` only for a fragment (`url(#id)`).
160pub fn svg_attr_ok(name: Text, value: Text): Bool ! {div, overflow} {
161  one_of_text(name, svg_attrs()) && svg_value_from(value, 0)
162}

168  )

171pub fn svg_attr_refusal(name: Text): Text {
172  concat(

178// ---- CSS custom properties ----------------------------------------------------------------
179
180// A number for a CSS custom property. The value is written from the number and its unit, so
181// it is always a single CSS number: it cannot end the declaration or name a URL.
182pub(open) enum Css {
183  Px(i64)
184  Pct(i64)
185  Num(i64)
186  Deg(i64)
187  // Removes the property.
188  Unset
189}
190
191// The CSS text of a value; "" for `Unset`.
192pub fn css_text(v: Css): Text {
193  match v {
194    Px(n) => concat(Text::of_i64(n), "px")
195    Pct(n) => concat(Text::of_i64(n), "%")
196    Num(n) => Text::of_i64(n)
197    Deg(n) => concat(Text::of_i64(n), "deg")
198    Unset => ""
199  }
200}

202fn var_rest(name: Text, i: u64): Bool ! {div, overflow} {
203  if i >= Text::len(name) { true } else {
204    let b := byte_of(name, i)
205    (letter_or_digit(b) || b == 45 || b == 95) && var_rest(name, i + 1)
206  }
207}
208
209// Whether `name` is a custom property delim sets: `--`, then letters, digits, `-` and `_`,
210// at most 64 bytes. A page's style sheet reads it with `var(--name)`; no other property of
211// `style` is written.
212pub fn var_ok(name: Text): Bool ! {div, overflow} {
213  Text::len(name) > 2 && Text::len(name) <= 64 && byte_of(
214    name,
215    0
216  ) == 45 && byte_of(name, 1) == 45 && var_rest(name, 2)
217}
