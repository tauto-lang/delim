34// The HTML renderer, for rendering on the server and ahead of time: it applies the patches
35// to a tree of its own and writes the tree out as HTML.
36
37struct(shared) HNode {
38  // The tag name; empty for a text node.
39  tag: Text,
40  text: Text,
41  // Whether `text` is HTML written as it is, from a `TrustedHtml`, instead of the children.
42  raw: Bool,
43  attrs: List(@name Text × @value Text),
44  // Newest first, so that appending a child takes one step.
45  children: List(u32)
46}
47
48struct(shared) Html {
49  nodes: Mutex(HashMap(u32, HNode)),
50  parents: Mutex(HashMap(u32, u32)),
51  // Whether each node carries its number, for the DOM renderer to hydrate.
52  marked: Bool,
53  // Whether each group carries its number in `data-g`, for a streamed page to fill it later.
54  grouped: Bool
55}

57fn html_doc(marked: Bool): Html ! {cells} {
58  Html {
59    nodes: perform mutex_new(HashMap::new()),
60    parents: perform mutex_new(HashMap::new()),
61    marked: marked,
62    grouped: false
63  }
64}

105fn without_attr(
106  l: List(@name Text × @value Text),
107  k: Text
108): List(@name Text × @value Text) ! {div, overflow} {
109  match l {
110    Nil => Nil
111    Cons(a, rest) => {
112      let name: Text := (a: @name Text)
113      if name == k { without_attr(rest, k) } else { Cons(a, without_attr(rest, k)) }
114    }
115  }

122    raw: x.raw,

153fn without_child(l: List(u32), c: u32): List(u32) ! {div, overflow} {
154  match l {
155    Nil => Nil
156    Cons(x, rest) => if x == c { rest } else { Cons(x, without_child(rest, c)) }

160// `c` placed before `b` in the children `l`, which are newest first: after `b` in `l`. With no
161// `b`, `c` comes last, which is first in `l`.
162fn before_child(l: List(u32), c: u32, b: u32): List(u32) ! {div, overflow} {
163  match after_child(l, c, b) {
164    Some(m) => m
165    None => Cons(c, l)
166  }
167}
168
169fn after_child(l: List(u32), c: u32, b: u32): Option(List(u32)) ! {div, overflow} {
170  match l {
171    Nil => None
172    Cons(x, rest) => if x == b { Some(Cons(x, Cons(c, rest))) } else {
173      match after_child(rest, c, b) {
174        Some(m) => Some(Cons(x, m))
175        None => None
176      }
177    }
178  }
179}

182  match l {
183    Nil => acc
184    Cons(x, rest) => reversed(rest, Cons(x, acc))

199fn html_attr(h: Html, n: u32, a: HtmlAttr): Unit ! {wait, div, overflow, cells} {
200  match a {
201    // An attribute `Attr::refused` names is not written.
202    Attr(k, v) => match Attr::refused(k, v) {
203      None => put_attr(h, n, k, v)
204      Some(why) => #()
205    }
206    Aria(role, l) => {
207      put_attr(h, n, "role", role)
208      if l == "" { #() } else { put_attr(h, n, "aria-label", l) }
209    }
210    Value(v) => put_attr(h, n, "value", v)
211    Disabled(b) => bool_attr(h, n, "disabled", b)
212    Checked(b) => bool_attr(h, n, "checked", b)
213    Selected(b) => bool_attr(h, n, "selected", b)
214    Multiple(b) => bool_attr(h, n, "multiple", b)
215    Hidden(b) => bool_attr(h, n, "hidden", b)
216    Required(b) => bool_attr(h, n, "required", b)
217    ReadOnly(b) => bool_attr(h, n, "readonly", b)
218    OptionalAttr(k, v) => match v {
219      None => clear_attr(h, n, k)
220      Some(text) => html_attr(h, n, Attr(k, text))
221    }
222    // Not written: `html_safe` has no SVG and no `style`, and a drag needs the DOM.
223    SvgAttr(k, v) => #()
224    Var(k, v) => #()
225    DragSource(b) => #()
226  }
227}

229fn html_apply(h: Html, ps: List(Patch(HtmlKind, HtmlAttr))): Unit ! {wait, div, overflow, cells} {
230  match ps {
231    Nil => #()
232    Cons(p, rest) => {
233      match p {
234        Create(n, k) => match k {
235          // A refused tag is written as an element that shows nothing.
236          Tag(tag) => put_hnode(
237            h,
238            n.id,
239            HNode {
240              tag: written_tag(tag),
241              text: "",
242              raw: false,
243              attrs: match tag_refused(tag) {
244                None => Nil
245                Some(why) => Cons(#(@name "hidden", @value ""), Nil)
246              },
247              children: Nil
248            }
249          )
250          Island(name, props, inner) => put_hnode(
251            h,
252            n.id,
253            HNode {
254              tag: "delim-island",
255              // Written as it is, not escaped.
256              text: trusted_text(inner),
257              raw: true,
258              attrs: Cons(
259                #(@name "data-props", @value props),
260                Cons(#(@name "data-name", @value name), Nil)
261              ),
262              children: Nil
263            }
264          )
265          // SVG is not written (`html_safe` has none): it shows nothing, as a refused tag.
266          Svg(tag) => put_hnode(
267            h,
268            n.id,
269            HNode {
270              tag: "delim-refused",
271              text: "",
272              raw: false,
273              attrs: Cons(#(@name "hidden", @value ""), Nil),
274              children: Nil
275            }
276          )
277          Trusted(html) => put_hnode(
278            h,
279            n.id,
280            HNode {
281              tag: "delim-trusted",
282              // Written as it is, not escaped: a `TrustedHtml` vouches for it.
283              text: trusted_text(html),
284              raw: true,
285              attrs: Cons(#(@name "style", @value "display:contents"), Nil),
286              children: Nil
287            }
288          )
289        }
290        CreateText(n, s) => put_hnode(
291          h,
292          n.id,
293          HNode { tag: "", text: s, raw: false, attrs: Nil, children: Nil }
294        )
295        // A group is written as the element the DOM renderer uses for it, so that the page
296        // and the DOM built over it have the same shape.
297        CreateGroup(n) => put_hnode(
298          h,
299          n.id,
300          HNode {
301            tag: "delim-group",
302            text: "",
303            raw: false,
304            attrs: Cons(#(@name "style", @value "display:contents"), Nil),
305            children: Nil
306          }
307        )
308        SetProp(n, a) => html_attr(h, n.id, a)
309        SetText(n, s) => with_hnode(
310          h,
311          n.id,
312          (x: HNode) => HNode {
313            tag: x.tag,
314            text: s,
315            raw: x.raw,
316            attrs: x.attrs,
317            children: x.children
318          }
319        )
320        Append(parent, c) => {
321          detach(h, c.id)
322          set_children(h, parent.id, Cons(c.id, children_of(h, parent.id)))
323          set_parent(h, c.id, parent.id)
324        }
325        InsertBefore(parent, c, b) => {
326          detach(h, c.id)
327          set_children(h, parent.id, before_child(children_of(h, parent.id), c.id, b.id))
328          set_parent(h, c.id, parent.id)
329        }
330        Remove(n) => detach(h, n.id)
331        Template(id, n) => #()
332      }
333      html_apply(h, rest)
334    }
335  }
336}

375fn attr_value(attrs: List(@name Text × @value Text), name: Text): Option(Text) {
376  match attrs {
377    Nil => None
378    Cons(a, rest) => {
379      let k: Text := (a: @name Text)
380      let v: Text := (a: @value Text)
381      if k == name { Some(v) } else { attr_value(rest, name) }
382    }
383  }

386fn option_text(h: Html, cs: List(u32)): Text ! {wait, div, overflow, cells} {
387  match cs {
388    Nil => ""
389    Cons(n, rest) => match hnode(h, n) {
390      None => option_text(h, rest)
391      Some(x) => concat(
392        if x.tag == "" { x.text } else { option_text(h, reversed(x.children, Nil)) },
393        option_text(h, rest)
394      )
395    }
396  }

399// Value on a select selects the first matching option, including options in optgroups.
400fn select_options(
401  h: Html,
402  cs: List(u32),
403  value: Text,
404  found: Ref(Bool)
405): Unit ! {wait, div, overflow, cells} {
406  match cs {
407    Nil => #()
408    Cons(n, rest) => {
409      match hnode(h, n) {
410        None => #()
411        Some(x) => if x.tag == "option" {
412          let v := match attr_value(x.attrs, "value") {
413            Some(v) => v
414            None => option_text(h, reversed(x.children, Nil))
415          }
416          let selected := not(perform ref_get(found)) && v == value
417          bool_attr(h, n, "selected", selected)
418          if selected { perform ref_set(found, true) } else { #() }
419        } else { select_options(h, reversed(x.children, Nil), value, found) }
420      }
421      select_options(h, rest, value, found)
422    }
423  }
424}

426// The node `n` as the trees `write.taut` writes. When the document is marked, a text node
427// follows a comment with its number, which also keeps two texts apart, and an element
428// carries its number in `data-b`.
429fn tree_of(h: Html, n: u32): List(HTree) ! {wait, div, overflow, cells} {
430  match hnode(h, n) {
431    None => Nil
432    Some(x) => {
433      let id := decimal(U64::of_u32(n))
434      if x.tag == "" {
435        if h.marked { Cons(HMark(bytes_of(id)), Cons(HText(bytes_of(x.text)), Nil)) } else {
436          Cons(HText(bytes_of(x.text)), Nil)
437        }
438      } else {
439        // The attributes were put newest first; they are written oldest first, then the mark.
440        let marks := if h.grouped && x.tag == "delim-group" {
441          Cons(#(@name "data-g", @value id), x.attrs)
442        } else { x.attrs }
443        let attrs := bytes_pairs(
444          if h.marked { Cons(#(@name "data-b", @value id), marks) } else { marks },
445          Nil
446        )
447        let tag := bytes_of(x.tag)
448        if x.tag == "select" {
449          match attr_value(x.attrs, "value") {
450            None => #()
451            Some(v) => select_options(h, reversed(x.children, Nil), v, perform ref_new(false))
452          }
453        } else { #() }
454        if x.tag == "textarea" {
455          match attr_value(x.attrs, "value") {
456            None => Cons(HEl(tag, attrs, trees_of(h, reversed(x.children, Nil))), Nil)
457            Some(v) => {
458              // HTML parsing discards one initial newline in a textarea.
459              let content := if starts_with(v, "\n") { concat("\n", v) } else { v }
460              Cons(HEl(tag, attrs, Cons(HText(bytes_of(content)), Nil)), Nil)
461            }
462          }
463        } else if x.raw {
464          // The text of a `TrustedHtml`, checked again: the node keeps it as text.
465          match checked_bytes(bytes_of(x.text)) {
466            Some(t) => Cons(HRaw(tag, attrs, t), Nil)
467            None => Cons(HEl(tag, attrs, Nil), Nil)
468          }
469        } else { Cons(HEl(tag, attrs, trees_of(h, reversed(x.children, Nil))), Nil) }
470      }
471    }
472  }
473}

497  match ws {
498    Nil => Nil
499    Cons(w, rest) => Cons(w.key, want_keys(rest))

503// Each key given to `load` in turn: the answers of a `loads` that loads one at a time.
504fn each_loaded<!e>(
505  keys: List(Text),
506  load: (key: Text) -> Result(Text, Text) ! {div, overflow} ∪ e
507): List(Result(Text, Text)) ! {div, overflow} ∪ e {
508  match keys {
509    Nil => Nil
510    Cons(k, rest) => {
511      let r := load(k)
512      Cons(r, each_loaded(rest, load))
513    }
514  }
515}
516
517// Settles each of `ws` with the answer at the same place in `rs`. A want with no answer, as
518// when the loader stopped waiting for it, is rejected.
519fn settle_all(ws: List(Want), rs: List(Result(Text, Text))): Unit ! {rt, div, overflow} {
520  match ws {
521    Nil => #()
522    Cons(w, rest) => match rs {
523      Nil => {
524        reject(w, "the loader gave no answer")
525        settle_all(rest, Nil)
526      }
527      Cons(r, more) => {
528        match r {
529          Ok(v) => resolve(w, v)
530          Err(why) => reject(w, why)
531        }
532        settle_all(rest, more)
533      }
534    }
535  }
536}

538// Loads what the view asks for until it asks for nothing more. The resources asked for in
539// one step are given to `loads` together, so that a server can load them at once.
540fn load_until_done<!e>(
541  t: Tree(HtmlKind, HtmlAttr),
542  loads: (keys: List(Text)) -> List(Result(Text, Text)) ! {div, overflow} ∪ e
543): Unit ! {rt, div, overflow} ∪ e {
544  match take_wants() {
545    Nil => #()
546    Cons(first, more) => {
547      let ws := Cons(first, more)
548      settle_all(ws, loads(want_keys(ws)))
549      load_until_done(t, loads)
550    }
551  }
552}

554fn render_loading<!e>(
555  h: Html,
556  draw: (t: Tree(HtmlKind, HtmlAttr), v: Vocab(HtmlKind, HtmlAttr)) -> Node ! {rt, div, overflow},
557  loading: Bool,
558  loads: (keys: List(Text)) -> List(Result(Text, Text)) ! {div, overflow} ∪ e
559): TrustedHtml ! {wait, div, overflow, cells} ∪ e {
560  let t := tree()
561  let root := reactive(() => {
562    let n := draw(t, web_vocab())
563    if loading { load_until_done(t, loads) } else { #() }
564    n
565  })
566  html_apply(h, take_patches(t))
567  // `write_safe` proves that `html_safe` accepts it, whatever the view made.
568  written_all(tree_of(h, root.id))
569}

573pub fn render_html_loading<!e>(

592// What one step of loading changed, as written before the step: the groups whose content is
593// to be sent again, and whether something outside every group changed, so that the whole view
594// is.
595struct(shared) Changed {
596  groups: List(u32),
597  whole: Bool
598}

600fn with_group(l: List(u32), g: u32): List(u32) ! {div, overflow} {
601  match l {
602    Nil => Cons(g, Nil)
603    Cons(x, rest) => if x == g { l } else { Cons(x, with_group(rest, g)) }
604  }
605}

607// The group already written that holds what changes at `n`, walking up from `n`. A node made
608// in this step is not written yet, and what is under it is sent with whatever holds it.
609fn changed_at(h: Html, root: u32, n: u32, c: Changed): Changed ! {wait, div, overflow, cells} {
610  match hnode(h, n) {
611    None => c
612    Some(x) => if x.tag == "delim-group" {
613      Changed { groups: with_group(c.groups, n), whole: c.whole }
614    } else if n == root { Changed { groups: c.groups, whole: true } } else {
615      match parent_of(h, n) {
616        None => c
617        Some(p) => changed_at(h, root, p, c)
618      }
619    }
620  }
621}

625  root: u32,
626  ps: List(Patch(HtmlKind, HtmlAttr)),
627  c: Changed
628): Changed ! {wait, div, overflow, cells} {
629  match ps {
630    Nil => c
631    Cons(p, rest) => changed_by(h, root, rest, match p {
632      Append(parent, x) => changed_at(h, root, parent.id, c)
633      InsertBefore(parent, x, b) => changed_at(h, root, parent.id, c)
634      Remove(n) => match parent_of(h, n.id) {
635        None => c
636        Some(q) => changed_at(h, root, q, c)
637      }
638      SetText(n, s) => changed_at(h, root, n.id, c)
639      SetProp(n, a) => changed_at(h, root, n.id, c)
640      CreateGroup(_) => c
641      Create(_, _) => c
642      CreateText(_, _) => c
643      Template(_, _) => c
644    })
645  }
646}
647
648fn has_group(l: List(u32), g: u32): Bool {
649  match l {
650    Nil => false
651    Cons(x, rest) => x == g || has_group(rest, g)
652  }
653}
654
655fn under_any(h: Html, gs: List(u32), n: u32): Bool ! {wait, div, overflow, cells} {
656  match parent_of(h, n) {
657    None => false
658    Some(p) => has_group(gs, p) || under_any(h, gs, p)
659  }
660}
661
662// The groups of `gs` not inside another of them, whose content is sent with that one's.
663fn outermost(h: Html, gs: List(u32), l: List(u32)): List(u32) ! {wait, div, overflow, cells} {
664  match l {
665    Nil => Nil
666    Cons(g, rest) => if under_any(h, gs, g) { outermost(h, gs, rest) } else {
667      Cons(g, outermost(h, gs, rest))
668    }
669  }
670}
671
672fn group_parts(h: Html, gs: List(u32)): Text ! {wait, div, overflow, cells} {
673  match gs {
674    Nil => ""
675    Cons(g, rest) => {
676      let inner := written_all(trees_of(h, reversed(children_of(h, g), Nil)))
677      concat(
678        concat(
679          concat(concat("<template data-for=\"", decimal(U64::of_u32(g))), "\">"),
680          concat(trusted_text(inner), "</template>")
681        ),
682        group_parts(h, rest)
683      )
684    }
685  }
686}

688// What `render_html_streaming` asks of `io`: the resource of a key, or that a part be sent.
689// One function does both, so that both run with the same effects.
690pub(open) enum(shared) Stream {
691  Fetch(Text),
692  Part(Text)
693}

695// One resource loaded, with its place among the keys asked for together.
696pub(open) struct(shared) LoadAnswer { index: u64, got: Result(Text, Text) }

698// The steps run outside the runtime, with each step's work in it by `with_runtime`, so that
699// no closure in the runtime holds `io`.
700fn stream_steps<!e>(
701  h: Html,
702  t: Tree(HtmlKind, HtmlAttr),
703  r: Runtime,
704  root: u32,
705  start: (keys: List(Text)) -> Unit ! {wait, div, overflow} ∪ e,
706  next: () -> Option(LoadAnswer) ! {wait, div, overflow} ∪ e,
707  io: (s: Stream) -> Result(Text, Text) ! {wait, div, overflow} ∪ e,
708  whole: Bool
709): Option(Bool) ! {wait, div, overflow, cells} ∪ e {
710  match with_runtime(r, () => take_wants()) {
711    Nil => Some(whole)
712    Cons(first, more) => {
713      let ws := Cons(first, more)
714      let state := perform ref_new(Some(whole))
715      start(want_keys(ws))
716      stream_answers(h, t, r, root, ws, next, io, state)
717      // What the loader stopped waiting for is rejected, and that change is sent too.
718      stream_settled(h, t, r, root, () => reject_unsettled(ws), io, state)
719      match perform ref_get(state) {
720        None => None
721        Some(w) => stream_steps(h, t, r, root, start, next, io, w)
722      }
723    }
724  }
725}
726
727// Sends what each answer of `next` changes, until it has no more or `io` refuses a part.
728fn stream_answers<!e>(
729  h: Html,
730  t: Tree(HtmlKind, HtmlAttr),
731  r: Runtime,
732  root: u32,
733  ws: List(Want),
734  next: () -> Option(LoadAnswer) ! {wait, div, overflow} ∪ e,
735  io: (s: Stream) -> Result(Text, Text) ! {wait, div, overflow} ∪ e,
736  state: Ref(Option(Bool))
737): Unit ! {wait, div, overflow, cells} ∪ e {
738  match next() {
739    None => #()
740    Some(a) => {
741      let going := match want_at(ws, a.index) {
742        None => true
743        Some(w) => if perform ref_get(w.settled) { true } else {
744          stream_settled(h, t, r, root, () => match a.got {
745            Ok(v) => resolve(w, v)
746            Err(why) => reject(w, why)
747          }, io, state)
748        }
749      }
750      if going { stream_answers(h, t, r, root, ws, next, io, state) } else { #() }
751    }
752  }
753}
754
755fn want_at(ws: List(Want), i: u64): Option(Want) ! {overflow} {
756  match ws {
757    Nil => None
758    Cons(w, rest) => if i == 0 { Some(w) } else { want_at(rest, i - 1) }
759  }
760}
761
762fn reject_unsettled(ws: List(Want)): Unit ! {rt, div, overflow} {
763  match ws {
764    Nil => #()
765    Cons(w, rest) => {
766      if perform ref_get(w.settled) { #() } else { reject(w, "the loader gave no answer") }
767      reject_unsettled(rest)
768    }
769  }
770}
771
772// Runs `settle` in the runtime and sends what it changed. `state` is whether the whole view is
773// to be sent again, or None once `io` refused a part; the answer is whether to go on.
774fn stream_settled<!e>(
775  h: Html,
776  t: Tree(HtmlKind, HtmlAttr),
777  r: Runtime,
778  root: u32,
779  settle: () -> Unit ! {rt, div, overflow},
780  io: (s: Stream) -> Result(Text, Text) ! {wait, div, overflow} ∪ e,
781  state: Ref(Option(Bool))
782): Bool ! {wait, div, overflow, cells} ∪ e {
783  match perform ref_get(state) {
784    None => false
785    Some(whole) => {
786      with_runtime(r, settle)
787      let ps := take_patches(t)
788      let c := changed_by(h, root, ps, Changed { groups: Nil, whole: whole })
789      html_apply(h, ps)
790      // Once the whole view is to be sent again, groups are not sent one by one.
791      let going := if c.whole { true } else {
792        match c.groups {
793          Nil => true
794          Cons(first, more) => {
795            let gs := Cons(first, more)
796            sent(io, group_parts(h, outermost(h, gs, gs)))
797          }
798        }
799      }
800      perform ref_set(state, if going { Some(c.whole) } else { None })
801      going
802    }
803  }
804}

806// The keys `start` was last given, each fetched from `io` in turn by `next`.
807fn fetched_next<!e>(
808  queue: Ref(List(Text)),
809  at: Ref(u64),
810  io: (s: Stream) -> Result(Text, Text) ! {wait, div, overflow} ∪ e
811): Option(LoadAnswer) ! {wait, div, overflow, cells} ∪ e {
812  match perform ref_get(queue) {
813    Nil => None
814    Cons(k, rest) => {
815      perform ref_set(queue, rest)
816      let i := perform ref_get(at)
817      perform ref_set(at, i + 1)
818      Some(LoadAnswer { index: i, got: io(Fetch(k)) })
819    }
820  }
821}
822
823fn sent<!e>(
824  io: (s: Stream) -> Result(Text, Text) ! {wait, div, overflow} ∪ e,
825  part: Text
826): Bool ! {wait, div, overflow} ∪ e {
827  match io(Part(part)) {
828    Ok(u) => true
829    Err(why) => false
830  }
831}

835// `render_html_loading` in parts, each given to `io` as a `Part` as soon as it is ready: first the view
836// with every suspense boundary showing its fallback, then, each time a resource loads, a
837// `<template data-for>` with the new content of each boundary it changed, to put in the
838// element of the same `data-g`. If a resource changes something outside every boundary, the
839// whole view is sent last instead, as `<template data-for="app">`. An `Err` for a part stops the
840// loading. `io` answers a `Fetch` as `render_html_loading`'s `load`.
841pub fn render_html_streaming<!e>(
842  draw: (t: Tree(HtmlKind, HtmlAttr), v: Vocab(HtmlKind, HtmlAttr)) -> Node ! {rt, div, overflow},
843  io: (s: Stream) -> Result(Text, Text) ! {div, overflow} ∪ e
844): Unit ! {wait, div, overflow, cells} ∪ e {
845  let queue := perform ref_new(Nil)
846  let at := perform ref_new(0)
847  render_html_streaming_together(
848    draw,
849    (keys: List(Text)) => {
850      perform ref_set(queue, keys)
851      perform ref_set(at, 0)
852    },
853    () => fetched_next(queue, at, (s: Stream) => io(s)),
854    (s: Stream) => io(s)
855  )
856}
857
858// `render_html_streaming` with the resources asked for at once given to `start` together
859// instead of to `io` one by one: a server can load them at the same time, send each
860// boundary as soon as its resources are there, and stop waiting for a slow one. `next` gives
861// the next answer for the keys `start` was last given, in any order, or None once it gives no
862// more; a key it gave no answer for is rejected. `io` is given only `Part`s.
863pub fn render_html_streaming_together<!e>(
864  draw: (t: Tree(HtmlKind, HtmlAttr), v: Vocab(HtmlKind, HtmlAttr)) -> Node ! {rt, div, overflow},
865  start: (keys: List(Text)) -> Unit ! {wait, div, overflow} ∪ e,
866  next: () -> Option(LoadAnswer) ! {wait, div, overflow} ∪ e,
867  io: (s: Stream) -> Result(Text, Text) ! {wait, div, overflow} ∪ e
868): Unit ! {wait, div, overflow, cells} ∪ e {
869  let h := Html {
870    nodes: perform mutex_new(HashMap::new()),
871    parents: perform mutex_new(HashMap::new()),
872    marked: false,
873    grouped: true
874  }
875  let t := tree()
876  let made := reactive(() => StreamView { root: draw(t, web_vocab()), runtime: current() })
877  let root := made.root.id
878  html_apply(h, take_patches(t))
879  if sent(io, trusted_text(written_all(tree_of(h, root)))) {
880    if stream_steps(h, t, made.runtime, root, start, next, io, false) == Some(true) {
881      sent(
882        io,
883        concat(
884          concat("<template data-for=\"app\">", trusted_text(written_all(tree_of(h, root)))),
885          "</template>"
886        )
887      )
888      #()
889    } else { #() }
890  } else { #() }
891}

893// An island in a page rendered on the server: `view(props)` rendered in a tree of its own,
894// with each node's number, which the browser's `hydrate` of the same view and `props` takes
895// over. `props` is all the island starts from, as text.
896pub fn island(
897  t: Tree(HtmlKind, HtmlAttr),
898  name: Text,
899  props: Text,
900  draw: (
901    t: Tree(HtmlKind, HtmlAttr),
902    v: Vocab(HtmlKind, HtmlAttr),
903    props: Text
904  ) -> Node ! {rt, div, overflow}
905): Node ! {rt, div, overflow} {
906  let inner := render_with(
907    html_doc(true),
908    (t2: Tree(HtmlKind, HtmlAttr), v: Vocab(HtmlKind, HtmlAttr)) => draw(t2, v, props)
909  )
910  el(t, Island(name, props, inner), Nil, Nil)
911}
