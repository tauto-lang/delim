4// HTML that is safe to write into a page as it is, as the kernel reads it.
5//
6// `html_safe` reads bytes as a browser's tokenizer would, but accepts only a small part of
7// HTML: text with no `<`; a start tag of an element `tag_allowed` lets through, with attributes
8// that `Attr::value_allowed` lets through, each written ` name` or ` name="value"` with no `"`
9// or `<` in the value; an end tag of such an element; and a comment `<!--` digits `-->`. Nothing it
10// accepts opens a script, a style sheet, a frame or foreign content, sets a handler, or writes
11// a URL that runs code. That is the one claim the proofs below rest on and cannot check: it is
12// about browsers, not about the program.
13//
14// A `TrustedHtml` carries a proof that `html_safe` accepts its bytes, so one cannot be made
15// for any other bytes: the proof is erased when the program runs.
16
17// ---- escaping ---------------------------------------------------------------------------------

51      true => amp_bytes()

57  pub fn html(w: List(Nat)): List(Nat) {

60      Cons(c, rest) => List::append(escape_html_byte(c), Escape::html(rest))

67    y: Bool,
68    c: Nat,

89    y: Bool,
90    c: Nat,
91    e: Nat::eq(c, 60) = y
92  ): scan(SText, Escape::lt_as(y, c)) = SText {
93    match y {
94      true => Refl
95      false => Escape::gt_text(Nat::eq(c, 62), c, e)
96    }
97  }

99  theorem amp_text(y: Bool, c: Nat): scan(SText, Escape::amp_as(y, c)) = SText {
100    match y {
101      true => Refl
102      false => Escape::lt_text(Nat::eq(c, 60), c, Refl)
103    }
104  }

106  // Escaped text read from text ends in text.
107  pub theorem html_scan(w: List(Nat)): scan(SText, Escape::html(w)) = SText {
108    match w {
109      Nil => Refl
110      Cons(c, rest) => Eq::trans(
111        Scan::append(SText, escape_html_byte(c), Escape::html(rest)),
112        Eq::trans(
113          Eq::congr((s: Scan) => scan(s, Escape::html(rest)), Escape::amp_text(Nat::eq(c, 38), c)),
114          Escape::html_scan(rest)
115        )
116      )
117    }
118  }
119
120  // Escaped text is safe HTML, whatever it escapes.
121  pub theorem safe(w: List(Nat)): html_safe(Escape::html(w)) = true {
122    Eq::congr((s: Scan) => Scan::done(s), Escape::html_scan(w))
123  }
124
125  // ---- trusted HTML -----------------------------------------------------------------------------
126}

128// A byte as HTML that shows it: `&`, `<`, `>` and `"` as character references.
129fn escape_html_byte(c: Nat): List(Nat) { Escape::amp_as(Nat::eq(c, 38), c) }

131// Where `html_safe` is in the bytes. Names and values are kept reversed.
132pub(open) inductive Scan: Type {
133  // In text.
134  SText: Scan
135  // After `<`.
136  SLt: Scan
137  // In the name of a start tag.
138  SOpen(name: List(Nat)): Scan
139  // In the name of an end tag, after `</`.
140  SClose(name: List(Nat)): Scan
141  // In a start tag, after its name or a quoted value.
142  STag: Scan
143  // In a start tag, after a space: an attribute's name comes.
144  SSpace: Scan
145  // In the name of an attribute.
146  SAttr(name: List(Nat)): Scan
147  // After `=`: `"` comes.
148  SEq(name: List(Nat)): Scan
149  // In a quoted value.
150  SValue(name: List(Nat), value: List(Nat)): Scan
151  // After `<!`.
152  SBang: Scan
153  // After `<!-`.
154  SDash: Scan
155  // In the digits of a comment; `true` once there is one.
156  SDigits(any: Bool): Scan
157  // After the first `-` that ends a comment.
158  SEnd1: Scan
159  // After `--`.
160  SEnd2: Scan
161  // Not safe HTML.
162  SFail: Scan
163}

165// A byte of a name: a lowercase letter, a digit or `-`.
166fn name_byte(c: Nat): Bool { Bool::or(is_lower(c), Bool::or(is_digit_byte(c), Nat::eq(c, 45))) }
167
168fn scan_if(y: Bool, s: Scan): Scan {
169  match y { true => s, false => SFail }
170}

172pub module Scan {
173  pub fn text_as(y: Bool): Scan {
174    match y { true => SLt, false => SText }
175  }
176
177  fn lt(c: Nat): Scan {
178    match Nat::eq(c, 47) {
179      true => SClose(Nil)
180      false => ⊢ Scan from match Nat::eq(c, 33) {
181        true => SBang
182        false => scan_if(is_lower(c), SOpen([c]))
183      }
184    }
185  }
186
187  // After the name of a start tag, at `c`.
188  fn open_end(name: List(Nat), c: Nat): Scan {
189    match Nat::eq(c, 32) {
190      true => scan_if(tag_allowed(List::reverse(name)), SSpace)
191      false => ⊢ Scan from match Nat::eq(c, 62) {
192        true => scan_if(tag_allowed(List::reverse(name)), SText)
193        false => SFail
194      }
195    }
196  }
197
198  fn open_as(y: Bool, name: List(Nat), c: Nat): Scan {
199    match y {
200      true => SOpen(Cons(c, name))
201      false => Scan::open_end(name, c)
202    }
203  }
204
205  fn close_as(y: Bool, name: List(Nat), c: Nat): Scan {
206    match y {
207      true => SClose(Cons(c, name))
208      false => ⊢ Scan from match Nat::eq(c, 62) {
209        true => scan_if(tag_allowed(List::reverse(name)), SText)
210        false => SFail
211      }
212    }
213  }
214
215  fn tag(c: Nat): Scan {
216    match Nat::eq(c, 32) {
217      true => SSpace
218      false => ⊢ Scan from match Nat::eq(c, 62) { true => SText, false => SFail }
219    }
220  }
221
222  // After the name of an attribute with no value, at `c`.
223  fn attr_end(name: List(Nat), c: Nat): Scan {
224    match Nat::eq(c, 61) {
225      true => SEq(name)
226      false => ⊢ Scan from match Nat::eq(c, 32) {
227        true => scan_if(Attr::value_allowed(List::reverse(name), Nil), SSpace)
228        false => ⊢ Scan from match Nat::eq(c, 62) {
229          true => scan_if(Attr::value_allowed(List::reverse(name), Nil), SText)
230          false => SFail
231        }
232      }
233    }
234  }
235
236  fn attr_as(y: Bool, name: List(Nat), c: Nat): Scan {
237    match y {
238      true => SAttr(Cons(c, name))
239      false => Scan::attr_end(name, c)
240    }
241  }
242
243  fn value_as(y: Bool, name: List(Nat), value: List(Nat), c: Nat): Scan {
244    match y {
245      true => scan_if(Attr::value_allowed(List::reverse(name), List::reverse(value)), STag)
246      // A `<` in a value is refused: after `<textarea>`, a browser reads a value's
247      // `</textarea>` as the end of the element.
248      false => scan_if(not(Nat::eq(c, 60)), SValue(name, Cons(c, value)))
249    }
250  }
251
252  fn digits(any: Bool, c: Nat): Scan {
253    match is_digit_byte(c) {
254      true => SDigits(true)
255      false => ⊢ Scan from match Nat::eq(c, 45) {
256        true => scan_if(any, SEnd1)
257        false => SFail
258      }
259    }
260  }
261
262  pub fn byte(s: Scan, c: Nat): Scan {
263    match s {
264      SText => Scan::text_as(Nat::eq(c, 60))
265      SLt => Scan::lt(c)
266      SOpen(name) => Scan::open_as(name_byte(c), name, c)
267      SClose(name) => Scan::close_as(name_byte(c), name, c)
268      STag => Scan::tag(c)
269      SSpace => scan_if(is_lower(c), SAttr([c]))
270      SAttr(name) => Scan::attr_as(name_byte(c), name, c)
271      SEq(name) => scan_if(Nat::eq(c, 34), SValue(name, Nil))
272      SValue(name, value) => Scan::value_as(Nat::eq(c, 34), name, value, c)
273      SBang => scan_if(Nat::eq(c, 45), SDash)
274      SDash => scan_if(Nat::eq(c, 45), SDigits(false))
275      SDigits(any) => Scan::digits(any, c)
276      SEnd1 => scan_if(Nat::eq(c, 45), SEnd2)
277      SEnd2 => scan_if(Nat::eq(c, 62), SText)
278      SFail => SFail
279    }
280  }
281
282  pub fn done(s: Scan): Bool {
283    match s {
284      SText => true
285      SLt => false
286      SOpen(name) => false
287      SClose(name) => false
288      STag => false
289      SSpace => false
290      SAttr(name) => false
291      SEq(name) => false
292      SValue(name, value) => false
293      SBang => false
294      SDash => false
295      SDigits(any) => false
296      SEnd1 => false
297      SEnd2 => false
298      SFail => false
299    }
300  }
301
302  pub theorem append(
303    s: Scan,
304    a: List(Nat),
305    b: List(Nat)
306  ): scan(s, List::append(a, b)) = scan(scan(s, a), b) {
307    match a {
308      Nil => Refl
309      Cons(c, rest) => Scan::append(Scan::byte(s, c), rest, b)
310    }
311  }
312}

314pub fn scan(s: Scan, w: List(Nat)): Scan {
315  match w {
316    Nil => s
317    Cons(c, rest) => scan(Scan::byte(s, c), rest)
318  }
319}

321// Whether `w` is safe HTML: read from text, it ends in text.
322pub fn html_safe(w: List(Nat)): Bool { Scan::done(scan(SText, w)) }

326// A byte that is not `<` keeps text text.
327theorem text_step(c: Nat, h: Nat::eq(c, 60) = false): scan(SText, [c]) = SText {
328  Eq::congr((y: Bool) => Scan::text_as(y), h)
329}

331// HTML `html_safe` accepts, with the proof.
332pub(open) inductive TrustedHtml: Type {
333  Mk(bytes: List(Nat), ok: html_safe(bytes) = true): TrustedHtml
334}
335
336pub fn trusted_bytes(h: TrustedHtml): List(Nat) {
337  match h {
338    Mk(b, ok) => b
339  }
340}

342// Bytes as HTML that shows them.
343pub fn escaped_bytes(
344  w: List(Nat)

347fn checked_as(y: Bool, w: List(Nat), e: html_safe(w) = y): Option(TrustedHtml) {
348  match y {
349    true => Some(TrustedHtml::Mk(w, e))
350    false => None
351  }
352}

354// `w` as trusted HTML, when `html_safe` accepts it.
355pub fn checked_bytes(w: List(Nat)): Option(TrustedHtml) { checked_as(html_safe(w), w, Refl) }
