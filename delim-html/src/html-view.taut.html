17// `view!` of the web elements: the dialect of `delim::view` for HTML. A tag is one of the
18// elements `policy.taut` lists (or a custom element with a dash in its name), an attribute one of
19// its attributes, `data-*` or `aria-*`; what the renderers would refuse is an error of the program.
20//
21//   view!(
22//     ```view
23//     $t <div class="row"><p>"Count: " $text_dyn(t, () => decimal(get(n)))</p>
24//       <button aria-label="Add" on:press=${(d: EventData) => update(n, (x: u64) => x + 1)}>"+"</button>
25//     </div>
26//     ```
27//   )
28//
29// The expression first is the tree. An attribute is a quoted text, fixed, or `${e}` with `e` a
30// text, followed; `on:press` and the other event names take a handler of the event's data. A
31// child is an element, a quoted text, or `$e` with `e` a node.
32
33pub fn view_url_attr_names(): List(Text) {
34  ["action", "cite", "formaction", "href", "poster", "src"]
35}
36
37fn text_in(t: Text, l: List(Text)): Bool ! {div, overflow} {
38  match l {
39    Nil => false
40    Cons(x, rest) => x == t || text_in(t, rest)
41  }
42}
43
44fn has_dash(t: Text, i: u64): Bool ! {div, overflow} {
45  match Text::byte(t, i) {
46    None => false
47    Some(b) => U64::of_u8(b) == 45 || has_dash(t, i + 1)
48  }
49}

256  fn byte_ok(c: u64, first: Bool): Bool {
257    (c >= 97 && c <= 122) || (not(first) && ((c >= 48 && c <= 57) || c == 45))
258  }

260  fn name_from(t: Text, i: u64): Bool ! {div, overflow} {
261    match Text::byte(t, i) {
262      None => i > 0
263      Some(b) => HtmlView::byte_ok(U64::of_u8(b), i == 0) && HtmlView::name_from(t, i + 1)
264    }
265  }

267  fn tag_refused(tag: Text): Option(Text) ! {div, overflow} {
268    if HtmlView::name_from(
269      tag,
270      0
271    ) && (text_in(tag, HtmlView::tag_names()) || has_dash(tag, 0)) { None } else {
272      Some(tag_refusal(tag))
273    }
274  }

276  fn attr_name_refused(name: Text): Option(Text) ! {div, overflow} {
277    if HtmlView::name_from(
278      name,
279      0
280    ) && (
281      text_in(
282        name,
283        HtmlView::attr_names()
284      ) || starts_with(name, "data-") || starts_with(name, "aria-")
285    ) { None } else { Some(Attr::refusal(name)) }
286  }

288  // The bytes of a URL before its first `/`, `?` or `#`, lowercased, without spaces and control
289  // bytes, as `url_head` in `policy.taut`.
290  fn url_head(t: Text, i: u64, acc: Text): Text ! {div, overflow} {
291    match Text::byte(t, i) {
292      None => acc
293      Some(b) => {
294        let c := U64::of_u8(b)
295        if c == 47 || c == 63 || c == 35 { acc } else if c <= 32 {
296          HtmlView::url_head(t, i + 1, acc)
297        } else {
298          let l := if c >= 65 && c <= 90 { c + 32 } else { c }
299          match Text::of_code_point(U32::of_u64(l)) {
300            Some(x) => HtmlView::url_head(t, i + 1, concat(acc, x))
301            // A byte of a longer character: it is not one of `&<>":`, so any stands for it.
302            None => HtmlView::url_head(t, i + 1, concat(acc, "~"))
303          }
304        }
305      }
306    }
307  }

309  fn find(t: Text, c: u64, i: u64): Option(u64) ! {div, overflow} {
310    match Text::byte(t, i) {
311      None => None
312      Some(b) => if U64::of_u8(b) == c { Some(i) } else { HtmlView::find(t, c, i + 1) }
313    }
314  }

316  pub fn url_ok(value: Text): Bool ! {div, overflow} {
317    let h := HtmlView::url_head(value, 0, "")
318    if HtmlView::find(
319      h,
320      38,
321      0
322    ) != None || HtmlView::find(
323      h,
324      60,
325      0
326    ) != None || HtmlView::find(h, 62, 0) != None || HtmlView::find(h, 34, 0) != None {
327      false
328    } else {
329      match HtmlView::find(h, 58, 0) {
330        None => true
331        Some(i) => match Text::slice(h, 0, i) {
332          Some(s) => text_in(s, HtmlView::scheme_names())
333          None => false
334        }
335      }
336    }
337  }

339  // Why the fixed `name="value"` may not be written, if it may not: what `Attr::refused` says,
340  // before the program runs.
341  fn attr_refused(name: Text, value: Text): Option(Text) ! {div, overflow} {
342    if name == "style" && value == "display:contents" { None } else {
343      match HtmlView::attr_name_refused(name) {
344        Some(why) => Some(why)
345        None => if text_in(name, view_url_attr_names()) && not(HtmlView::url_ok(value)) {
346          Some(url_refusal())
347        } else { None }
348      }
349    }
350  }

352  // Under `view!` without a tree, the handler is kept with `on_event`, so that what the
353  // `provide`s above handle is handled again when it runs.
354  fn handler(
355    u: Option(SynExpr),
356    kind: SynExpr,
357    h: SynExpr
358  ): SynExpr ! {macro_error, div, overflow} {
359    match u {
360      Some(v) => {
361        let f := View::handler_fn(h)
362        quote { on_event($kind, $f) }
363      }
364      None => View::handler(kind, h)
365    }
366  }

368  fn attr(u: Option(SynExpr), a: view::Attr): SynExpr ! {macro_error, div, overflow} {
369    let name := a.name.text
370    match Text::slice(name, 0, 3) {
371      Some(head) => if head == "on:" {
372        let kind := View::event(a.name.span, match Text::slice(name, 3, Text::len(name)) {
373          Some(e) => e
374          None => ""
375        })
376        match a.value {
377          Str(tok) => Syn::error(tok.span, "an event takes a handler, `${…}`")
378          TautoExpr(h) => HtmlView::handler(u, kind, h)
379        }
380      } else { HtmlView::plain_attr(u, a.name.span, name, a.value) }
381      None => HtmlView::plain_attr(u, a.name.span, name, a.value)
382    }
383  }

385  // The view of `t`, when `t` is the tree `view!` without a tree took from `ui`.
386  fn ui_of(t: SynExpr): Option(SynExpr) {
387    match t {
388      ExVar(n) => if n.name == "delim_view_tree" {
389        Some(ExVar(SynIdent { declaration: None, name: "delim_view_ui", ctx: n.ctx, span: n.span }))
390      } else { None }
391      _ => None
392    }
393  }

395  // A plain attribute: its name is checked here, and a fixed value too, so what the renderers
396  // would refuse is an error of the program instead.
397  fn plain_attr(
398    u: Option(SynExpr),
399    at: SynSpan,
400    name: Text,
401    v: view::Value
402  ): SynExpr ! {macro_error, div, overflow} {
403    match HtmlView::attr_name_refused(name) {
404      Some(why) => Syn::error(at, why)
405      None => #()
406    }
407    let n := Syn::lift_text(name, at)
408    match v {
409      Str(tok) => {
410        let literal := View::unquote(tok)
411        match HtmlView::attr_refused(name, literal) {
412          Some(why) => Syn::error(tok.span, why)
413          None => #()
414        }
415        let s := Syn::lift_text(literal, tok.span)
416        quote { Static(Attr($n, $s)) }
417      }
418      TautoExpr(e) => if name == "value" {
419        HtmlView::dyn_attr(u, quote { Value($e) })
420      } else if name == "disabled" {
421        HtmlView::dyn_attr(u, quote { Disabled($e) })
422      } else if name == "checked" { HtmlView::dyn_attr(u, quote { Checked($e) })
423      } else if name == "selected" { HtmlView::dyn_attr(u, quote { Selected($e) })
424      } else if name == "multiple" { HtmlView::dyn_attr(u, quote { Multiple($e) })
425      } else if name == "hidden" { HtmlView::dyn_attr(u, quote { Hidden($e) })
426      } else if name == "required" { HtmlView::dyn_attr(u, quote { Required($e) })
427      } else if name == "readonly" { HtmlView::dyn_attr(u, quote { ReadOnly($e) })
428      } else { HtmlView::dyn_attr(u, quote { Attr($n, AttrText::attr_text($e)) }) }
429    }
430  }

448  // `(delim_view_tree) => c`, the view built on the tree it is given.
449  pub fn from_ui(
450    d: ViewDialect,
451    c: view::Child,
452    at: SynSpan
453  ): SynExpr ! {macro_error, div, overflow} {
454    let tree := SynIdent {
455      declaration: None,
456      name: "delim_view_tree",
457      ctx: SynCtx { scopes: Nil },
458      span: Syn::call_span()
459    }
460    let body := View::child(d, ExVar(tree), c)
461    ExLam([BName(tree, Some(quote { Tree(HtmlKind, HtmlAttr) }), None)], body, at)
462  }

464  // `@mount f(a: ${x})` of the web elements.
465  pub fn html_mount(t: SynExpr, m: view::Mount): SynExpr ! {macro_error, div, overflow} {
466    let props := match m.mount_args {
467      None => Nil
468      Some(l) => Cons(l.mount_arg, HtmlView::prop_tail(l.mount_arg_tail))
469    }
470    let call := ExApp(ExVar(Syn::ident(m.name)), HtmlView::props(props), m.name.span)
471    match HtmlView::ui_of(t) {
472      Some(u) => quote { view_value(() => $call) }
473      None => quote { view_value_in($t, () => $call) }
474    }
475  }

486  pub fn html_kind(name: SynToken): SynExpr ! {macro_error, div, overflow} {

488      Some(why) => Syn::error(name.span, why)

535  // `@mount f(a: ${x}, b: ${y})` calls `f` once, with each prop a `ReadSignal` that follows
536  // its expression, so the child is not built again when a prop changes. The props are given
537  // in the order they are written; their names say which parameter each one is.
538  fn props(l: List(view::MountArg)): List(SynExpr) ! {macro_error, div, overflow} {
539    match l {
540      Nil => Nil
541      Cons(p, rest) => {
542        let n := p.name.text
543        if Text::len(n) > 1 && Text::slice(
544          n,
545          Text::len(n) - 1,
546          Text::len(n)
547        ) == Some(":") { #() } else {
548          Syn::error(p.name.span, "a prop of @mount is written `name: ${…}`")
549        }
550        let e := p.tauto_expr
551        Cons(quote { readonly(memo(() => $e)) }, HtmlView::props(rest))
552      }
553    }
554  }

557pub macro fn view(v: view::View): SynExpr ! {macro_error, div, overflow} {
558  let d := HtmlView::html()
559  match v.tree_input {
560    Some(input) => View::child(d, input, v.child)
561    // The tree is taken from `ui` once, so the closures the view makes (of `@if`, `@for`,
562    // events) hold the tree and do not perform `ui` themselves.
563    None => {
564      let build := HtmlView::from_ui(d, v.child, v.span)
565      quote { view_in($build) }
566    }
567  }
