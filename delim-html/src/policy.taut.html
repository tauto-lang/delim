3// What the renderers may write, decided on the bytes so that the kernel reads it. Names are
4// lists of byte values, with the name in a comment: the file does not import `std::text`,
5// whose names would meet the prelude's in every file that imports this one: a tag name
6// from a list of elements that run no code, an attribute name from a list of attributes that
7// run none, and a URL whose scheme is one that loads a page or an address. Anything else is
8// refused. The lists name what is allowed, so an element or attribute a browser adds later
9// is refused until it is added here.
10
11pub fn same_bytes(a: List(Nat), b: List(Nat)): Bool {
12  match a {
13    Nil => ⊢ Bool from match b {
14      Nil => true
15      Cons(y, ys) => false
16    }
17    Cons(x, xs) => ⊢ Bool from match b {
18      Nil => false
19      Cons(y, ys) => Bool::and(Nat::eq(x, y), same_bytes(xs, ys))
20    }
21  }
22}
23
24pub fn begins_with(w: List(Nat), p: List(Nat)): Bool {
25  match p {
26    Nil => true
27    Cons(y, ys) => ⊢ Bool from match w {
28      Nil => false
29      Cons(x, xs) => Bool::and(Nat::eq(x, y), begins_with(xs, ys))
30    }
31  }
32}
33
34fn one_of(w: List(Nat), l: List(List(Nat))): Bool {
35  match l {
36    Nil => false
37    Cons(t, rest) => Bool::or(same_bytes(w, t), one_of(w, rest))
38  }
39}

41fn has_byte(w: List(Nat), c: Nat): Bool {
42  match w {
43    Nil => false
44    Cons(x, rest) => Bool::or(Nat::eq(x, c), has_byte(rest, c))
45  }
46}

48pub fn is_lower(c: Nat): Bool { Bool::and(Nat::le(97, c), Nat::le(c, 122)) }

50pub fn is_digit_byte(c: Nat): Bool { Bool::and(Nat::le(48, c), Nat::le(c, 57)) }

52// The bytes after the first of a name: lowercase letters, digits and `-`.
53fn name_rest_ok(w: List(Nat)): Bool {
54  match w {
55    Nil => true
56    Cons(c, rest) => Bool::and(
57      Bool::or(is_lower(c), Bool::or(is_digit_byte(c), Nat::eq(c, 45))),
58      name_rest_ok(rest)
59    )
60  }
61}

63// A name of a tag or an attribute: a lowercase letter, then lowercase letters, digits and `-`.
64fn name_ok(w: List(Nat)): Bool {
65  match w {
66    Nil => false
67    Cons(c, rest) => Bool::and(is_lower(c), name_rest_ok(rest))
68  }

315pub module Attr {
316  // Whether the attribute `w` may be written: one of `attrs`, or a `data-…` or `aria-…`
317  // attribute.
318  pub fn allowed(w: List(Nat)): Bool {
319    Bool::and(
320      name_ok(w),
321      Bool::or(
322        one_of(w, attrs()),
323        Bool::or(begins_with(w, data_prefix()), begins_with(w, aria_prefix()))
324      )
325    )
326  }
327
328  pub fn is_url(w: List(Nat)): Bool { one_of(w, url_attrs()) }
329
330  // ---- URLs -------------------------------------------------------------------------------------
331
332  // Whether `name="value"` may be written. `style` is written only as `display:contents`, which
333  // the renderers give the elements that hold a group, an island or trusted HTML.
334  pub fn value_allowed(name: List(Nat), value: List(Nat)): Bool {
335    Bool::or(
336      Bool::and(
337        Attr::allowed(name),
338        ⊢ Bool from match Attr::is_url(name) {
339          true => url_allowed(value)
340          false => true
341        }
342      ),
343      Bool::and(same_bytes(name, style_name()), same_bytes(value, display_contents()))
344    )
345  }
346}

360fn ascii_lower(c: Nat): Nat {
361  match Bool::and(Nat::le(65, c), Nat::le(c, 90)) {
362    true => Nat::add(c, 32)
363    false => c
364  }
365}

367// The bytes of a URL before its first `/`, `?` or `#`, lowercased, without the spaces and
368// control bytes a browser skips (it skips fewer, so what it reads as a scheme is read here
369// too).
370fn url_head(w: List(Nat)): List(Nat) {
371  match w {
372    Nil => Nil
373    Cons(c, rest) => ⊢ List(Nat) from match Bool::or(
374      Nat::eq(c, 47),
375      Bool::or(Nat::eq(c, 63), Nat::eq(c, 35))
376    ) {
377      true => Nil
378      false => ⊢ List(Nat) from match Nat::le(c, 32) {
379        true => url_head(rest)
380        false => Cons(ascii_lower(c), url_head(rest))
381      }
382    }
383  }
384}

386// The bytes before the first `:`.
387fn before_colon(w: List(Nat)): List(Nat) {
388  match w {
389    Nil => Nil
390    Cons(c, rest) => ⊢ List(Nat) from match Nat::eq(c, 58) {
391      true => Nil
392      false => Cons(c, before_colon(rest))
393    }
394  }
395}

406// `&`, `<`, `>` and `"`: a character reference could spell a `:` or a scheme, and escaping
407// the others would add an `&`.
408fn has_markup(h: List(Nat)): Bool {
409  Bool::or(Bool::or(has_byte(h, 38), has_byte(h, 60)), Bool::or(has_byte(h, 62), has_byte(h, 34)))
410}
411
412fn head_allowed(h: List(Nat)): Bool {
413  match has_markup(h) {
414    true => false
415    false => ⊢ Bool from match has_byte(h, 58) {
416      true => one_of(before_colon(h), schemes())
417      // No scheme: a path relative to the page.
418      false => true
419    }
420  }
421}

424// `https`, `mailto` or `tel`; and no `&`, `<`, `>` or `"` before its first `/`, `?` or `#`.
