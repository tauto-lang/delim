6// What the DOM renderer and the HTML one share: element kinds are tag names and properties
7// are attributes.
8
9pub(open) enum HtmlKind {
10  Tag(Text)
11  // An island named `name`, with the text its view starts from and its first render as
12  // HTML: the browser hydrates it alone, the rest of the page stays as the server wrote it.
13  Island(Text, Text, TrustedHtml)
14  // HTML written as it is, by `trusted`.
15  Trusted(TrustedHtml)
16  // An SVG element (`svg.taut` lists them), made in the SVG namespace by the DOM renderer.
17  // The HTML renderer writes it as a refused element: `html_safe` has no SVG.
18  Svg(Text)
19}

21// HTML written into a page as it is: a `TrustedHtml` (`safe.taut`), whose bytes come with a
22// proof that `html_safe` accepts them. Every other text a view shows is escaped.
23
24// Text as HTML that shows it: `&`, `<`, `>` and `"` escaped.
25pub fn escaped_html(t: Text): TrustedHtml ! {div, overflow} { escaped_bytes(bytes_of(t)) }
26
27// `html` as trusted HTML, when `html_safe` accepts it: the output of a renderer such as a
28// Markdown or highlighting one, checked as it is written into the page. `None` when it has
29// anything `html_safe` refuses, such as a script, a handler, or a URL that runs code.
30pub fn checked_html(html: Text): Option(TrustedHtml) ! {div, overflow} {
31  checked_bytes(bytes_of(html))
32}
33
34// The HTML as text.
35pub fn trusted_text(h: TrustedHtml): Text ! {div, overflow} { text_of_byte_list(trusted_bytes(h)) }

37pub(open) enum HtmlAttr {
38  Attr(Text, Text)
39  // A role for assistive technology and the element's name; an empty name sets none.
40  Aria(Text, Text)
41  // The live DOM value; HTML uses textarea text, select option selection, or an attribute.
42  Value(Text)
43  Disabled(Bool)
44  Checked(Bool)
45  Selected(Bool)
46  Multiple(Bool)
47  Hidden(Bool)
48  Required(Bool)
49  ReadOnly(Bool)
50  // None removes the attribute; names and present values still pass the policy.
51  OptionalAttr(Text, Option(Text))
52  // An attribute of an SVG element, from the list in `svg.taut`.
53  SvgAttr(Text, Text)
54  // A CSS custom property (`--name`) of the element, for its style sheet to read with
55  // `var(--name)`: how a view places and sizes an element. Only the DOM renderer sets it.
56  Var(Text, Css)
57  // Whether a press on the element (or inside it) that moves becomes a drag: the element then
58  // holds the pointer and gets `DragStart`, `DragMove` and `DragEnd`, and the node it is let
59  // go over gets `Drop`. Only the DOM renderer has drags.
60  DragSource(Bool)
61}

63pub fn web_vocab(): Vocab(HtmlKind, HtmlAttr) {
64  Vocab {
65    column: Tag("div"),
66    row: Tag("div"),
67    pressable: Tag("button"),
68    input: Tag("input"),
69    role: (r: Role) => match r {
70      Button(l) => Aria("button", l)
71      Link(l) => Aria("link", l)
72      Checkbox(l) => Aria("checkbox", l)
73      TextField(l) => Aria("textbox", l)
74      Tab(l) => Aria("tab", l)
75      MenuItem(l) => Aria("menuitem", l)
76      Heading(level) => Aria("heading", "")
77      ListBox => Aria("listbox", "")
78      ListEntry => Aria("option", "")
79      Region(l) => Aria("region", l)
80      Presentation => Aria("presentation", "")
81    },
82    name: (l: Text) => Attr("aria-label", l),
83    disabled: (b: Bool) => Disabled(b),
84    value: (v: Text) => Value(v)
85  }

88fn digit(d: u64): Text ! {overflow} {
89  match Text::slice("0123456789", d, d + 1) {
90    Some(x) => x
91    None => "0"
92  }
93}

95pub fn decimal(n: u64): Text ! {div, overflow} {
96  if n < 10 { digit(n) } else { concat(decimal(div(n, 10)), digit(mod(n, 10))) }
97}

99// ---- what may be written --------------------------------------------------------------------
100// Both renderers write only what `policy.taut` allows: a tag from its list of elements, an
101// attribute from its list of attributes, and a URL relative to the page or with the scheme
102// `http`, `https`, `mailto` or `tel`.
103
104pub fn holds(y: Bool): Bool {
105  match y { true => true, false => false }
106}

108// The bytes of `t`, as `policy.taut` reads them.
109pub fn bytes_of(t: Text): List(Nat) ! {div, overflow} { bytes_before(t, Text::len(t), Nil) }
110
111fn bytes_before(t: Text, i: u64, acc: List(Nat)): List(Nat) ! {div, overflow} {
112  if i == 0 { acc } else {
113    match Text::byte(t, i - 1) {
114      Some(b) => bytes_before(t, i - 1, Cons(Nat::of_u64(U64::of_u8(b)), acc))
115      None => acc
116    }
117  }
118}

120// Keep byte conversion tail recursive: a rendered document can have many more
121// bytes than the host stack can hold, even when its DOM tree is shallow.
122fn reverse_u8s(w: List(u8), acc: List(u8)): List(u8) ! {div, overflow} {
123  match w {
124    Nil => acc
125    Cons(c, rest) => reverse_u8s(rest, Cons(c, acc))
126  }
127}
128fn u8s_before(w: List(Nat), acc: List(u8)): List(u8) ! {div, overflow} {
129  match w {
130    Nil => reverse_u8s(acc, Nil)
131    Cons(c, rest) => u8s_before(rest, Cons(U8::of_u64(U64::of_nat(c)), acc))
132  }
133}
134fn u8s_of(w: List(Nat)): List(u8) ! {div, overflow} { u8s_before(w, Nil) }

136// The text of bytes that came from a text: "" if they are not UTF-8.
137pub fn text_of_byte_list(w: List(Nat)): Text ! {div, overflow} {
138  match Text::of_utf8(u8s_of(w)) {
139    Some(t) => t
140    None => ""
141  }
142}

148pub module Attr {
149  fn ok(name: Text, value: Text): Bool ! {div, overflow} {
150    holds(Attr::value_allowed(bytes_of(name), bytes_of(value)))
151  }
152
153  pub fn refusal(name: Text): Text ! {div, overflow} {
154    if starts_with(name, "on") { "a handler is an `on:` event, not an `on…` attribute" } else {
155      concat(
156        concat("`", name),
157        "` is not an attribute delim writes: it may run code, or it is not one of HTML"
158      )
159    }
160  }
161
162  // Why the attribute `name` may not be written, if it may not.
163  fn name_refused(name: Text): Option(Text) ! {div, overflow} {
164    if holds(Attr::allowed(bytes_of(name))) { None } else { Some(Attr::refusal(name)) }
165  }
166
167  // Why `name="value"` may not be written, if it may not.
168  pub fn refused(name: Text, value: Text): Option(Text) ! {div, overflow} {
169    if Attr::ok(name, value) { None } else {
170      match Attr::name_refused(name) {
171        Some(why) => Some(why)
172        None => Some(url_refusal())
173      }
174    }
175  }
176}
177
178pub fn starts_with(t: Text, prefix: Text): Bool ! {div, overflow} {
179  match Text::slice(t, 0, Text::len(prefix)) {
180    Some(head) => head == prefix
181    None => false
182  }
183}

185// What the renderers and `view!` say about a refused tag, attribute and URL.
186pub fn tag_refusal(tag: Text): Text {
187  concat(
188    concat("`", tag),
189    "` is not an element delim writes: it may run code, or it is not one of HTML"
190  )
191}
192
193pub fn url_refusal(): Text {
194  "a URL is written only relative to the page or with `http:`, `https:`, `mailto:` or `tel:`, and with no `&`, `<`, `>` or `\"` before its path"
195}
196
197// Why the tag `tag` may not be written, if it may not.
198pub fn tag_refused(tag: Text): Option(Text) ! {div, overflow} {
199  if tag_ok(tag) { None } else { Some(tag_refusal(tag)) }
200}

202// The tag written for `tag`: itself, or, when it is refused, an element that shows nothing.
203pub fn written_tag(tag: Text): Text ! {div, overflow} {
204  if tag_ok(tag) { tag } else { "delim-refused" }
205}

208pub fn url_ok(value: Text): Bool ! {div, overflow} { holds(url_allowed(bytes_of(value))) }
