23// What a component of the web elements performs while it builds: the tree, keeping its parts
24// for later, and the signals. A component that reads a context names it too:
25// `{Delim, context(T)}`.
26pub effect Delim { ui(HtmlKind, HtmlAttr), keeping, rt, div }

28pub fn current_tree(): Tree(HtmlKind, HtmlAttr) ! {ui(HtmlKind, HtmlAttr)} { perform ui() }

30// `body` on the tree `t`, at the root: what it keeps runs later with the contexts and the tree
31// that stood where it was kept.
32pub fn with_ui<A>(
33  t: Tree(HtmlKind, HtmlAttr),
34  vocabulary: Vocab(HtmlKind, HtmlAttr),
35  body: () -> A ! {Delim}
36): A ! {rt, div} {
37  match rooted(() => with_tree(t, body)) {
38    Some(a) => a
39    // The build itself returns through `rooted`; only a kept part run later ends otherwise.
40    None => with_ui(t, vocabulary, body)
41  }
42}

44// The handler `h` of an event, kept so that it runs with what stood where it was made.
45pub fn on_event<P: Type, !r>(
46  kind: EventKind,
47  h: (d: EventData) -> Unit ! {rt, div} ∪ r
48): Prop(P) ! {ui(HtmlKind, HtmlAttr), keeping, rt, div} ∪ r {
49  let t := current_tree()
50  let run := keep_run(() => h(current_event(t)))
51  Held(kind, hold(run))
52}

120// Where a `suspense` below an `error_boundary` reports a failure that happens when it builds its
121// body again, after the `error_boundary` has returned: the boundary shows `failed(why)` for it.
122pub struct(shared) Failure(E: Type) { cell: Signal(Option(E)) }
123
124// `body()`, or `failed(why)` if `body` fails with `fail(why)`: before it returns, or later, when a
125// `suspense` in it builds its body again and that fails. A `suspense` needs one above it.
126pub fn error_boundary<E: Shared + NoBorrow, !e>(
127  b: {
128    failed: (why: E) -> Node ! {Delim} ∪ e,
129    body: () -> Node ! {Delim, context(Failure(E)), fail(E)} ∪ e
130  }
131): Node ! {Delim} ∪ e {
132  let t := current_tree()
133  let failure: Signal(Option(E)) := signal(None)
134  let build := keep((f: Option(E)) => match f {
135    Some(why) => (b.failed)(why)
136    None => match provide(Failure { cell: failure }, () => handle { Ok((b.body)()) } with {
137      fail<E>(why) => Err(why)
138      return n => n
139    }) {
140      Ok(n) => n
141      Err(why) => (b.failed)(why)
142    }
143  })
144  switch_on_change(t, () => get(failure), build)
145}

147// `body()` once every resource it reads is settled. While one is pending, `pending()` is shown
148// in its place, and `body` is built again when that resource settles; a cancelled one shows
149// `cancelled()`. The boundary owns only what it shows: waiting here never cancels a resource.
150// A failure of `body` goes to the `error_boundary` above, which shows its `failed`.
151enum Tried { Shown(Node), Waiting, Stopped }
152
153pub fn suspense<A: Shared + NoBorrow, E: Shared + NoBorrow, !e>(
154  s: {
155    pending: () -> Node ! {Delim} ∪ e,
156    cancelled: () -> Node ! {Delim} ∪ e,
157    body: () -> Node ! {Delim, resource_wait(A, E), resource_cancelled, fail(E)} ∪ e
158  }
159): Node ! {Delim, context(Failure(E))} ∪ e {
160  let t := current_tree()
161  let reported: Failure(E) := use_context()
162  let attempt := signal(0)
163  let build := keep((n: u64) => {
164    // The arms only note what happened: what is shown instead is built after the `handle`,
165    // where no continuation of `body` is held while it keeps parts of its own.
166    let tried := handle { Shown((s.body)()) } with {
167      resource_wait<A, E>(resource), k => {
168        watch(() => match state(resource) {
169          Pending => #()
170          Ready(_) => set(attempt, n + 1)
171          Failed(_) => set(attempt, n + 1)
172          Cancelled => set(attempt, n + 1)
173        })
174        Waiting
175      }
176      resource_cancelled() => Stopped
177      fail<E>(why) => {
178        set(reported.cell, Some(why))
179        // The `error_boundary` builds again and drops this; nothing is shown meanwhile.
180        let g := node_of(t)
181        emit(t, CreateGroup(g))
182        Shown(g)
183      }
184      return shown => shown
185    }
186    match tried {
187      Shown(node) => node
188      Waiting => (s.pending)()
189      Stopped => (s.cancelled)()
190    }
191  })
192  switch_on(t, () => get(attempt), build)
193}
