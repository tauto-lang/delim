5// The HTML renderer's output, as the kernel reads it: a tree of elements, texts, numbered
6// marks and trusted HTML, written as bytes. `write_safe` proves that whatever the tree holds,
7// `html_safe` accepts what is written: texts and attribute values are escaped, trusted HTML
8// carries its own proof, and each tag and mark is written only when `html_safe` accepts it on
9// its own. The renderer gives only tags and attributes the policy allows, so in practice none
10// is left out.
11
12pub(open) inductive HTree: Type {
13  // A text, escaped when written.
14  HText(bytes: List(Nat)): HTree
15  // A comment with a node's number, which the DOM renderer hydrates by.
16  HMark(digits: List(Nat)): HTree
17  // An element with attributes (name, value) and children. An element `tag_void` names has
18  // no end tag, and its children are not written.
19  HEl(tag: List(Nat), attrs: List(@name List(Nat) × @value List(Nat)), children: List(HTree)): HTree
20  // An element whose content is trusted HTML, written as it is.
21  HRaw(tag: List(Nat), attrs: List(@name List(Nat) × @value List(Nat)), html: TrustedHtml): HTree
22}

24// Reverse onto a suffix with constant host stack usage. HTML byte lists are
25// often much longer than the DOM tree is deep.
26fn reverse_bytes(xs: List(Nat), suffix: List(Nat)): List(Nat) {
27  match xs {
28    Nil => suffix
29    Cons(x, rest) => reverse_bytes(rest, Cons(x, suffix))
30  }
31}
32fn append_bytes(xs: List(Nat), suffix: List(Nat)): List(Nat) {
33  reverse_bytes(reverse_bytes(xs, Nil), suffix)
34}
35theorem reverse_bytes_twice(
36  xs: List(Nat),
37  acc: List(Nat),
38  suffix: List(Nat)
39): reverse_bytes(reverse_bytes(xs, acc), suffix) = reverse_bytes(acc, List::append(xs, suffix)) {
40  match xs {
41    Nil => Refl
42    Cons(x, rest) => reverse_bytes_twice(rest, Cons(x, acc), suffix)
43  }
44}
45theorem append_bytes_eq(
46  xs: List(Nat),
47  suffix: List(Nat)
48): append_bytes(xs, suffix) = List::append(xs, suffix) {
49  reverse_bytes_twice(xs, Nil, suffix)
50}

54fn guard_as(y: Bool, w: List(Nat)): List(Nat) {
55  match y { true => w, false => Nil }

58// `w` when `html_safe` accepts it on its own; nothing otherwise.
59fn guard(w: List(Nat)): List(Nat) { guard_as(html_safe(w), w) }

61fn value_as(empty: Bool, escaped: List(Nat)): List(Nat) {
62  match empty {
63    true => Nil
64    false => Cons(61, Cons(34, append_bytes(escaped, [34])))
65  }
66}

68fn is_nil(w: List(Nat)): Bool {
69  match w {
70    Nil => true
71    Cons(c, rest) => false
72  }
73}

75// ` name="value"`, or ` name` for an empty value, when the policy allows it.
76fn attr_as(y: Bool, name: List(Nat), escaped: List(Nat)): List(Nat) {
77  match y {
78    true => Cons(32, append_bytes(name, value_as(is_nil(escaped), escaped)))
79    false => Nil
80  }
81}

83pub module Attr {
84  pub fn bytes(name: List(Nat), escaped: List(Nat)): List(Nat) {
85    attr_as(Attr::value_allowed(name, escaped), name, escaped)
86  }
87}
88
89fn attrs_bytes(l: List(@name List(Nat) × @value List(Nat))): List(Nat) {
90  match l {
91    Nil => Nil
92    Cons(a, rest) => append_bytes(
93      Attr::bytes((a: @name List(Nat)), Escape::html((a: @value List(Nat)))),
94      attrs_bytes(rest)
95    )
96  }
97}

99fn start_tag(tag: List(Nat), attrs: List(@name List(Nat) × @value List(Nat))): List(Nat) {
100  guard(Cons(60, append_bytes(tag, append_bytes(attrs_bytes(attrs), [62]))))
101}
102
103fn end_tag(tag: List(Nat)): List(Nat) { guard(Cons(60, Cons(47, append_bytes(tag, [62])))) }
104
105fn element_as(void: Bool, start: List(Nat), inner: List(Nat), end: List(Nat)): List(Nat) {
106  match void {
107    true => start
108    false => append_bytes(start, append_bytes(inner, end))
109  }
110}
111
112fn element(
113  tag: List(Nat),
114  attrs: List(@name List(Nat) × @value List(Nat)),
115  inner: List(Nat)
116): List(Nat) {
117  element_as(tag_void(tag), start_tag(tag, attrs), inner, end_tag(tag))
118}
119
120// `<!--digits-->`.
121fn mark_bytes(digits: List(Nat)): List(Nat) {
122  guard(Cons(60, Cons(33, Cons(45, Cons(45, append_bytes(digits, [45, 45, 62]))))))
123}

125mutual {
126  // The tree as HTML.
127  pub fn write(t: HTree): List(Nat) {
128    match t {
129      HText(b) => Escape::html(b)
130      HMark(d) => mark_bytes(d)
131      HEl(tag, attrs, cs) => element(tag, attrs, write_all(cs))
132      HRaw(tag, attrs, h) => element(tag, attrs, trusted_bytes(h))
133    }
134  }
135
136  pub fn write_all(ts: List(HTree)): List(Nat) {
137    match ts {
138      Nil => Nil
139      Cons(t, rest) => append_bytes(write(t), write_all(rest))
140    }
141  }
142}

144// ---- what is proved ---------------------------------------------------------------------------
145
146// Bytes that, read from text, end in text.
147fn keeps_text(w: List(Nat)): std::prelude::Prop { scan(SText, w) = SText }
148
149theorem keeps_text_append(
150  a: List(Nat),
151  b: List(Nat),
152  ha: keeps_text(a),
153  hb: keeps_text(b)
154): keeps_text(append_bytes(a, b)) {
155  Eq::trans(
156    Eq::congr((w: List(Nat)) => scan(SText, w), append_bytes_eq(a, b)),
157    Eq::trans(Scan::append(SText, a, b), Eq::trans(Eq::congr((s: Scan) => scan(s, b), ha), hb))
158  )
159}
160
161// Where `Scan::done` says yes, the reading is in text.
162theorem done_text(s: Scan, h: Scan::done(s) = true): s = SText {
163  match s {
164    SText => Refl
165    SLt => False::elim(false_ne_true(h))
166    SOpen(name) => False::elim(false_ne_true(h))
167    SClose(name) => False::elim(false_ne_true(h))
168    STag => False::elim(false_ne_true(h))
169    SSpace => False::elim(false_ne_true(h))
170    SAttr(name) => False::elim(false_ne_true(h))
171    SEq(name) => False::elim(false_ne_true(h))
172    SValue(name, value) => False::elim(false_ne_true(h))
173    SBang => False::elim(false_ne_true(h))
174    SDash => False::elim(false_ne_true(h))
175    SDigits(any) => False::elim(false_ne_true(h))
176    SEnd1 => False::elim(false_ne_true(h))
177    SEnd2 => False::elim(false_ne_true(h))
178    SFail => False::elim(false_ne_true(h))
179  }
180}

182// What `html_safe` accepts keeps text.
183theorem safe_keeps_text(w: List(Nat), h: html_safe(w) = true): keeps_text(w) {
184  done_text(scan(SText, w), h)
185}

187theorem guard_as_keeps_text(
188  y: Bool,
189  w: List(Nat),
190  e: html_safe(w) = y
191): keeps_text(guard_as(y, w)) {
192  match y {
193    true => safe_keeps_text(w, e)
194    false => Refl
195  }
196}

198theorem guard_keeps_text(w: List(Nat)): keeps_text(guard(w)) {
199  guard_as_keeps_text(html_safe(w), w, Refl)
200}

202theorem element_as_keeps_text(
203  void: Bool,
204  start: List(Nat),
205  inner: List(Nat),
206  end: List(Nat),
207  hs: keeps_text(start),
208  hi: keeps_text(inner),
209  he: keeps_text(end)
210): keeps_text(element_as(void, start, inner, end)) {
211  match void {
212    true => hs
213    false => keeps_text_append(
214      start,
215      append_bytes(inner, end),
216      hs,
217      keeps_text_append(inner, end, hi, he)
218    )
219  }
220}
221
222theorem element_keeps_text(
223  tag: List(Nat),
224  attrs: List(@name List(Nat) × @value List(Nat)),
225  inner: List(Nat),
226  hi: keeps_text(inner)
227): keeps_text(element(tag, attrs, inner)) {
228  element_as_keeps_text(
229    tag_void(tag),
230    start_tag(tag, attrs),
231    inner,
232    end_tag(tag),
233    guard_keeps_text(Cons(60, append_bytes(tag, append_bytes(attrs_bytes(attrs), [62])))),
234    hi,
235    guard_keeps_text(Cons(60, Cons(47, append_bytes(tag, [62]))))
236  )
237}
238
239theorem trusted_keeps_text(h: TrustedHtml): keeps_text(trusted_bytes(h)) {
240  match h {
241    Mk(b, ok) => safe_keeps_text(b, ok)
242  }
243}

245mutual {
246  theorem write_keeps_text(t: HTree): keeps_text(write(t)) {
247    match t {
248      HText(b) => Escape::html_scan(b)
249      HMark(d) => guard_keeps_text(
250        Cons(60, Cons(33, Cons(45, Cons(45, append_bytes(d, [45, 45, 62])))))
251      )
252      HEl(tag, attrs, cs) => element_keeps_text(tag, attrs, write_all(cs), write_all_keeps_text(cs))
253      HRaw(tag, attrs, h) => element_keeps_text(tag, attrs, trusted_bytes(h), trusted_keeps_text(h))
254    }
255  }
256
257  theorem write_all_keeps_text(ts: List(HTree)): keeps_text(write_all(ts)) {
258    match ts {
259      Nil => Refl
260      Cons(t, rest) => keeps_text_append(
261        write(t),
262        write_all(rest),
263        write_keeps_text(t),
264        write_all_keeps_text(rest)
265      )
266    }
267  }
268}

270// Whatever the tree holds, `html_safe` accepts what is written.
271pub theorem write_safe(t: HTree): html_safe(write(t)) = true {
272  Eq::congr((s: Scan) => Scan::done(s), write_keeps_text(t))
273}
274
275// The tree as trusted HTML.
276fn written(t: HTree): TrustedHtml { TrustedHtml::Mk(write(t), write_safe(t)) }
277
278pub theorem write_all_safe(ts: List(HTree)): html_safe(write_all(ts)) = true {
279  Eq::congr((s: Scan) => Scan::done(s), write_all_keeps_text(ts))
280}
281
282// The trees one after another as trusted HTML.
283pub fn written_all(
284  ts: List(HTree)
285): TrustedHtml { TrustedHtml::Mk(write_all(ts), write_all_safe(ts)) }
