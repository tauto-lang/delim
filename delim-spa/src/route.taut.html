32syntax!(route_syntax, ```par
33%start Decl
34%title "route"
35%%
36Decl: Name Path { Parameter };
37Path: Literal { PathPart };
38PathPart: Literal | Capture;
39Capture: LBrace Name Colon Name RBrace;
40Parameter: Separator Name Equals LBrace Name Colon Name [ Default ] RBrace;
41Separator: Question | Amp;
42Default: Equals DefaultValue;
43DefaultValue: Name | TautoExpr;
44Question: '?';
45Amp: '&';
46Equals: '=';
47LBrace: '{';
48RBrace: '}';
49Colon: ':';
50Literal: /\/[A-Za-z0-9_~.\/-]*/;

54module RouteMacro {
55  fn local_ident(
56    name: Text,
57    at: SynSpan
58  ): SynIdent { SynIdent { declaration: None, name: name, ctx: SynCtx { scopes: Nil }, span: at } }
59  struct Field {
60    key: SynToken,
61    name: SynToken,
62    ty: SynToken,
63    fallback: Option(SynExpr),
64    segment: Bool
65  }
66  struct PathPiece { literal: Option(Text), field: Option(Field) }
67
68  fn contains(values: List(Text), value: Text): Bool ! {div} {
69    match values { Nil => false
70      Cons(item, rest) => item == value || contains(rest, value) }
71  }
72  fn scoped(value: Text, index: u64): Bool ! {overflow, div} {
73    match Text::byte(value, index) { None => false
74      Some(byte) => U64::of_u8(byte) == 58 || scoped(value, index + 1) }
75  }
76  fn ends_with_slash(value: Text): Bool ! {overflow, div} {
77    match Text::byte(value, Text::len(value) - 1) { None => false
78      Some(byte) => U64::of_u8(byte) == 47 }
79  }
80  // Each capture fills one whole segment: the text before it ends with `/`, and the next piece starts with `/`.
81  fn path_pieces(
82    items: List(route_syntax::PathPart),
83    names: List(Text),
84    after_slash: Bool
85  ): List(PathPiece) ! {overflow, macro_error, div} {
86    match items {
87      Nil => Nil
88      Cons(item, rest) => match item {
89        route_syntax::PathPart::Literal(token) =>
90        Cons(
91          PathPiece { literal: Some(token.text), field: None },
92          path_pieces(rest, names, ends_with_slash(token.text))
93        )
94        route_syntax::PathPart::Capture(capture) => {
95          if not(after_slash) {
96            Syn::error(capture.span, "route: a path parameter must fill a whole segment after /")
97          } else { #() }
98          if scoped(capture.name.text, 0) {
99            Syn::error(capture.name.span, "route: field names must be unqualified identifiers")
100          } else { #() }
101          if contains(
102            names,
103            capture.name.text
104          ) { Syn::error(capture.name.span, "route: duplicate field name") } else { #() }
105          Cons(
106            PathPiece {
107              literal: None,
108              field: Some(
109                Field {
110                  key: capture.name,
111                  name: capture.name,
112                  ty: capture.name_2,
113                  fallback: None,
114                  segment: true
115                }
116              )
117            },
118            path_pieces(rest, Cons(capture.name.text, names), false)
119          )
120        }
121      }
122    }
123  }
124  fn piece_fields(items: List(PathPiece)): List(Field) ! {div} {
125    match items { Nil => Nil
126      Cons(item, rest) => match item.field { None => piece_fields(rest)
127        Some(field) => Cons(field, piece_fields(rest)) } }
128  }
129  fn field_names_of(items: List(Field)): List(Text) ! {div} {
130    match items { Nil => Nil
131      Cons(item, rest) => Cons(item.name.text, field_names_of(rest)) }
132  }
133  fn fields(
134    items: List(route_syntax::Parameter),
135    keys: List(Text),
136    names: List(Text),
137    first: Bool
138  ): List(Field) ! {overflow, macro_error, div} {
139    match items {
140      Nil => Nil
141      Cons(item, rest) => {
142        let valid_separator := match item.separator {
143          route_syntax::Separator::Question(_) => first
144          route_syntax::Separator::Amp(_) => not(first)
145        }
146        if not(valid_separator) {
147          Syn::error(item.span, "route: query parameters start with ? and continue with &")
148        } else { #() }
149        if scoped(item.name.text, 0) || scoped(item.name_2.text, 0) {
150          Syn::error(item.span, "route: query keys and field names must be unqualified identifiers")
151        } else { #() }
152        if contains(
153          keys,
154          item.name.text
155        ) { Syn::error(item.name.span, "route: duplicate query key") } else { #() }
156        if contains(
157          names,
158          item.name_2.text
159        ) { Syn::error(item.name_2.span, "route: duplicate field name") } else { #() }
160        let fallback := match item.default {
161          None => None
162          Some(value) => Some(match value.default_value {
163            route_syntax::DefaultValue::Name(token) => ExVar(Syn::ident(token))
164            route_syntax::DefaultValue::TautoExpr(expression) => expression
165          })
166        }
167        Cons(
168          Field {
169            key: item.name,
170            name: item.name_2,
171            ty: item.name_3,
172            fallback: fallback,
173            segment: false
174          },
175          fields(rest, Cons(item.name.text, keys), Cons(item.name_2.text, names), false)
176        )
177      }
178    }
179  }
180  fn global(name: Text): SynExpr { ExVar(Syn::global(name)) }
181  fn apply(name: Text, args: List(SynExpr), at: SynSpan): SynExpr { ExApp(global(name), args, at) }
182  fn named(
183    name: SynIdent,
184    args: List(SynExpr),
185    at: SynSpan
186  ): SynExpr { ExApp(ExVar(name), args, at) }
187  fn method(
188    owner: SynIdent,
189    suffix: Text
190  ): SynIdent { Syn::decl_name(owner, concat(owner.name, concat("::", suffix))) }
191  fn codec(ty: SynToken): SynExpr {
192    if ty.text == "Text" { apply("text_codec", Nil, ty.span) } else if ty.text == "u64" {
193      apply("u64_codec", Nil, ty.span)
194    } else if ty.text == "Bool" { apply("route_bool_codec", Nil, ty.span) } else {
195      named(Syn::ident_like(Syn::ident(ty), concat(ty.text, "::route_codec")), Nil, ty.span)
196    }
197  }
198  fn data_fields(items: List(Field)): List(Option(SynIdent) × SynExpr) ! {div} {
199    match items { Nil => Nil
200      Cons(item, rest) => Cons(
201        #(Some(Syn::ident(item.name)), ExVar(Syn::ident(item.ty))),
202        data_fields(rest)
203      ) }
204  }
205  fn bindings(items: List(Field)): List(SynExpr × Text) ! {div} {
206    match items { Nil => Nil
207      Cons(item, rest) => Cons(
208        #(
209          ExVar(local_ident(concat("_route_field_", item.name.text), item.name.span)),
210          item.name.text
211        ),
212        bindings(rest)
213      ) }
214  }
215  fn result_type(owner: SynIdent, at: SynSpan): SynExpr {
216    apply("Result", [ExVar(owner), global("RouteError")], at)
217  }

223    name: Text,

227  fn read_fields(
228    owner: SynIdent,
229    items: List(Field),
230    all: List(Field),
231    matched: SynIdent,
232    at: SynSpan
233  ): SynExpr ! {div} {
234    match items {
235      Nil => apply("Ok", [ExRecord(owner, bindings(all), at)], at)
236      Cons(item, rest) => {
237        let local := local_ident(concat("_route_field_", item.name.text), item.name.span)
238        let error := local_ident("_route_error", at)
239        let decoded := local_ident("_route_decoded", at)
240        let values := ExField(ExVar(matched), if item.segment { "segments" } else { "query" }, at)
241        let args := [codec(item.ty), values, ExText(item.key.text, item.key.span)]
242        let read := match item.fallback {
243          None => apply(if item.segment { "route_read_segment" } else { "route_read" }, args, at)
244          Some(value) => apply("route_read_or", Syn::append(args, [value]), at)
245        }
246        let ty := apply("Result", [ExVar(Syn::ident(item.ty)), global("RouteError")], at)
247        ExBlock(
248          [
249            StLet(decoded, Some(ty), read, false, false, None, at),
250            StExpr(
251              ExMatch(
252                ExVar(decoded),
253                Some(result_type(owner, at)),
254                [
255                  arm(ctor("Err", error), apply("Err", [ExVar(error)], at)),
256                  arm(ctor("Ok", local), read_fields(owner, rest, all, matched, at))
257                ],
258                None,
259                None,
260                at
261              )
262            )
263          ],
264          at
265        )
266      }
267    }
268  }
269  fn part_list(pieces: List(PathPiece), at: SynSpan): SynExpr ! {div} {
270    match pieces {
271      Nil => global("Nil")
272      Cons(piece, rest) => {
273        let part := match piece.field {
274          Some(field) => apply("RouteCapture", [ExText(field.name.text, field.name.span)], at)
275          None => apply("RouteLiteral", [ExText(match piece.literal { Some(text) => text
276            None => "" }, at)], at)
277        }
278        apply("Cons", [part, part_list(rest, at)], at)
279      }
280    }
281  }
282  fn parse_body(
283    owner: SynIdent,
284    pieces: List(PathPiece),
285    items: List(Field),
286    target: SynIdent,
287    at: SynSpan
288  ): SynExpr ! {div} {
289    let matched := local_ident("_route_match", at)
290    let error := local_ident("_route_error", at)
291    ExMatch(
292      apply("route_match", [ExVar(target), part_list(pieces, at)], at),
293      Some(result_type(owner, at)),
294      [
295        arm(ctor("Err", error), apply("Err", [ExVar(error)], at)),
296        arm(ctor("Ok", matched), read_fields(owner, items, items, matched, at))
297      ],
298      None,
299      None,
300      at
301    )
302  }
303  fn href_fields(items: List(Field), value: SynIdent, first: Bool, at: SynSpan): SynExpr ! {div} {
304    match items {
305      Nil => ExText("", at)
306      Cons(item, rest) => {
307        let encoded := encode_field(item, value, at)
308        let pair := apply("route_query_item", [ExText(item.key.text, item.key.span), encoded], at)
309        apply(
310          "concat",
311          [
312            ExText(if first { "?" } else { "&" }, at),
313            apply("concat", [pair, href_fields(rest, value, false, at)], at)
314          ],
315          at
316        )
317      }
318    }
319  }
320  fn encode_field(item: Field, value: SynIdent, at: SynSpan): SynExpr {
321    let encoder := ExField(codec(item.ty), "encode", at)
322    ExApp(encoder, [ExField(ExVar(value), item.name.text, item.name.span)], at)
323  }
324  fn href_path(
325    pieces: List(PathPiece),
326    value: SynIdent,
327    tail: SynExpr,
328    at: SynSpan
329  ): SynExpr ! {div} {
330    match pieces {
331      Nil => tail
332      Cons(piece, rest) => {
333        let part := match piece.field {
334          Some(field) => apply("route_segment_item", [encode_field(field, value, at)], at)
335          None => ExText(match piece.literal { Some(text) => text
336            None => "" }, at)
337        }
338        apply("concat", [part, href_path(rest, value, tail, at)], at)
339      }
340    }
341  }
342  fn function(
343    name: SynIdent,
344    params: List(SynIdent),
345    types: List(Option(SynExpr)),
346    result: SynExpr,
347    effects: List(SynRowEntry),
348    body: SynExpr,
349    at: SynSpan
350  ): SynStmt ! {div} {
351    StFn(SynFn { name: name, generics: Nil, generic_types: Nil, bounds: Nil, params: params,
352      r#borrows: nones(params), param_types: types, result: Some(result), effects: Some(effects), where_: Nil, body: body, span: at, unused: None, unused_generics: Nil, unused_params: Nil })
353  }
354  fn nones(params: List(SynIdent)): List(Option(Bool)) ! {div} {
355    match params { Nil => Nil
356      Cons(_, rest) => Cons(None, nones(rest)) }
357  }
358  pub fn make(decl: route_syntax::Decl): List(SynStmt) ! {overflow, macro_error, div} {
359    let owner := Syn::decl_name(Syn::ident(decl.name), decl.name.text)
360    let at := decl.span
361    let pieces := path_pieces(
362      Cons(route_syntax::PathPart::Literal(decl.path.literal), decl.path.path_part),
363      Nil,
364      false
365    )
366    let segments := piece_fields(pieces)
367    let query_items := fields(decl.parameter, Nil, field_names_of(segments), true)
368    let items := List::append(segments, query_items)
369    let value := local_ident("_route_value", at)
370    let target := local_ident("_route_target", at)
371    let href := method(owner, "href")
372    let parse := method(owner, "parse")
373    let spec := method(owner, "route")
374    [
375      StStruct(SynData { kind: Some("shared"), name: owner, generics: Nil, unused_generics: Nil,
376        ctors: [SynCtor { name: owner, fields: data_fields(items) }], span: at }),
377      function(href, [value], [Some(ExVar(owner))], global("Text"), [EffName("div", None), EffName("overflow", None)],
378        href_path(pieces, value, href_fields(query_items, value, true, at), at), at),
379      function(parse, [target], [Some(global("Text"))], result_type(owner, at), [EffName("div", None), EffName("overflow", None)],
380        parse_body(owner, pieces, items, target, at), at),
381      function(spec, Nil, Nil, apply("RouteSpec", [ExVar(owner)], at), Nil,
382        ExRecord(Syn::global("RouteSpec"), [#(ExVar(href), "href"), #(ExVar(parse), "parse")], at), at)
383    ]
384  }
385}

387pub macro fn route(decl: route_syntax::Decl): List(SynStmt) ! {overflow, macro_error, div} {
388  RouteMacro::make(decl)
