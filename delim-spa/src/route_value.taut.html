4// Custom types provide Type::route_codec(): Codec(Type), using the same codec as primitives.
5pub fn route_bool_codec(): Codec(Bool) {
6  Codec { schema: "route-value/v1",
7    decode: (value: Text) => if value == "true" { Some(true) } else if value == "false" {
8      Some(false)
9    } else { None },
10    encode: (value: Bool) => if value { "true" } else { "false" }
11  }
12}

22fn route_slice(value: Text, start: u64, end: u64): Text {
23  match Text::slice(value, start, end) { Some(part) => part
24    None => "" }
25}
26fn route_find(value: Text, byte: u64, start: u64): u64 ! {overflow, div} {
27  match Text::byte(value, start) {
28    None => start
29    Some(c) => if U64::of_u8(c) == byte { start } else { route_find(value, byte, start + 1) }
30  }
31}
32fn route_plus(value: Text, start: u64): Text ! {overflow, div} {
33  let end := route_find(value, 43, start)
34  let before := route_slice(value, start, end)
35  if end < Text::len(value) { concat(before, concat(" ", route_plus(value, end + 1))) } else { before }
36}
37fn route_decode(value: Text): Option(Text) ! {overflow, div} {
38  percent_decode(route_plus(value, 0))
39}

41fn route_query_parts(
42  query: Text,
43  start: u64
44): Result(List(RouteQueryValue), RouteError) ! {overflow, div} {
45  if start >= Text::len(query) { Ok(Nil) } else {
46    let end := route_find(query, 38, start)
47    let pair := route_slice(query, start, end)
48    let equals := route_find(pair, 61, 0)
49    let name := route_decode(route_slice(pair, 0, equals))
50    let value := route_decode(
51      if equals < Text::len(pair) { route_slice(pair, equals + 1, Text::len(pair)) } else { "" }
52    )
53    match name {
54      None => Err(MalformedQuery)
55      Some(key) => match value {
56        None => Err(MalformedQuery)
57        Some(text) => match route_query_parts(query, end + 1) {
58          Err(error) => Err(error)
59          Ok(rest) => if pair == "" { Ok(rest) } else {
60            Ok(Cons(RouteQueryValue { name: key, value: text }, rest))
61          }
62        }
63      }
64    }
65  }
66}

68// A static path must match exactly. Fragments are not part of the route or its query.
69pub fn route_query(
70  target: Text,
71  path: Text
72): Result(List(RouteQueryValue), RouteError) ! {overflow, div} {
73  let fragment := route_find(target, 35, 0)
74  let clean := route_slice(target, 0, fragment)
75  let question := route_find(clean, 63, 0)
76  if route_slice(clean, 0, question) != path { Err(NoRoute) } else {
77    route_query_parts(
78      if question < Text::len(clean) { route_slice(clean, question + 1, Text::len(clean)) } else { "" },
79      0
80    )
81  }
82}

93fn route_starts(value: Text, at: u64, prefix: Text): Bool ! {overflow} {
94  route_slice(
95    value,
96    at,
97    at + Text::len(prefix)
98  ) == prefix && at + Text::len(prefix) <= Text::len(value)
99}
100fn route_path_parts(
101  path: Text,
102  at: u64,
103  parts: List(RoutePart)
104): Result(List(RouteQueryValue), RouteError) ! {overflow, div} {
105  match parts {
106    Nil => if at == Text::len(path) { Ok(Nil) } else { Err(NoRoute) }
107    Cons(part, rest) => match part {
108      RouteLiteral(text) => if route_starts(
109        path,
110        at,
111        text
112      ) { route_path_parts(path, at + Text::len(text), rest) } else { Err(NoRoute) }
113      RouteCapture(name) => {
114        let end := route_find(path, 47, at)
115        if end == at { Err(NoRoute) } else {
116          match percent_decode(route_slice(path, at, end)) {
117            None => Err(MalformedQuery)
118            Some(text) => match route_path_parts(path, end, rest) {
119              Err(error) => Err(error)
120              Ok(values) => Ok(Cons(RouteQueryValue { name: name, value: text }, values))
121            }
122          }
123        }
124      }
125    }
126  }
127}
128// The path must match the declared parts. A capture is one whole, non-empty segment,
129// percent-decoded once; `+` stays `+` in a path. Fragments are not part of the route.
130pub fn route_match(
131  target: Text,
132  parts: List(RoutePart)
133): Result(RouteMatch, RouteError) ! {overflow, div} {
134  let fragment := route_find(target, 35, 0)
135  let clean := route_slice(target, 0, fragment)
136  let question := route_find(clean, 63, 0)
137  match route_path_parts(route_slice(clean, 0, question), 0, parts) {
138    Err(error) => Err(error)
139    Ok(segments) => match route_query_parts(
140      if question < Text::len(clean) { route_slice(clean, question + 1, Text::len(clean)) } else { "" },
141      0
142    ) {
143      Err(error) => Err(error)
144      Ok(query) => Ok(RouteMatch { segments: segments, query: query })
145    }
146  }
147}
148// A segment the codec rejects is another path, so it is NoRoute and other routes are tried.
149pub fn route_read_segment<A: Shared>(
150  codec: Codec(A),
151  values: List(RouteQueryValue),
152  name: Text
153): Result(A, RouteError) ! {overflow, div} {
154  match route_parameter(values, name) {
155    None => Err(NoRoute)
156    Some(text) => ⊢ Result(A, RouteError) from match (codec.decode)(text) {
157      None => Err(NoRoute)
158      Some(value) => Ok(value)
159    }
160  }
161}
162// `.` and `..` are escaped so that a browser does not read the segment as a directory step.
163pub fn route_segment_item(value: Text): Text ! {overflow, div} {
164  if value == "." { "%2E" } else if value == ".." { "%2E%2E" } else { percent_encode(value) }
165}
166pub fn route_parameter(values: List(RouteQueryValue), name: Text): Option(Text) ! {div} {
167  match values { Nil => None
168    Cons(item, rest) => if item.name == name { Some(item.value) } else {
169      route_parameter(rest, name)
170    } }
171}
172pub fn route_read<A: Shared>(
173  codec: Codec(A),
174  values: List(RouteQueryValue),
175  name: Text
176): Result(A, RouteError) ! {overflow, div} {
177  match route_parameter(values, name) {
178    None => Err(MissingParameter(name))
179    Some(text) => ⊢ Result(A, RouteError) from match (codec.decode)(text) {
180      None => Err(InvalidParameter(name))
181      Some(value) => Ok(value)
182    }
183  }
184}
185pub fn route_read_or<A: Shared>(
186  codec: Codec(A),
187  values: List(RouteQueryValue),
188  name: Text,

192    None => Ok(fallback)

194      None => Ok(fallback)

196    }
197  }

212pub fn route_choice<A: Shared, R: Shared>(spec: RouteSpec(A), inject: (A) -> R): RouteChoice(R) {
213  RouteChoice { parse: (target: Text) => match (spec.parse)(target) {
214      Err(error) => Err(error)
215      Ok(value) => Ok(inject(value))
216    } }
217}
218pub fn choose_route<R: Shared>(
219  choices: List(RouteChoice(R)),
220  target: Text
221): Result(R, RouteError) ! {overflow, div} {
222  match choices {
223    Nil => Err(NoRoute)
224    Cons(choice, rest) => match (choice.parse)(target) {
225      Ok(value) => Ok(value)
226      Err(error) => match error {
227        NoRoute => choose_route(rest, target)
228        MalformedQuery => Err(error)
229        MissingParameter(_) => Err(error)
230        InvalidParameter(_) => Err(error)
231      }
232    }
233  }
