1// Codecs: how a value crosses between the browser and the server as text. A server function
2// and a resource carry the codecs of what they take and give, so both sides read the other
3// with the same codec, and a value that does not decode is refused rather than misread.
4
5pub(open) struct Codec(A: Type) {
6  // Stable wire-format descriptor; custom mappings MUST version their own schema.
7  schema: Text,
8  encode: (a: A) -> Text ! {overflow, div},
9  decode: (t: Text) -> Option(A) ! {overflow, div}
10}
11
12pub fn text_codec(): Codec(Text) {
13  Codec { schema: "text/utf8/v1", encode: (a: Text) => a, decode: (t: Text) => Some(t) }
14}
15
16fn digit_text(d: u64): Text ! {overflow} {
17  match Text::slice("0123456789", d, d + 1) {
18    Some(x) => x
19    None => "0"
20  }
21}
22
23pub fn u64_text(n: u64): Text ! {overflow, div} {
24  if n < 10 { digit_text(n) } else { concat(u64_text(div(n, 10)), digit_text(mod(n, 10))) }
25}
26
27// The number the decimal digits of `t` from `i` write, or `None` for anything else, or for
28// a number above `u64`'s range.
29fn u64_of_digits(t: Text, i: u64, acc: u64): Option(u64) ! {overflow, div} {
30  match Text::byte(t, i) {
31    None => Some(acc)
32    Some(b) => {
33      let c := U64::of_u8(b)
34      if c < 48 || c > 57 { None } else if acc > div(18446744073709551615 - (c - 48), 10) {
35        None
36      } else { u64_of_digits(t, i + 1, acc * 10 + (c - 48)) }
37    }
38  }
39}
40
41pub fn u64_of_text(t: Text): Option(u64) ! {overflow, div} {
42  if Text::len(t) == 0 { None } else { u64_of_digits(t, 0, 0) }
43}
44
45pub fn u64_codec(): Codec(u64) {
46  Codec {
47    schema: "u64/decimal/v1",
48    encode: (a: u64) => u64_text(a),
49    decode: (t: Text) => u64_of_text(t)
50  }
51}
52
53pub fn bool_codec(): Codec(Bool) {
54  Codec {
55    schema: "bool/true-false/v1",
56    encode: (a: Bool) => if a { "true" } else { "false" },
57    decode: (t: Text) => if t == "true" { Some(true) } else if t == "false" { Some(false) } else {
58      None
59    }
60  }
61}
62
63pub fn unit_codec(): Codec(Unit) {
64  Codec {
65    schema: "unit/empty/v1",
66    encode: (a: Unit) => "",
67    decode: (t: Text) => if t == "" { Some(#()) } else { None }
68  }
69}

71// A list, as each item's encoding after its length in bytes and `:`, so that an item may
72// hold any text.
73pub fn list_codec<A: Shared>(c: Codec(A)): Codec(List(A)) {
74  Codec {
75    schema: concat("list/length/v1:", c.schema),
76    encode: (a: List(A)) => list_text(c, a),
77    decode: (t: Text) => list_of_text(c, t, 0, Nil)
78  }
79}

81// Written into one buffer in a loop, so that a long list takes neither stack nor time
82// beyond its length.
83fn list_text<A: Shared>(c: Codec(A), l: List(A)): Text ! {overflow, div} {
84  var b := Buf::new()
85  var r := l
86  var go := true
87  while go {
88    match r {
89      Nil => { go := false }
90      Cons(x, rest) => {
91        let e := (c.encode)(x)
92        let n := concat(u64_text(Text::len(e)), ":")
93        Buf::push(&mut b, &n)
94        Buf::push(&mut b, &e)
95        r := rest
96      }
97    }
98  }
99  Text::of(b)
100}
101
102fn reverse_onto<A>(l: List(A), acc: List(A)): List(A) {
103  match l {
104    Nil => acc
105    Cons(x, rest) => reverse_onto(rest, Cons(x, acc))
106  }
107}

109// The length before the `:` at `i`, and where the item starts.
110fn length_at(t: Text, i: u64, acc: u64): Option(u64 × u64) ! {overflow, div} {
111  match Text::byte(t, i) {
112    None => None
113    Some(b) => {
114      let c := U64::of_u8(b)
115      if c == 58 { Some(#(acc, i + 1)) } else if c < 48 || c > 57 {
116        None
117      } else if acc > div(18446744073709551615 - (c - 48), 10) { None } else {
118        length_at(t, i + 1, acc * 10 + (c - 48))
119      }
120    }
121  }
122}
123
124// The items from `i`, the ones read so far newest first in `acc`. Each step calls itself last,
125// so a list of many items runs as a loop rather than a frame for each item. A length larger
126// than what is left of `t` is refused before `start + len` is added, so it cannot overflow.
127fn list_of_text<A: Shared>(
128  c: Codec(A),
129  t: Text,
130  i: u64,
131  acc: List(A)
132): Option(List(A)) ! {overflow, div} {
133  if i == Text::len(t) { Some(reverse_onto(acc, Nil)) } else {
134    match length_at(t, i, 0) {
135      None => None
136      Some(p) => {
137        let #(len, start) := p
138        if start == i + 1 || len > Text::len(t) - start { None } else {
139          match Text::slice(t, start, start + len) {
140            None => None
141            Some(item) => match (c.decode)(item) {
142              None => None
143              Some(x) => list_of_text(c, t, start + len, Cons(x, acc))
144            }
145          }
146        }
147      }
148    }
149  }
150}

152// An optional value: empty for none, and `+` before the value's encoding for one.
153pub fn option_codec<A: Shared>(c: Codec(A)): Codec(Option(A)) {
154  Codec {
155    schema: concat("option/plus/v1:", c.schema),
156    encode: (a: Option(A)) => match a {
157      None => ""
158      Some(x) => concat("+", (c.encode)(x))
159    },
160    decode: (t: Text) => if t == "" { Some(None) } else {
161      match Text::slice(t, 0, 1) {
162        Some(h) => if h == "+" {
163          match Text::slice(t, 1, Text::len(t)) {
164            None => None
165            Some(rest) => match (c.decode)(rest) {
166              None => None
167              Some(x) => Some(Some(x))
168            }
169          }
170        } else { None }
171        None => None
172      }
173    }
174  }
175}

177// The codec of a type made from another's, such as a struct from a list or a pair: `to`
178// may refuse what it is given.
179pub fn mapped_codec<A: Shared, B: Shared>(
180  schema: Text,
181  c: Codec(A),
182  to: (a: A) -> Option(B) ! {overflow, div},
183  convert_back: (b: B) -> A ! {overflow, div}
184): Codec(B) {
185  Codec {
186    schema: concat(concat("mapped/v1:", schema_part(schema)), schema_part(c.schema)),
187    encode: (b: B) => (c.encode)(convert_back(b)),
188    decode: (t: Text) => match (c.decode)(t) {
189      None => None
190      Some(a) => to(a)
191    }
192  }
193}

195fn pair_text<A: Shared, B: Shared>(a: Codec(A), b: Codec(B), p: A × B): Text ! {overflow, div} {
196  match p {
197    #(x, y) => {
198      let e := (a.encode)(x)
199      concat(concat(concat(u64_text(Text::len(e)), ":"), e), (b.encode)(y))
200    }
201  }
202}

204// Two values, the first after its length in bytes and `:`.
205pub fn pair_codec<A: Shared, B: Shared>(a: Codec(A), b: Codec(B)): Codec(A × B) {
206  Codec {
207    schema: concat(
208      concat(concat(concat("pair/length/v1:", Text::of_u64(Text::len(a.schema))), ":"), a.schema),
209      b.schema
210    ),
211    encode: (p: A × B) => pair_text(a, b, p),
212    decode: (t: Text) => match length_at(t, 0, 0) {
213      None => None
214      Some(l) => {
215        let #(len, start) := l
216        if start == 1 { None } else {
217          match Text::slice(t, start, start + len) {
218            None => None
219            Some(x) => match Text::slice(t, start + len, Text::len(t)) {
220              None => None
221              Some(y) => match (a.decode)(x) {
222                None => None
223                Some(va) => match (b.decode)(y) {
224                  None => None
225                  Some(vb) => Some(#(va, vb))
226                }
227              }
228            }
229          }
230        }
231      }
232    }
233  }
234}
235
236fn schema_part(value: Text): Text {
237  concat(concat(Text::of_u64(Text::len(value)), ":"), value)
238}

240// The fields of a form, each its name and its text, in the order the form has them. A server
241// function that takes this can be sent from a page without JavaScript (`web_form.taut`).
242pub fn form_codec(): Codec(List(Text × Text)) { list_codec(pair_codec(text_codec(), text_codec())) }
