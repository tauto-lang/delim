1// Percent-encoding of path segments and query values, so that a key or a route parameter
2// arrives as it was written, whatever bytes it has.
3
4fn hex_value(c: u64): Option(u64) ! {overflow} {
5  if c >= 48 && c <= 57 { Some(c - 48) } else if c >= 65 && c <= 70 {
6    Some(c - 55)
7  } else if c >= 97 && c <= 102 {
8    Some(c - 87)
9  } else { None }
10}
11
12fn reverse_bytes(l: List(u8), acc: List(u8)): List(u8) {
13  match l {
14    Nil => acc
15    Cons(b, rest) => reverse_bytes(rest, Cons(b, acc))
16  }
17}
18
19// The bytes of `t` from `i` with each `%XY` read as one byte, newest first in `acc`; `None`
20// for a `%` without two hex digits. With `plus`, as in a form, a `+` is a space.
21fn decoded_bytes(t: Text, i: u64, acc: List(u8), plus: Bool): Option(List(u8)) ! {overflow, div} {
22  match Text::byte(t, i) {
23    None => Some(reverse_bytes(acc, Nil))
24    Some(b) => if U64::of_u8(b) == 37 {
25      match Text::byte(t, i + 1) {
26        None => None
27        Some(h) => match Text::byte(t, i + 2) {
28          None => None
29          Some(l) => match hex_value(U64::of_u8(h)) {
30            None => None
31            Some(hv) => match hex_value(U64::of_u8(l)) {
32              None => None
33              Some(lv) => decoded_bytes(t, i + 3, Cons(U8::of_u64(hv * 16 + lv), acc), plus)
34            }
35          }
36        }
37      }
38    } else if plus && U64::of_u8(b) == 43 {
39      decoded_bytes(t, i + 1, Cons(U8::of_u64(32), acc), plus)
40    } else {
41      decoded_bytes(t, i + 1, Cons(b, acc), plus)
42    }
43  }
44}

46// `t` with its percent-escapes read; `None` when they are broken or the bytes are not UTF-8.
47pub fn percent_decode(t: Text): Option(Text) ! {overflow, div} {
48  match decoded_bytes(t, 0, Nil, false) {
49    None => None
50    Some(bs) => Text::of_utf8(bs)
51  }
52}
53
54// A name or value of an `application/x-www-form-urlencoded` body: `+` is a space, then the
55// percent-escapes are read.
56pub fn form_decode(t: Text): Option(Text) ! {overflow, div} {
57  match decoded_bytes(t, 0, Nil, true) {
58    None => None
59    Some(bs) => Text::of_utf8(bs)
60  }
61}

63pub fn hex_digit(n: u64): Text ! {overflow} {
64  match Text::slice("0123456789ABCDEF", n, n + 1) {
65    Some(d) => d
66    None => "0"
67  }
68}

70// Letters, digits and `-._~` stay; every other byte is written `%XY`.
71fn unreserved(c: u64): Bool {
72  (c >= 48 && c <= 57) || (c >= 65 && c <= 90) || (c >= 97 && c <= 122) || c == 45 || c == 46 || c == 95 || c == 126
73}
74
75// Written into one buffer in a loop, so that the time is linear in the length of `t`.
76pub fn percent_encode(t: Text): Text ! {overflow, div} {
77  var b := Buf::new()
78  var i: u64 := 0
79  let n := Text::len(t)
80  while i < n {
81    let c := match Text::byte(t, i) {
82      Some(x) => U64::of_u8(x)
83      None => 0
84    }
85    let piece := if unreserved(c) { match Text::slice(t, i, i + 1) {
86        Some(x) => x
87        None => ""
88      } } else { concat(concat("%", hex_digit(div(c, 16))), hex_digit(mod(c, 16))) }
89    Buf::push(&mut b, &piece)
90    i := i + 1
91  }
92  Text::of(b)
93}
