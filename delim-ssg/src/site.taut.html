12pub(open) struct(shared) Page {
13  title: Text,
14  // The language of the document, as `<html lang>`; empty for none.
15  lang: Text,
16  // What goes in `<head>` after the title.

18  view: (t: Tree(HtmlKind, HtmlAttr), v: Vocab(HtmlKind, HtmlAttr)) -> Node ! {rt, div}

21// An element of `<head>`. Values are escaped, and a URL delim's `url_ok` refuses is left out.
22// A stylesheet and a script are code the page runs: they are not made from a user's value.
23pub(open) enum Head {
24  // `<link rel="stylesheet" href>`.
25  Stylesheet(Text)
26  // `<link rel="icon" href type sizes>`.
27  Icon(Text, Text, Text)
28  // `<meta name content>`.
29  Meta(Text, Text)
30  // `<script type="module" src>`.
31  Script(Text)
32  // `<meta property content>`, as Open Graph uses (`og:title`).
33  Property(Text, Text)
34  // `<link rel href>` for a relation that loads nothing, such as `canonical`.
35  Related(Text, Text)
36  // `<link rel="alternate" hreflang href>`: the same page in another language.
37  Alternate(Text, Text)
38  // `<script src>`, run before the page is drawn, for what must apply before the first paint.
39  BlockingScript(Text)
40}

42// How `generate` names the file of a page other than `/`.
43pub(open) enum Layout {
44  // `<path>.html`.
45  Flat
46  // `<path>/index.html`, which a static host serves at `<path>/`.
47  Directories
48}

50fn quoted(name: Text, value: Text): Text ! {overflow, div} {
51  concat(concat(concat(" ", name), concat("=\"", Escape::attr(value))), "\"")
52}
53
54fn head_element(h: Head): Text ! {overflow, div} {
55  match h {
56    Stylesheet(href) => if url_ok(href) {
57      concat(concat("<link rel=\"stylesheet\"", quoted("href", href)), ">")
58    } else { "" }
59    Icon(href, mime, sizes) => if url_ok(href) {
60      concat(
61        concat(
62          concat(concat("<link rel=\"icon\"", quoted("href", href)), quoted("type", mime)),
63          quoted("sizes", sizes)
64        ),
65        ">"
66      )
67    } else { "" }
68    Meta(name, content) => concat(
69      concat(concat("<meta", quoted("name", name)), quoted("content", content)),
70      ">"
71    )
72    Script(src) => if url_ok(src) {
73      concat(concat("<script type=\"module\"", quoted("src", src)), "></script>")
74    } else { "" }
75    Property(property, content) => concat(
76      concat(concat("<meta", quoted("property", property)), quoted("content", content)),
77      ">"
78    )
79    Related(rel, href) => if url_ok(href) && link_rel_ok(rel) {
80      concat(concat(concat("<link", quoted("rel", rel)), quoted("href", href)), ">")
81    } else { "" }
82    Alternate(lang, href) => if url_ok(href) {
83      concat(
84        concat(concat("<link rel=\"alternate\"", quoted("hreflang", lang)), quoted("href", href)),
85        ">"
86      )
87    } else { "" }
88    BlockingScript(src) => if url_ok(src) {
89      concat(concat("<script", quoted("src", src)), "></script>")
90    } else { "" }
91  }
92}

94// The relations `Related` writes: ones that load no code or style into the page.
95fn link_rel_ok(rel: Text): Bool {
96  rel == "canonical" || rel == "author" || rel == "license" || rel == "help" || rel == "me"
97}

99fn head_elements(l: List(Head)): Text ! {overflow, div} {
100  match l {
101    Nil => ""
102    Cons(h, rest) => concat(head_element(h), head_elements(rest))
103  }
104}

106// The document up to the view: the head, and `<div id="app">`, which the DOM renderer mounts on.
107pub fn document_open(p: Page): Text ! {overflow, div} {
108  let open := if p.lang == "" { "<html>" } else {
109    concat(concat("<html", quoted("lang", p.lang)), ">")
110  }
111  concat(
112    concat(
113      concat(concat("<!doctype html>\n", open), "<head><meta charset=\"utf-8\"><title>"),
114      escape_text(p.title)
115    ),
116    concat(concat("</title>", head_elements(p.head)), "</head><body><div id=\"app\">")
117  )
118}

120// The whole document of a page: the rendered view in `<div id="app">`, with the resources it
121// asks for loaded by `load`. A resource that fails to load fails the page, with the first
122// reason, so that no file is written with a part missing.
123pub fn document<!e>(
124  p: Page,
125  load: (key: Text) -> Result(Text, Text) ! {overflow, div} ∪ e
126): Result(Text, Text) ! {overflow, wait, div, cells, ref_new, ref_get, ref_set} ∪ e {
127  let failed: Ref(Option(Text)) := perform ref_new(None)
128  let noted := (key: Text) => match load(key) {
129    Ok(v) => Ok(v)
130    Err(why) => {
131      match perform ref_get(failed) {
132        None => perform ref_set(failed, Some(concat(concat(key, ": "), why))),
133        Some(w) => #()
134      }
135      Err(why)
136    }
137  }
138  let body := render_html_loading(p.view, noted)
139  match perform ref_get(failed) {
140    Some(why) => Err(why)
141    None => Ok(concat(document_open(p), concat(body, "</div></body></html>\n")))
142  }
143}

145// Whether `path` may name a file under the output: it starts with `/`, and no segment is
146// empty, `.` or `..`.
147pub fn file_path_ok(path: Text): Bool ! {overflow, div} {
148  match Text::byte(path, 0) {
149    Some(b) => U64::of_u8(b) == 47 && segments_ok(path, 1, 1)
150    None => false
151  }
152}
153
154fn segments_ok(t: Text, begin: u64, i: u64): Bool ! {overflow, div} {
155  match Text::byte(t, i) {
156    None => segment_ok(t, begin, i)
157    Some(b) => if U64::of_u8(b) == 47 {
158      segment_ok(t, begin, i) && segments_ok(t, i + 1, i + 1)
159    } else { segments_ok(t, begin, i + 1) }
160  }
161}
162
163fn segment_ok(t: Text, begin: u64, to: u64): Bool ! {div} {
164  match Text::slice(t, begin, to) {
165    None => false
166    Some(s) => not(s == "" || s == "." || s == "..")
167  }
168}

170// The directory part of `path`, up to its last `/`.
171pub fn parent_dir(path: Text): Text ! {overflow, div} { parent_until(path, 0, 0) }
172
173fn parent_until(t: Text, i: u64, last: u64): Text ! {overflow, div} {
174  match Text::byte(t, i) {
175    None => match Text::slice(t, 0, last) {
176      Some(d) => d
177      None => ""
178    }
179    Some(b) => parent_until(t, i + 1, if U64::of_u8(b) == 47 { i } else { last })
180  }
181}

183// Writes `contents` to `dir` + `file`, making the directories it is in.
184pub fn write_under(
185  dir: Text,
186  file: Text,
187  contents: Text
188): Result(Unit, Text) ! {overflow, fs_write, fs_create_dir, div} {
189  if not(file_path_ok(file)) { Err(concat("not a path under the output: ", file)) } else {
190    let full := concat(dir, file)
191    match perform fs_create_dir(parent_dir(full)) {
192      Err(e) => Err(io_message(e))
193      Ok(u) => write_file(full, contents)
194    }
195  }
196}

198// `write_under` for bytes, as an image, a font or a `.wasm` is.
199pub fn write_bytes_under(
200  dir: Text,
201  file: Text,
202  contents: Bytes
203): Result(Unit, Text) ! {overflow, fs_write_bytes, fs_create_dir, div} {
204  if not(file_path_ok(file)) { Err(concat("not a path under the output: ", file)) } else {
205    let full := concat(dir, file)
206    match perform fs_create_dir(parent_dir(full)) {
207      Err(e) => Err(io_message(e))
208      Ok(u) => match perform fs_write_bytes(full, contents) {
209        Err(e) => Err(io_message(e))
210        Ok(v) => Ok(#())
211      }
212    }
213  }
214}

216// The file a page at `path` is written to.
217pub fn page_file(layout: Layout, path: Text): Text ! {div} {
218  if path == "/" { "/index.html" } else {
219    match layout {
220      Flat => concat(path, ".html")
221      Directories => concat(path, "/index.html")
222    }
223  }
224}

226// `generate`, also writing `missing` as `/404.html`, the page static hosts answer a path they
227// have no file for with.
228pub fn generate_missing<!e>(
229  dir: Text,
230  layout: Layout,
231  pages: List(Text × Page),
232  missing: Option(Page),
233  load: (key: Text) -> Result(Text, Text) ! {overflow, div} ∪ e,
234  assets: List(@path Text × @body Bytes)
235): Result(Unit, Text) ! {
236  overflow,
237  wait,
238  fs_write,
239  fs_write_bytes,
240  fs_create_dir,
241  div,
242  cells,
243  ref_new,
244  ref_get,
245  ref_set
246} ∪ e {
247  match missing {
248    None => generate(dir, layout, pages, load, assets)
249    Some(p) => match document(p, load) {
250      Err(why) => Err(concat("/404.html: ", why))
251      Ok(d) => match write_under(dir, "/404.html", d) {
252        Err(why) => Err(why)
253        Ok(u) => generate(dir, layout, pages, load, assets)
254      }
255    }
256  }
257}

259// The first path that two of `pages` share, or that is not a path a file can be written at.
260pub fn bad_path(pages: List(Text × Page)): Option(Text) ! {overflow, div} {
261  match pages {
262    Nil => None
263    Cons(#(path, p), rest) => if path != "/" && not(file_path_ok(path)) {
264      Some(concat(path, ": not a path a page can be written at"))
265    } else if has_path(rest, path) {
266      Some(concat(path, ": two pages have this path"))
267    } else { bad_path(rest) }
268  }
269}

271fn has_path(pages: List(Text × Page), path: Text): Bool {
272  match pages {
273    Nil => false
274    Cons(#(q, p), rest) => q == path || has_path(rest, path)
275  }
276}

278// Writes each of `pages` to `dir`, as `index.html` for `/` and by `layout` otherwise, and then
279// each of `assets` at its path. Nothing is written when two pages share a path or a path is
280// not one a file can be written at.
281pub fn generate<!e>(
282  dir: Text,
283  layout: Layout,
284  pages: List(Text × Page),
285  load: (key: Text) -> Result(Text, Text) ! {overflow, div} ∪ e,
286  assets: List(@path Text × @body Bytes)
287): Result(Unit, Text) ! {
288  overflow,
289  wait,
290  fs_write,
291  fs_write_bytes,
292  fs_create_dir,
293  div,
294  cells,
295  ref_new,
296  ref_get,
297  ref_set
298} ∪ e {
299  match bad_path(pages) {
300    Some(why) => Err(why)
301    None => write_pages(dir, layout, pages, load, assets)
302  }
303}

305fn write_pages<!e>(
306  dir: Text,
307  layout: Layout,
308  pages: List(Text × Page),
309  load: (key: Text) -> Result(Text, Text) ! {overflow, div} ∪ e,
310  assets: List(@path Text × @body Bytes)
311): Result(Unit, Text) ! {
312  overflow,
313  wait,
314  fs_write,
315  fs_write_bytes,
316  fs_create_dir,
317  div,
318  cells,
319  ref_new,
320  ref_get,
321  ref_set
322} ∪ e {
323  match pages {
324    Nil => write_assets(dir, assets)
325    Cons(#(path, p), rest) => match document(p, load) {
326      Err(why) => Err(concat(concat(path, ": "), why))
327      Ok(d) => match write_under(dir, page_file(layout, path), d) {
328        Err(why) => Err(why)
329        Ok(u) => write_pages(dir, layout, rest, load, assets)
330      }
331    }
332  }
333}
334
335fn write_assets(
336  dir: Text,
337  assets: List(@path Text × @body Bytes)
338): Result(Unit, Text) ! {overflow, fs_write_bytes, fs_create_dir, div} {
339  match assets {
340    Nil => Ok(#())
341    Cons(#(@path p, @body b), rest) => match write_bytes_under(dir, p, b) {
342      Err(why) => Err(why)
343      Ok(u) => write_assets(dir, rest)
344    }
345  }
346}

348// An asset of text, as a stylesheet or a script is.
349pub fn text_asset(path: Text, body: Text): @path Text × @body Bytes {
350  #(@path path, @body Bytes::of_text(body))

353// The path of a request target, without its query.
354pub fn path_of(target: Text): Text ! {overflow, div} { path_until(target, 0) }

356fn path_until(t: Text, i: u64): Text ! {overflow, div} {
357  match Text::byte(t, i) {
358    None => t
359    Some(b) => if U64::of_u8(b) == 63 { match Text::slice(t, 0, i) {
360        Some(p) => p
361        None => t
362      } } else { path_until(t, i + 1) }
363  }
