7// A page and the address it is written at. A page with arguments is a function that gives its
8// route, such as `fn post_route(p: Post): Route`; a link is made from the same route, so a link
9// only names an address that is written.
10pub(open) struct(shared) Route { path: Text, page: Page }
11
12// The route of `page` at the path of `segments`, each escaped: `["posts", "a b"]` is
13// `/posts/a%20b`, and no segment is `[]`, which is `/`.
14pub fn route(segments: List(Text), page: Page): Route ! {overflow, div} {
15  Route { path: path_of_segments(segments), page: page }
16}
17
18// `/` and the segments, each escaped, after a `/`.
19pub fn path_of_segments(segments: List(Text)): Text ! {overflow, div} {
20  match segments {
21    Nil => "/"
22    Cons(_, _) => joined(segments)
23  }
24}
25
26fn joined(l: List(Text)): Text ! {overflow, div} {
27  match l {
28    Nil => ""
29    Cons(s, rest) => concat(concat("/", percent_encode(s)), joined(rest))
30  }
31}
32
33// A link to `r`.
34pub fn link(t: Tree(HtmlKind, HtmlAttr), r: Route, children: List(Node)): Node ! {rt, div} {
35  el(t, Tag("a"), [Static(Attr("href", r.path))], children)
36}
37
38// The route of each of `rs`, for a site that names its pages by a type of its own: the path of
39// each from its segments, and its page.
40pub fn routes_of<R: Shared>(
41  rs: List(R),
42  segments: (r: R) -> List(Text) ! {overflow, div},
43  page: (r: R) -> Page ! {overflow, div}
44): List(Route) ! {overflow, div} {
45  match rs {
46    Nil => Nil
47    Cons(r, rest) => Cons(route(segments(r), page(r)), routes_of(rest, segments, page))
48  }
49}
