7// The data a page reads while it is written: a key the view asks for, and how its value is
8// written as text.
9pub(open) struct Res(B: Type) { key: Text, value: Codec(B) }
10
11// What `generate` runs for a key, with the value's type erased by the codec. It may perform
12// the effects `e` besides `Div`, such as reading files.
13pub struct(shared) Answer(!e) { name: Text, run: () -> Result(Text, Text) ! {overflow, div} ∪ e }
14
15// The loader of `r`. An `Err` stops `generate`, with its reason.
16pub fn loading<B: Shared, !e>(
17  r: Res(B),
18  run: () -> Result(B, Text) ! {overflow, div} ∪ e
19): Answer(!e) {
20  Answer {
21    name: r.key,
22    run: () => match run() {
23      Ok(b) => Ok((r.value.encode)(b))
24      Err(why) => Err(why)
25    }
26  }
27}
28
29// Runs the loader named `name`; a name no loader has is refused.
30pub fn load<!e>(l: List(Answer(!e)), name: Text): Result(Text, Text) ! {overflow, div} ∪ e {
31  match l {
32    Nil => Err("no resource has this key")
33    Cons(a, rest) => if a.name == name { (a.run)() } else { load(rest, name) }
34  }
35}
36
37// The value of `r` in the page being written on `t`: pending until `generate` has loaded it,
38// then the decoded value. A value that does not decode is a failure.
39pub fn use_res_in_tree<K: Shared, P: Shared, B: Shared>(
40  t: Tree(K, P),
41  r: Res(B)
42): Resource(B, Text) ! {rt, div} {
43  let source := resource(t, r.key)
44  resource_handle(memo(() => match get(source) {
45    Loading => Pending
46    Loaded(value) => match (r.value.decode)(value) {
47      Some(b) => Ready(b)
48      None => Failed(concat(r.key, ": the value does not decode"))
49    }
50    LoadFailed(why) => Failed(why)
51  }))
52}
