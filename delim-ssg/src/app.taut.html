28// The framework's entry: a program that writes its pages ahead of time (`generate <dir>`).
29
30pub(open) struct(shared) App(!e) {
31  // The pages written by `generate`, each at its route's path.
32  routes: List(Route),
33  // The loaders of the resources the pages read, made by `loading`.
34  resources: List(Answer(!e)),
35  // Files written by `generate` at their paths, such as `text_asset("/style.css", css)` or
36  // `#(@path "/app.wasm", @body wasm)`. A path starts with `/` and has no `.` or `..` segment.
37  assets: List(@path Text × @body Bytes),
38  // How `generate` names the files of the pages.
39  layout: Layout,
40  // The page `generate` writes as `/404.html`, which static hosts answer a path they have no
41  // file for with; `None` writes none.
42  not_found: Option(Page)
43}
44
45pub fn usage(): Text { "usage: generate <dir>" }
46
47fn pairs_of(rs: List(Route)): List(Text × Page) {
48  match rs {
49    Nil => Nil
50    Cons(r, rest) => Cons(#(r.path, r.page), pairs_of(rest))
51  }
52}
53
54// `generate <dir>` from the command line. It needs `--allow-write` and nothing else.
55pub fn generate_app<!e>(a: App(!e)): Unit ! {
56  overflow,
57  wait,
58  argv,
59  stdout,
60  stderr,
61  exit,
62  fs_write,
63  fs_write_bytes,
64  fs_create_dir,
65  div,
66  mutex_claim,
67  mutex_new,
68  mutex_put,
69  mutex_take_claimed,
70  ref_get,
71  ref_new,
72  ref_set,
73  weak_get,
74  weak_of,
75  hash_seed
76} ∪ e {
77  match perform argv() {
78    Cons(cmd, Cons(arg, rest)) => if cmd == "generate" { generate_to(a, arg) } else {
79      print(usage())
80    }
81    Nil => print(usage())
82    Cons(_, Nil) => print(usage())
83  }
84}

86// Writes the pages and assets of `a` under `dir`. A failure is written to standard error and
87// ends the program with exit code 1.
88pub fn generate_to<!e>(a: App(!e), dir: Text): Unit ! {
89  overflow,
90  wait,
91  stdout,
92  stderr,
93  exit,
94  fs_write,
95  fs_write_bytes,
96  fs_create_dir,
97  div,
98  mutex_claim,
99  mutex_new,
100  mutex_put,
101  mutex_take_claimed,
102  ref_get,
103  ref_new,
104  ref_set,
105  weak_get,
106  weak_of,
107  hash_seed
108} ∪ e {
109  match generate_missing(
110    dir,
111    a.layout,
112    pairs_of(a.routes),
113    a.not_found,
114    (k: Text) => load(a.resources, k),
115    a.assets
116  ) {
117    Ok(u) => print("generated")
118    Err(why) => {
119      perform stderr(why)
120      perform exit(1)
121    }
122  }
123}
