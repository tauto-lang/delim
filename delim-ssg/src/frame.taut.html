7// Nested layouts. A `Frame` is what pages of one section share: head entries and a wrapper
8// around the page's own view. Frames nest: `nested(outer, inner)` puts the inner frame's
9// wrapper inside the outer one, so a site's shell, a section's nav and a page build one tree
10// and share one suspense and one set of resources.
11
12pub(open) struct Frame {
13  head: List(Head),
14  wrap: (t: Tree(HtmlKind, HtmlAttr), v: Vocab(HtmlKind, HtmlAttr), inner: Node) -> Node ! {rt, div}
15}
16
17fn heads_then(a: List(Head), b: List(Head)): List(Head) {
18  match a {
19    Nil => b
20    Cons(h, rest) => Cons(h, heads_then(rest, b))
21  }
22}
23
24// `p` inside `f`: `f`'s head entries come first, the title and language stay `p`'s.
25pub fn framed(f: Frame, p: Page): Page {
26  Page {
27    title: p.title,
28    lang: p.lang,
29    head: heads_then(f.head, p.head),
30    view: (t: Tree(HtmlKind, HtmlAttr), v: Vocab(HtmlKind, HtmlAttr)) => {
31      let inner := (p.view)(t, v)
32      (f.wrap)(t, v, inner)
33    }
34  }
35}
36
37// A frame that puts `inner`'s result inside `outer`'s wrapper.
38pub fn nested(outer: Frame, inner: Frame): Frame {
39  Frame {
40    head: heads_then(outer.head, inner.head),
41    wrap: (t: Tree(HtmlKind, HtmlAttr), v: Vocab(HtmlKind, HtmlAttr), n: Node) => {
42      let mid := (inner.wrap)(t, v, n)
43      (outer.wrap)(t, v, mid)
44    }
45  }
46}
