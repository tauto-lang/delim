25// A button named `label`, showing it, running `press` when pressed.
26pub fn button<K: Shared + NoBorrow, P: Shared + NoBorrow>(
27  t: Tree(K, P),
28  v: Vocab(K, P),
29  label: Text,
30  press: () -> Unit ! {rt, div}
31): Node ! {rt, div, overflow} where DecEq(P) {
32  el(
33    t,
34    v.pressable,
35    Cons(Static((v.role)(Button(label))), Cons(On(Press, (d: EventData) => press()), Nil)),
36    Cons(text(t, label), Nil)
37  )
38}

40// An input field named `label` showing `value`, and giving each committed text to `edit`.
41pub fn text_input<K: Shared + NoBorrow, P: Shared + NoBorrow>(
42  t: Tree(K, P),
43  v: Vocab(K, P),
44  label: Text,
45  value: delim::src::signal::Signal(Text),
46  edit: (s: Text) -> Unit ! {rt, div}
47): Node ! {rt, div, overflow} where DecEq(P) {
48  el(
49    t,
50    v.input,
51    Cons(
52      Static((v.role)(TextField(label))),
53      Cons(
54        Dyn(() => (v.value)(get(value))),
55        Cons(On(Edit, (d: EventData) => match d {
56          NoData => #()
57          KeyData(k) => #()
58          TextData(s) => edit(s)
59          Pointer(x, y, b) => #()
60          ScrollData(dx, dy) => #()
61        }), Nil)
62      )
63    ),
64    Nil
65  )
66}
