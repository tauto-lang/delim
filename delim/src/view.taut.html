28//

100// `tag` says the token before was `<` or `</`.
101fn view_colour(ts: List(SynTok), tag: Bool): List(SynHighlight) ! {div} {
102  match ts {
103    Nil => Nil
104    Cons(t, rest) => if t.skipped { view_colour(rest, tag) } else if t.kind == "Name" {
105      view_mark(t, if tag { "type" } else { "property" }, view_colour(rest, false))
106    } else if t.kind == "Str" {
107      view_mark(t, "string", view_colour(rest, false))
108    } else if t.kind == "TautoExpr" {
109      view_colour(rest, false)
110    } else if t.kind == "KwIf" || t.kind == "KwElse" || t.kind == "KwFor" || t.kind == "KwMatch" || t.kind == "KwMount" {
111      view_mark(t, "keyword", view_colour(rest, false))
112    } else { view_mark(t, "operator", view_colour(rest, t.kind == "Lt" || t.kind == "LtSlash")) }
113  }
114}

150// `<name attrs (/> | > children </name>)` from the tokens, with what follows.
151fn view_read_element(ts: List(SynTok)): Option(ViewRead) ! {div, overflow} {
152  match ts {
153    Cons(lt, Cons(name, rest)) => if lt.kind == "Lt" && name.kind == "Name" {
154      view_read_attrs(name.text, Nil, rest)
155    } else { None }
156    Nil => None
157    Cons(_, Nil) => None

161// Read a complete child, including directive blocks. Embedded Tauto expressions are opaque.
162fn view_read_el(ts: List(SynTok)): Option(ViewRead) ! {div, overflow} {
163  match ts {
164    Nil => None
165    Cons(t, rest) => if t.kind == "Lt" { view_read_element(ts) } else if t.kind == "FragOpen" {
166      view_read_parts("fragment", "", Nil, rest)
167    } else if t.kind == "KwIf" || t.kind == "KwFor" || t.kind == "KwMatch" || t.kind == "KwMount" {
168      view_read_header(t.text, rest)
169    } else if t.kind == "Str" || t.kind == "TautoExpr" {
170      Some(ViewRead { node: LayLeaf(t.text), rest: rest })
171    } else { None }
172  }

175fn view_header_piece(head: Text, token: SynTok): Text ! {div, overflow} {
176  if head == "" {
177    token.text
178  } else if token.kind == "LParen" || token.kind == "RParen" || token.kind == "Comma" {
179    concat(head, token.text)
180  } else if Text::slice(head, Text::len(head) - 1, Text::len(head)) == Some("(") {
181    concat(head, token.text)
182  } else { concat(head, concat(" ", token.text)) }
183}

189fn view_read_header(head: Text, ts: List(SynTok)): Option(ViewRead) ! {div, overflow} {
190  match ts {
191    Nil => None
192    Cons(t, rest) => if t.kind == "LBrace" { view_read_parts("block", head, Nil, rest) } else {
193      view_read_header(view_header_piece(head, t), rest)
194    }
195  }

198fn view_read_parts(
199  kind: Text,
200  head: Text,
201  children: List(ViewLayout),
202  ts: List(SynTok)
203): Option(ViewRead) ! {div, overflow} {
204  match ts {
205    Nil => None
206    Cons(t, rest) => if (kind == "fragment" && t.kind == "FragClose") || (kind == "block" && t.kind == "RBrace") {
207      let node := if kind == "fragment" {
208        LayEl("", Nil, Some(view_rev_nodes(children, Nil)))
209      } else { LayBlock(head, view_rev_nodes(children, Nil)) }
210      match rest {
211        Cons(other, more) => if kind == "block" && other.kind == "KwElse" {
212          let branch := match more {
213            Cons(next, _) => if next.kind == "KwIf" { view_read_el(more) } else {
214              view_read_header("", more)
215            }
216            Nil => None
217          }
218          match branch {
219            Some(b) => Some(ViewRead { node: LayChain(node, b.node), rest: b.rest })
220            None => None
221          }
222        } else { Some(ViewRead { node: node, rest: rest }) }
223        Nil => Some(ViewRead { node: node, rest: rest })
224      }
225    } else {
226      let child := view_read_el(ts)
227      // A match arm has a pattern header followed by a block.
228      let next := match child { Some(c) => Some(c)
229        None => if kind == "block" { view_read_header("", ts) } else { None } }
230      match next {
231        Some(c) => view_read_parts(kind, head, Cons(c.node, children), c.rest)
232        None => None
233      }
234    }
235  }
236}

238fn view_read_attrs(
239  name: Text,
240  attrs: List(Text),
241  ts: List(SynTok)
242): Option(ViewRead) ! {div, overflow} {
243  match ts {
244    Nil => None
245    Cons(t, rest) => if t.kind == "SlashGt" {
246      Some(ViewRead { node: LayEl(name, view_rev(attrs, Nil), None), rest: rest })
247    } else if t.kind == "Gt" {
248      view_read_children(name, view_rev(attrs, Nil), Nil, rest)
249    } else if t.kind == "Name" {
250      match rest {
251        Cons(eq, Cons(v, more)) => if eq.kind == "Equals" {
252          view_read_attrs(name, Cons(concat(t.text, concat("=", v.text)), attrs), more)
253        } else { None }
254        Nil => None
255        Cons(_, Nil) => None
256      }
257    } else { None }
258  }
259}

261// A `</name>` that closes another element is the macro's error; the formatter leaves it be.
262fn view_read_children(
263  name: Text,
264  attrs: List(Text),
265  children: List(ViewLayout),
266  ts: List(SynTok)
267): Option(ViewRead) ! {div, overflow} {
268  match ts {
269    Nil => None
270    Cons(t, rest) => if t.kind == "LtSlash" {
271      match rest {
272        Cons(close, Cons(gt, more)) => if close.kind == "Name" && gt.kind == "Gt" && close.text == name {
273          Some(
274            ViewRead { node: LayEl(name, attrs, Some(view_rev_nodes(children, Nil))), rest: more }
275          )
276        } else { None }
277        Nil => None
278        Cons(_, Nil) => None
279      }
280    } else {
281      match view_read_el(ts) {
282        Some(r) => view_read_children(name, attrs, Cons(r.node, children), r.rest)
283        None => None
284      }
285    }
286  }
287}

304fn view_flat(n: ViewLayout): Text ! {div} {
305  match n {
306    LayLeaf(t) => t
307    LayBlock(head, children) => match children {
308      Nil => concat(view_block_open(head), "}")
309      Cons(_, _) => concat(
310        view_block_open(head),
311        concat(" ", concat(view_join(view_flats(children), " "), " }"))
312      )
313    }
314    LayChain(first, second) => concat(view_flat(first), concat(" @else ", view_flat(second)))
315    LayEl(name, attrs, None) => concat(view_open_flat(name, attrs), "/>")
316    LayEl(name, attrs, Some(children)) => concat(
317      view_open_flat(name, attrs),
318      concat(">", concat(view_join(view_flats(children), " "), concat("</", concat(name, ">"))))
319    )
320  }
321}

323fn view_flats(l: List(ViewLayout)): List(Text) ! {div} {
324  match l {
325    Nil => Nil
326    Cons(n, rest) => Cons(view_flat(n), view_flats(rest))
327  }

330fn view_has_newline(t: Text, i: u64): Bool ! {div, overflow} {
331  match Text::byte(t, i) {

345// An element whose first line starts at `col` and whose other lines at `indent`: on one line if
346// it fits, else its attributes and its children each on a line of their own, two spaces in.
347fn view_lay(n: ViewLayout, indent: u64, col: u64, width: u64): Text ! {div, overflow} {
348  let flat := view_flat(n)
349  if view_fits(flat, col, width) { flat } else {
350    match n {
351      LayLeaf(t) => t
352      LayBlock(head, children) => concat(
353        view_block_open(head),
354        concat(
355          view_lay_children(children, indent + 2, width),
356          concat(concat("\n", view_spaces(indent)), "}")
357        )
358      )
359      LayChain(first, second) => concat(
360        view_lay(first, indent, col, width),
361        concat(" @else ", view_lay(second, indent, indent + 7, width))
362      )
363      LayEl(name, attrs, rest) => {
364        let inner := concat("\n", view_spaces(indent + 2))
365        let one := view_open_flat(name, attrs)
366        let open := if attrs == Nil || view_fits(concat(one, "/>"), col, width) { one } else {
367          concat(concat("<", name), concat(inner, view_join(attrs, inner)))
368        }
369        match rest {
370          None => concat(open, "/>")
371          Some(Nil) => concat(open, concat("></", concat(name, ">")))
372          Some(children) => concat(
373            open,
374            concat(
375              ">",
376              concat(
377                view_lay_children(children, indent + 2, width),
378                concat(concat("\n", view_spaces(indent)), concat("</", concat(name, ">")))
379              )
380            )
381          )
382        }
383      }
384    }
385  }
386}

388fn view_lay_children(l: List(ViewLayout), indent: u64, width: u64): Text ! {div, overflow} {
389  match l {
390    Nil => ""
391    Cons(n, rest) => concat(
392      concat("\n", view_spaces(indent)),
393      concat(view_lay(n, indent, indent, width), view_lay_children(rest, indent, width))
394    )
395  }

398fn view_format(ts: List(SynTok), width: u64): Option(Text) ! {div, overflow} {
399  match view_solid(ts) {
400    Nil => None
401    Cons(first, rest) => if first.kind == "TautoExpr" {
402      match view_read_el(rest) {
403        Some(r) => if r.rest == Nil {
404          Some(
405            concat(first.text, concat(" ", view_lay(r.node, 0, Text::len(first.text) + 1, width)))
406          )
407        } else { None }
408        None => None
409      }
410    } else { None }
411  }

420// ---- what the renderers write, checked before the program runs --------------------------------
421// A macro function cannot run `policy.taut`, whose byte values are `Nat`s, so its lists are
422// written here again as texts; a test checks that both say the same.
423
424// What a renderer says of its elements, for the `view!` the renderer defines: the kind a tag
425// names, the properties its attributes give (the tree is the one the view is built on), and
426// what `${…}` and `@mount` make. The syntax, `@if`, `@for`, `@match` and fragments are the
427// same for every renderer (`View::child`).
428pub(open) struct ViewDialect {
429  kind: (name: SynToken) -> SynExpr ! {macro_error, div, overflow},
430  attrs: (t: SynExpr, l: List(view::Attr)) -> List(SynExpr) ! {macro_error, div, overflow},
431  value: (t: SynExpr, e: SynExpr) -> SynExpr ! {macro_error, div, overflow},
432  mount: (t: SynExpr, m: view::Mount) -> SynExpr ! {macro_error, div, overflow},
433  // `f`, the function of an `@if`, `@for` or `@match` that the renderer runs again when the signals
434  // it reads change, as a function that runs it with what the view handles (its contexts).
435  lift: (t: SynExpr, f: SynExpr) -> SynExpr ! {macro_error, div, overflow},
436  // `f`, a function that builds a part of an `@if`, `@for` or `@match` later, from the value it
437  // is given, as a function that builds it with what the view handles.
438  keep: (t: SynExpr, f: SynExpr) -> SynExpr ! {macro_error, div, overflow}
439}

462  pub fn event(at: SynSpan, name: Text): SynExpr ! {macro_error, div} {
463    if Text::slice(name, 0, 7) == Some("action:") {

465        None => "" }

467        let value := Syn::lift_text(action, at)

471      quote { Focus }
472    } else if name == "blur" {
473      quote { Blur }
474    } else if name == "key" { quote { Key } } else if name == "edit" {
475      quote { Edit }
476    } else if name == "scroll" {
477      quote { Scroll }
478    } else if name == "submit" {
479      quote { Submit }
480    } else if name == "pointerdown" { quote { PointerDown } } else if name == "pointerup" {
481      quote { PointerUp }
482    } else if name == "pointermove" { quote { PointerMove } } else if name == "pointerenter" {
483      quote { PointerEnter }
484    } else if name == "pointerleave" { quote { PointerLeave } } else if name == "dragstart" {

499  // A handler is `(d: EventData) => …`, or, when it needs nothing from the event, `() => …`,
500  // or, for a handler of committed text (`on:edit`), `(value: Text) => …`, or of a checkbox's
501  // state, `(checked: Bool) => …`: as a function of the event's data.
502  pub fn handler_fn(h: SynExpr): SynExpr ! {macro_error, div} {
503    match h {
504      ExLam(Nil, _, _) => quote { (delim_event: EventData) => ($h)() }
505      ExLam(Cons(BName(_, Some(ExVar(ty)), _), Nil), _, _) => if ty.name == "Text" {
506        quote { (delim_event: EventData) => ($h)(event_text(delim_event)) }
507      } else if ty.name == "Bool" {
508        quote { (delim_event: EventData) => ($h)(event_text(delim_event) == "true") }
509      } else { h }
510      _ => h
511    }
512  }

548  fn conditional(
549    d: ViewDialect,
550    t: SynExpr,
551    c: view::Conditional
552  ): SynExpr ! {macro_error, div, overflow} {
553    let e := c.tauto_expr
554    let cond := (d.lift)(t, quote { () => $e })
555    let yes := View::block(d, t, c.block)
556    match c.otherwise {
557      None => {
558        let build := (d.lift)(t, quote { () => $yes })
559        quote { show($t, $cond, $build) }
560      }
561      Some(other) => {
562        let no := match other.else_body {
563          Block(b) => View::block(d, t, b)
564          Conditional(nested) => View::conditional(d, t, nested)
565        }
566        let build := (d.keep)(t, quote { (value: Bool) => if value { $yes } else { $no } })
567        quote { switch_on($t, $cond, $build) }
568      }
569    }
570  }

582    match p {
583      TautoExpr(e) => View::expr_pattern(e)
584      NameOpt { name: tok, pattern_args: args, span: _ } => {
585        let name := Syn::ident(tok)
586        match args {
587          Some(a) => PtCtor(name, match a.pattern_list {
588            None => Nil
589            Some(ps) => Cons(View::pattern(ps.pattern), View::pattern_tail(ps.pattern_tail))
590          })
591          None => if tok.text == "_" { PtWild } else if tok.text == "true" {
592            PtBool(true)
593          } else if tok.text == "false" { PtBool(false) } else if View::constructor_name(
594            tok.text,
595            0
596          ) { PtCtor(name, Nil) } else { PtBind(name, None) }
597        }
598      }
599    }
600  }
601
602  fn pattern_tail(ps: List(view::PatternTail)): List(SynPattern) ! {macro_error, div, overflow} {
603    match ps {
604      Nil => Nil
605      Cons(p, rest) => Cons(View::pattern(p.pattern), View::pattern_tail(rest))
606    }
607  }
608
609  fn expr_pattern(e: SynExpr): SynPattern ! {macro_error, div, overflow} {
610    match e {
611      ExVar(n) => if n.name == "_" { PtWild } else { PtCtor(n, Nil) }
612      ExBool(b, _) => PtBool(b)
613      ExText(s, _) => PtText(s)
614      ExApp(ExVar(n), args, _) => PtCtor(n, View::expr_patterns(args))
615      _ => Syn::error(Syn::call_span(), "view! pattern: use a name, constructor, Bool or Text")
616    }
617  }
618
619  fn expr_patterns(es: List(SynExpr)): List(SynPattern) ! {macro_error, div, overflow} {
620    match es {
621      Nil => Nil
622      Cons(ExVar(n), rest) => Cons(
623        if n.name == "_" { PtWild } else { PtBind(n, None) },
624        View::expr_patterns(rest)
625      )
626      Cons(e, rest) => Cons(View::expr_pattern(e), View::expr_patterns(rest))
627    }
628  }

630  fn arms(
631    d: ViewDialect,
632    t: SynExpr,
633    xs: List(view::Arm)
634  ): List(SynArm) ! {macro_error, div, overflow} {
635    match xs {
636      Nil => Nil
637      Cons(a, rest) => Cons(SynArm {
638        pattern: View::pattern(a.pattern), guard: None, goal: None, body: View::block(d, t, a.block)
639      }, View::arms(d, t, rest))
640    }
641  }
642
643  pub fn child(d: ViewDialect, t: SynExpr, c: view::Child): SynExpr ! {macro_error, div, overflow} {
644    match c {
645      Element(e) => View::element(d, t, e)
646      Fragment(f) => View::group(d, t, f.child)
647      Conditional(c) => View::conditional(d, t, c)
648      Each(e) => {
649        if e.in_word.name.text == "in" { #() } else {
650          Syn::error(e.in_word.span, "@for expects `in`")
651        }
652        if e.key_word.name.text == "key" { #() } else {
653          Syn::error(e.key_word.span, "@for requires `key` and a key function")
654        }
655        let x := e.tauto_expr
656        let items := (d.lift)(t, quote { () => $x })
657        let key := e.tauto_expr_2
658        let body := View::block(d, t, e.block)
659        let row := (d.keep)(t, ExLam([BName(Syn::ident(e.name), None, None)], body, e.name.span))
660        quote { keyed_each_signal($t, $items, $key, $row) }
661      }
662      Selection(s) => {
663        let x := s.tauto_expr
664        let source := (d.lift)(t, quote { () => $x })
665        let value := match quote { (delim_match_value) => delim_match_value } {
666          ExLam(Cons(BName(n, _, _), _), _, _) => n
667          _ => Syn::error(s.span, "view! could not construct match binder")
668        }
669        let body := ExMatch(
670          ExVar(value),
671          None,
672          View::arms(d, t, s.arm),
673          None,
674          None,
675          Syn::call_span()
676        )
677        let build := (d.keep)(t, ExLam([BName(value, None, None)], body, Syn::call_span()))
678        quote { switch_on_change($t, $source, $build) }
679      }
680      Str(tok) => {
681        let s := Syn::lift_text(View::unquote(tok), tok.span)
682        quote { text($t, $s) }
683      }
684      Mount(m) => (d.mount)(t, m)
685      TautoExpr(e) => (d.value)(t, e)
686    }
687  }
688
689  fn element(d: ViewDialect, t: SynExpr, e: view::Element): SynExpr ! {macro_error, div, overflow} {
690    let kind := (d.kind)(e.name)
691    let props := View::list((d.attrs)(t, e.attr))
692    let children := match e.rest {
693      SlashGt(tok) => quote { Nil }
694      Body(b) => {
695        if b.name.text == e.name.text { #() } else {
696          Syn::error(
697            b.name.span,
698            concat(concat("`</", b.name.text), concat(">` closes `<", concat(e.name.text, ">`")))
699          )
700        }
701        View::list(View::children(d, t, b.child))
702      }
703    }
704    quote { el($t, $kind, $props, $children) }
705  }
