18// Fine-grained reactivity: signals, memos, watchers and the owner tree.
19//
20// A signal holds its value in a `Ref` and knows only its id. Who reads which signal and the
21// owner tree are kept by the runtime in hash maps of ids, and each watcher's function is kept
22// by the runtime too, so a signal never points at the code that reads it and no strong
23// reference cycle forms. The runtime is reached through the effect `rt`, which `reactive`
24// provides. Its runtime lookup is direct; table locks leave `wait` to the host.
25
26// A watcher: the function to run again when a signal it read changes.
27struct Watcher { id: u64, run: () -> Unit ! {rt, div} }
28struct Cleanup { owner: u64, run: () -> Unit ! {rt, div} }
29
30// A timer the renderer has to start: run `run` after `ms` milliseconds, while `live` is true.
31// `every` is set when the timer starts again after each run.
32pub(open) struct(shared) Timer {
33  ms: u64,
34  every: Bool,
35  live: Ref(Bool),
36  run: () -> Unit ! {rt, div}
37}

56pub(open) struct(shared) Runtime {
57  // The resources asked for since the renderer last took them, newest
58  // first.
59  wants: Ref(List(Want)),
60  next: Ref(u64),
61  // The watcher whose run is reading signals now, if any.
62  observer: Ref(Option(u64)),
63  // The owner that watchers and cleanups created now belong to.
64  owner: Ref(u64),
65  // signal → the watchers that read it in their last run; watcher → the signals it read.
66  readers: Mutex(HashMap(u64, List(u64))),
67  // (signal, watcher) → 1 while the watcher reads the signal, 2 once it stopped but still stands in
68  // the signal's `readers`, 3 once it was dropped from there. Absent: never read it.
69  reading: Mutex(HashMap(u64, u64)),
70  // id → the name `label` gave it, for `graph`.
71  names: Mutex(HashMap(u64, Text)),
72  sources: Mutex(HashMap(u64, List(u64))),
73  // watcher → its function; a disposed watcher maps to `None`.
74  watchers: Mutex(HashMap(u64, Option(Watcher))),
75  // owner → the owners and watchers created under it, newest first. A watcher is the owner
76  // of what its run creates.
77  children: Mutex(HashMap(u64, List(u64))),
78  // owner or watcher → the owner it was created under, so that ending it takes it out of
79  // that owner's `children`.
80  parents: Mutex(HashMap(u64, u64)),
81  // signal → how many watchers read it now (state 1 in `reading`). A signal no watcher reads
82  // any more is taken out of `readers` and `reading`, so that a signal that ends with its part
83  // leaves nothing behind.
84  live: Mutex(HashMap(u64, u64)),
85  // owner → its cleanups, newest first.
86  cleanups: Mutex(HashMap(u64, List(Cleanup))),
87  // Watchers to run again once the current write is done, newest first, and whether each is
88  // queued.
89  pending: Ref(List(u64)),
90  queued: Mutex(HashMap(u64, Bool)),
91  // Whether a write is already running the pending watchers.
92  flushing: Ref(Bool),
93  // watcher → how many times the current write has run it.
94  runs: Mutex(HashMap(u64, u64)),
95  // The watchers writes stopped running again since `take_stalled`, newest first.
96  stalled: Ref(List(u64)),
97  // The timers asked for since the renderer last took them, newest first.
98  timers: Ref(List(Timer)),
99  // Why the host gave up on a call into the runtime since `take_failures`, newest first.
100  failures: Ref(List(Text)),
101  // id → a function kept for the host to run later (an event's handler). A tree names its
102  // handlers by these ids, so it holds no function that may hold the tree.
103  held: Mutex(HashMap(u64, Watcher))
104}

138fn has_id(l: List(u64), x: u64): Bool {
139  match l {
140    Nil => false
141    Cons(y, t) => if y == x { true } else { has_id(t, x) }
142  }

145fn without_id(l: List(u64), x: u64): List(u64) {
146  match l {
147    Nil => Nil
148    Cons(y, t) => if y == x { without_id(t, x) } else { Cons(y, without_id(t, x)) }
149  }

152fn reverse_ids(l: List(u64), acc: List(u64)): List(u64) {
153  match l {
154    Nil => acc
155    Cons(x, t) => reverse_ids(t, Cons(x, acc))
156  }

159fn ids_at(t: Mutex(HashMap(u64, List(u64))), k: u64): List(u64) ! {wait, mutex_claim} {
160  alone_with(t, (m) => match HashMap::get(&m, &k) {
161    None => Nil
162    Some(l) => own(&l)
163  })
164}

166// An empty list is not kept, so that an id that ends leaves nothing in the table.
167fn set_ids(t: Mutex(HashMap(u64, List(u64))), k: u64, l: List(u64)): Unit ! {wait, mutex_claim} {
168  alone_with(t, (m) => {
169    match l {
170      Nil => { let gone := HashMap::remove(&mut m, &k); #() }
171      Cons(_, _) => { HashMap::insert(&mut m, k, l); #() }
172    }
173  })
174}

176fn push_id(t: Mutex(HashMap(u64, List(u64))), k: u64, x: u64): Unit ! {wait, mutex_claim} {
177  set_ids(t, k, Cons(x, ids_at(t, k)))
178}

180fn is_queued(r: Runtime, w: u64): Bool ! {wait, mutex_claim} {

182    None => false

187fn set_queued(r: Runtime, w: u64, b: Bool): Unit ! {wait, mutex_claim} {
188  alone_with(r.queued, (m) => {
189    if b { HashMap::insert(&mut m, w, b); #() } else { let gone := HashMap::remove(&mut m, &w); #() }

193// Forgets what `w` read in its last run.
194fn unsubscribe(r: Runtime, w: u64): Unit ! {wait, overflow, mutex_claim} {
195  drop_reader(r, ids_at(r.sources, w), w)
196  set_ids(r.sources, w, Nil)

199fn drop_reader(r: Runtime, sources: List(u64), w: u64): Unit ! {wait, overflow, mutex_claim} {
200  match sources {
201    Nil => #()
202    Cons(s, rest) => {
203      set_reading(r, s, w, 2)
204      if add_live(r, s, false) == 0 { forget_readers(r, s) } else { #() }
205      drop_reader(r, rest, w)
206    }
207  }
208}

210fn reading_key(s: u64, w: u64): u64 ! {overflow} { s * 4294967296 + w }

212// One more (`up`) or one fewer watcher reads `s`; answers how many read it now.
213fn add_live(r: Runtime, s: u64, up: Bool): u64 ! {wait, overflow, mutex_claim} {
214  let n := alone_with(r.live, (m) => match HashMap::get(&m, &s) {
215    None => 0
216    Some(x) => own(&x)
217  })
218  let next := if up { n + 1 } else if n == 0 { 0 } else { n - 1 }
219  alone_with(r.live, (m) => {
220    if next == 0 { let gone := HashMap::remove(&mut m, &s); #() } else { HashMap::insert(&mut m, s, next); #() }
221  })
222  next
223}

225// No watcher reads `s` now: what `readers` and `reading` keep for it goes.
226fn forget_readers(r: Runtime, s: u64): Unit ! {wait, overflow, mutex_claim} {
227  forget_each(r, s, ids_at(r.readers, s))
228  set_ids(r.readers, s, Nil)
229}
230
231fn forget_each(r: Runtime, s: u64, l: List(u64)): Unit ! {wait, overflow, mutex_claim} {
232  match l {
233    Nil => #()
234    Cons(w, rest) => {
235      set_reading(r, s, w, 0)
236      forget_each(r, s, rest)
237    }
238  }
239}

241fn reading_of(r: Runtime, s: u64, w: u64): u64 ! {wait, overflow, mutex_claim} {
242  let key := reading_key(s, w)
243  alone_with(r.reading, (m) => match HashMap::get(&m, &key) {
244    None => 0
245    Some(n) => own(&n)
246  })
247}

249// 3 (dropped from the signal's readers) is kept as no entry, which reads as 0: the watcher
250// is not among the readers either way.
251fn set_reading(r: Runtime, s: u64, w: u64, n: u64): Unit ! {wait, overflow, mutex_claim} {
252  let key := reading_key(s, w)
253  alone_with(r.reading, (m) => {
254    if n == 3 || n == 0 { let gone := HashMap::remove(&mut m, &key); #() } else { HashMap::insert(&mut m, key, n); #() }
255  })
256}

258// The watchers that read the signal `s` now, oldest first. A watcher that stopped reading it
259// stays in the list until here, so that stopping costs nothing; this also drops it.
260fn live_readers(r: Runtime, s: u64): List(u64) ! {wait, overflow, mutex_claim} {
261  let oldest := keep_live(r, s, ids_at(r.readers, s), Nil)
262  set_ids(r.readers, s, reverse_ids(oldest, Nil))
263  oldest
264}
265
266fn keep_live(
267  r: Runtime,
268  s: u64,
269  l: List(u64),
270  acc: List(u64)
271): List(u64) ! {wait, overflow, mutex_claim} {
272  match l {
273    Nil => acc
274    Cons(w, t) => {
275      let n := reading_of(r, s, w)
276      if n == 1 { keep_live(r, s, t, Cons(w, acc)) } else {
277        if n == 2 { set_reading(r, s, w, 3) } else { #() }
278        keep_live(r, s, t, acc)
279      }
280    }
281  }
282}

319// Keeps `f` in the runtime until `release`, and gives the id `run_held` runs it by.
320pub fn hold(f: () -> Unit ! {rt, div}): u64 ! {rt} {
321  let r := perform access_runtime()
322  let id := fresh(r)
323  alone_with(r.held, (m) => { HashMap::insert(&mut m, id, Watcher { id: id, run: f }); #() })
324  id
325}
326
327// Runs what `hold` kept under `id`, if it is still kept.
328pub fn run_held(id: u64): Unit ! {rt, div} {
329  let r := perform access_runtime()
330  let found: Option(Watcher) := alone_with(r.held, (m) => match HashMap::get(&m, &id) {
331    None => None
332    Some(x) => Some(own(&x))
333  })
334  match found { Some(w) => (w.run)(), None => #() }
335}
336
337// Forgets what `hold` kept under `id`.
338pub fn release(id: u64): Unit ! {rt} {
339  let r := perform access_runtime()
340  alone_with(r.held, (m) => { let gone := HashMap::remove(&mut m, &id); #() })
341}

346  let n := perform ref_get(r.next)

348  n

381pub fn reactive<T, !e>(body: &(() -> T ! {rt} ∪ e)): T ! {div, wait, overflow, cells} ∪ e {

385    return v => v
386  }

442fn get_signal<T: Shared + NoBorrow>(s: delim::src::signal::Signal(T)): T ! {rt} {
443  let r := perform access_runtime()
444  match perform ref_get(r.observer) {
445    None => #()
446    Some(w) => {
447      let n := reading_of(r, s.id, w)
448      if n == 1 { #() } else {
449        if n == 2 { #() } else { push_id(r.readers, s.id, w) }
450        set_reading(r, s.id, w, 1)
451        let now := add_live(r, s.id, true)
452        push_id(r.sources, w, s.id)
453      }
454    }
455  }
456  perform ref_get(s.cell)
457}

459// How many times one write may run a watcher. A watcher that writes a signal it reads makes
460// itself stale again; past this it is not queued, and `take_stalled` names it.
461fn run_limit(): u64 { 100 }
462
463fn runs_of(r: Runtime, w: u64): u64 ! {wait, mutex_claim} {
464  alone_with(r.runs, (m) => match HashMap::get(&m, &w) {
465    None => 0
466    Some(n) => own(&n)
467  })
468}
469
470fn count_run(r: Runtime, w: u64): Unit ! {wait, mutex_claim} {
471  alone_with(r.runs, (m) => {
472    let n := match HashMap::get(&m, &w) {
473      None => 0
474      Some(x) => own(&x)
475    }
476    // The count of runs stays at its greatest when it cannot grow.
477    HashMap::insert(&mut m, w, handle { n + 1 } with { overflow() => n })
478    #()
479  })
480}

482fn queue(r: Runtime, ws: List(u64)): Unit ! {wait, ref_get, ref_set, mutex_claim} {
483  match ws {
484    Nil => #()
485    Cons(w, t) => {
486      if is_queued(r, w) { #() } else if runs_of(r, w) >= run_limit() {
487        let ss := perform ref_get(r.stalled)
488        if has_id(ss, w) { #() } else { perform ref_set(r.stalled, Cons(w, ss)) }
489      } else {
490        set_queued(r, w, true)
491        perform ref_set(r.pending, Cons(w, perform ref_get(r.pending)))
492      }
493      queue(r, t)
494    }
495  }
496}

498// Writes the signal and runs again, once each, the watchers that read it.
499pub fn set<T: Shared + NoBorrow>(s: delim::src::signal::Signal(T), v: T): Unit ! {rt, div} {
500  let r := perform access_runtime()
501  perform ref_set(s.cell, v)
502  queue(r, live_readers(r, s.id))
503  if perform ref_get(r.flushing) { #() } else {
504    perform ref_set(r.flushing, true)
505    flush(r)
506    end_flush(r)
507  }
508}

510// Writes only when the value changes, without subscribing the current watcher.
511pub fn set_eq<T: Shared + NoBorrow>(
512  s: delim::src::signal::Signal(T),
513  v: T
514): Unit ! {rt, div} where DecEq(T) {
515  if peek(s) == v { #() } else { set(s, v) }

518fn end_flush(r: Runtime): Unit ! {wait, ref_set, mutex_claim} {
519  alone_with(r.runs, (m) => {
520    *m := HashMap::with_seed(HashMap::seed(&m))
521    #()
522  })
523  perform ref_set(r.flushing, false)

526// The watchers that writes stopped running again since the last call, because one write
527// had run them too many times, as when a watcher writes a signal it reads; newest first.
528pub fn take_stalled(): List(u64) ! {rt} {
529  let r := current()
530  let ws := perform ref_get(r.stalled)
531  perform ref_set(r.stalled, Nil)
532  ws
533}

535// Puts the runtime back to rest after the host abandoned a call into it halfway, as when a
536// host function threw inside a watcher: the write in progress ends, the watchers it still
537// had queued are dropped and named by `take_stalled`, and no watcher counts as running.
538// Without this, a write left `flushing` and every later write would run no watcher. The host
539// calls it with why it gave up, which `take_failures` gives back.
540pub fn recover(why: Text): Unit ! {rt} {
541  let r := perform access_runtime()
542  perform ref_set(r.failures, Cons(why, perform ref_get(r.failures)))
543  drop_pending(r, perform ref_get(r.pending))
544  perform ref_set(r.pending, Nil)
545  perform ref_set(r.observer, None)
546  perform ref_set(r.owner, root_owner())
547  end_flush(r)
548}
549
550fn drop_pending(r: Runtime, ws: List(u64)): Unit ! {wait, ref_get, ref_set, mutex_claim} {
551  match ws {
552    Nil => #()
553    Cons(w, t) => {
554      set_queued(r, w, false)
555      let ss := perform ref_get(r.stalled)
556      if has_id(ss, w) { #() } else { perform ref_set(r.stalled, Cons(w, ss)) }
557      drop_pending(r, t)
558    }
559  }
560}

570// Runs `f` with its writes held back: the watchers they make stale run once each, after `f`.
571pub fn batch<T>(f: () -> T ! {rt, div}): T ! {rt, div} {
572  let r := perform access_runtime()
573  if perform ref_get(r.flushing) { f() } else {
574    perform ref_set(r.flushing, true)
575    let v := f()
576    flush(r)
577    end_flush(r)
578    v
579  }
580}

582pub fn update<T: Shared + NoBorrow>(
583  s: delim::src::signal::Signal(T),
584  f: (T) -> T ! {div}
585): Unit ! {rt, div} {
586  set(s, f(peek(s)))
587}
588
589// Runs the queued watchers, oldest first, until a run queues no more.
590fn flush(r: Runtime): Unit ! {rt, div} {
591  match perform ref_get(r.pending) {
592    Nil => #()
593    Cons(w, t) => {
594      let batch := reverse_ids(perform ref_get(r.pending), Nil)
595      perform ref_set(r.pending, Nil)
596      run_batch(r, batch)
597      flush(r)
598    }
599  }
600}
601
602fn run_batch(r: Runtime, ws: List(u64)): Unit ! {rt, div} {
603  var l := ws
604  var more := true
605  while more {
606    match l {
607      Nil => { more := false }
608      Cons(w, t) => {
609        set_queued(r, w, false)
610        count_run(r, w)
611        run_watcher(r, w)
612        l := t
613      }
614    }
615  }
616}
617
618// Disposes what the last run of `w` created, forgets what it read, and runs it again.
619fn run_watcher(r: Runtime, w: u64): Unit ! {rt, div} {
620  match find_watcher(r, w) {
621    None => #()
622    Some(x) => {
623      dispose_children(r, w)
624      run_cleanups(r, w)
625      unsubscribe(r, w)
626      let saved_observer := perform ref_get(r.observer)
627      let saved_owner := perform ref_get(r.owner)
628      perform ref_set(r.observer, Some(w))
629      perform ref_set(r.owner, w)
630      (x.run)()
631      perform ref_set(r.observer, saved_observer)
632      perform ref_set(r.owner, saved_owner)
633    }
634  }
635}

637// Runs `f` now, and again whenever a signal it read changes. The watcher belongs to the
638// current owner and ends with it.
639pub fn watch(f: () -> Unit ! {rt, div}): Unit ! {rt, div, overflow} {
640  let r := perform access_runtime()
641  let w := fresh(r)
642  adopt(r, perform ref_get(r.owner), w)
643  set_watcher(r, w, Some(Watcher { id: w, run: () => f() }))
644  run_watcher(r, w)
645}

647// A signal that holds `f()` and follows the signals `f` reads. Its readers run on every
648// change of a source, even when `f()` gives the same value again; `memo_eq` skips those.
649pub fn memo<T: Shared + NoBorrow>(
650  f: () -> T ! {rt, div}
651): delim::src::signal::Signal(T) ! {rt, div} {
652  let r := perform access_runtime()
653  let m := delim::src::signal::Signal { id: fresh(r), cell: perform ref_new(untracked(f)) }
654  // The first run only subscribes: nothing has read `m` yet, so the write reaches no one.
655  watch(() => set(m, f()))
656  m
657}
658
659// A memo whose readers run only when `f()` gives a value different from the one it holds.
660pub fn memo_eq<T: Shared + NoBorrow>(
661  f: () -> T ! {rt, div}
662): delim::src::signal::Signal(T) ! {rt, div} where DecEq(T) {
663  let r := perform access_runtime()
664  let m := delim::src::signal::Signal { id: fresh(r), cell: perform ref_new(untracked(f)) }
665  watch(() => {
666    let v := f()
667    if v == peek(m) { #() } else { set(m, v) }
668  })
669  m
670}

672// Runs `f` without subscribing the current watcher to what it reads.
673pub fn untracked<T>(f: () -> T ! {rt, div}): T ! {rt, div, overflow} {
674  let r := perform access_runtime()
675  let saved := perform ref_get(r.observer)
676  perform ref_set(r.observer, None)
677  let v := f()
678  perform ref_set(r.observer, saved)
679  v
680}

682// Runs `f` when the current owner ends: before its watcher runs again, or when it is disposed.
683pub fn on_cleanup(f: () -> Unit ! {rt, div}): Unit ! {rt} {
684  let r := perform access_runtime()
685  let o := perform ref_get(r.owner)
686  set_cleanups(r, o, Cons(Cleanup { owner: o, run: () => f() }, cleanups_at(r, o)))
687}

691fn add_timer(ms: u64, every: Bool, f: () -> Unit ! {rt, div}): Unit ! {rt} {
692  let r := perform access_runtime()
693  let live := perform ref_new(true)
694  on_cleanup(() => perform ref_set(live, false))
695  perform ref_set(
696    r.timers,
697    Cons(Timer { ms: ms, every: every, live: live, run: f }, perform ref_get(r.timers))
698  )
699}
700
701// Runs `f` once, `ms` milliseconds from now, unless the current owner ends first. The renderer
702// starts the timer (`take_timers`); one that never starts one, such as the HTML writer, never runs it.
703pub fn after(ms: u64, f: () -> Unit ! {rt, div}): Unit ! {rt} { add_timer(ms, false, f) }
704
705// Runs `f` every `ms` milliseconds until the current owner ends.
706pub fn every(ms: u64, f: () -> Unit ! {rt, div}): Unit ! {rt} { add_timer(ms, true, f) }
707
708fn reverse_timers(l: List(Timer), acc: List(Timer)): List(Timer) {
709  match l {
710    Nil => acc
711    Cons(x, rest) => reverse_timers(rest, Cons(x, acc))
712  }
713}
714
715// The timers asked for since the last call, oldest first, for the renderer to start.
716pub fn take_timers(): List(Timer) ! {rt} {
717  let r := perform access_runtime()
718  let ts := perform ref_get(r.timers)
719  perform ref_set(r.timers, Nil)
720  reverse_timers(ts, Nil)
721}

723// The value created in a scope and its owner, for `dispose`.
724pub(open) struct Scoped(T: Type) { value: T, owner: u64 }
725
726// Runs `f` as a new owner and gives back its result and the owner, for `dispose`.
727pub fn scope<T>(f: () -> T ! {rt, div}): Scoped(T) ! {rt, div} {
728  let r := perform access_runtime()
729  let o := fresh(r)
730  adopt(r, perform ref_get(r.owner), o)
731  let saved := perform ref_get(r.owner)
732  perform ref_set(r.owner, o)
733  let v := f()
734  perform ref_set(r.owner, saved)
735  Scoped { value: v, owner: o }
736}

738// Ends the owner `o`: what it created ends first, then its cleanups run, newest first.
739pub fn dispose(o: u64): Unit ! {rt, div, overflow} {
740  let r := perform access_runtime()
741  dispose_owner(r, o)
742}
743
744fn dispose_owner(r: Runtime, o: u64): Unit ! {rt, div} {
745  dispose_children(r, o)
746  run_cleanups(r, o)
747  unsubscribe(r, o)
748  set_watcher(r, o, None)
749  disown(r, o)
750}
751
752// `c` was created under `o`.
753fn adopt(r: Runtime, o: u64, c: u64): Unit ! {wait, mutex_claim} {
754  push_id(r.children, o, c)
755  alone_with(r.parents, (m) => { HashMap::insert(&mut m, c, o); #() })
756}
757
758// Takes `c` out of the children of the owner it was created under. When that owner is ending
759// too, its list is already empty.
760fn disown(r: Runtime, c: u64): Unit ! {wait, mutex_claim} {
761  let p := alone_with(r.parents, (m) => match HashMap::remove(&mut m, &c) {
762    None => None
763    Some(kv) => match kv { #(k, v) => Some(v) }
764  })
765  match p {
766    None => #()
767    Some(o) => set_ids(r.children, o, without_id(ids_at(r.children, o), c))
768  }
769}
770
771fn dispose_children(r: Runtime, o: u64): Unit ! {rt, div} {
772  let cs := ids_at(r.children, o)
773  set_ids(r.children, o, Nil)
774  dispose_all(r, cs)
775}
776
777fn dispose_all(r: Runtime, l: List(u64)): Unit ! {rt, div} {
778  match l {
779    Nil => #()
780    Cons(c, t) => {
781      dispose_owner(r, c)
782      dispose_all(r, t)
783    }
784  }
785}
786
787fn run_cleanups(r: Runtime, o: u64): Unit ! {rt, div} {
788  let l := cleanups_at(r, o)
789  set_cleanups(r, o, Nil)
790  run_each(l)
791}
792
793fn run_each(l: List(Cleanup)): Unit ! {rt, div} {
794  match l {
795    Nil => #()
796    Cons(c, t) => {
797      (c.run)()
798      run_each(t)
799    }
800  }
801}

803// Runs `f` under the owner `o`, outside any watcher: what it creates belongs to `o`.
804pub fn in_owner<T>(o: u64, f: () -> T ! {rt, div}): T ! {rt, div} {
805  let r := perform access_runtime()
806  let saved_observer := perform ref_get(r.observer)
807  let saved_owner := perform ref_get(r.owner)
808  perform ref_set(r.observer, None)
809  perform ref_set(r.owner, o)
810  let v := f()
811  perform ref_set(r.observer, saved_observer)
812  perform ref_set(r.owner, saved_owner)
813  v
814}

818fn digit(d: u64): Text ! {overflow} {
819  match Text::slice("0123456789", d, d + 1) {
820    Some(x) => x
821    None => "0"
822  }
823}

825fn num(n: u64): Text ! {div, overflow} {
826  if n < 10 { digit(n) } else { concat(num(div(n, 10)), digit(mod(n, 10))) }
827}

829// Gives the signal a name that `graph` shows. Returns the signal.
830pub fn label<T: Shared + NoBorrow>(
831  s: delim::src::signal::Signal(T),
832  name: Text
833): delim::src::signal::Signal(T) ! {rt} {
834  let r := perform access_runtime()
835  alone_with(r.names, (m) => {
836    HashMap::insert(&mut m, s.id, name)
837    #()
838  })
839  s
840}
841
842fn name_of(r: Runtime, id: u64): Text ! {wait, div, overflow, mutex_claim} {
843  let n := alone_with(r.names, (m) => match HashMap::get(&m, &id) {
844    None => ""
845    Some(x) => own(&x)
846  })
847  if n == "" { num(id) } else { concat(concat(n, "#"), num(id)) }
848}
849
850fn names_of(r: Runtime, ids: List(u64)): Text ! {wait, div, overflow, mutex_claim} {
851  match ids {
852    Nil => ""
853    Cons(i, rest) => match rest {
854      Nil => name_of(r, i)
855      Cons(j, more) => concat(concat(name_of(r, i), ", "), names_of(r, rest))
856    }
857  }
858}
859
860fn graph_line(r: Runtime, id: u64): Text ! {wait, div, overflow, mutex_claim} {
861  let readers := keep_live(r, id, ids_at(r.readers, id), Nil)
862  let sources := ids_at(r.sources, id)
863  match find_watcher(r, id) {
864    Some(w) => concat(
865      concat(concat("watcher ", name_of(r, id)), " reads ["),
866      concat(names_of(r, sources), "]\n")
867    )
868    None => match readers {
869      Nil => ""
870      Cons(w, rest) => concat(
871        concat(concat("signal ", name_of(r, id)), " is read by ["),
872        concat(names_of(r, readers), "]\n")
873      )
874    }
875  }
876}
877
878fn graph_from(r: Runtime, id: u64, end: u64): Text ! {wait, div, overflow, mutex_claim} {
879  if id >= end { "" } else { concat(graph_line(r, id), graph_from(r, id + 1, end)) }
880}

882// The signals that some watcher reads and the watchers with what they read, one line each, in
883// the order they were made. Name signals with `label` to find them.
884pub fn graph(): Text ! {rt, div, overflow} {
885  let r := perform access_runtime()
886  graph_from(r, 1, perform ref_get(r.next))
