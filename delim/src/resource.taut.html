20// Resources and suspense. A resource is text named by a key, which the renderer loads with
21// what the program gives it: on the server, `render_html_loading`'s `load`; in the browser,
22// the loader set on the DOM host. Its value is text so that what the server loaded can be
23// sent to the browser as it is.
24
25fn reverse_wants(l: List(Want), acc: List(Want)): List(Want) {
26  match l {
27    Nil => acc
28    Cons(w, rest) => reverse_wants(rest, Cons(w, acc))
29  }
30}
31
32// The resources asked for since the last call, oldest first. The renderer loads each and
33// gives the result to `resolve`.
34pub fn take_wants(): List(Want) ! {rt} {
35  let r := current()
36  let ws := perform ref_get(r.wants)
37  perform ref_set(r.wants, Nil)
38  reverse_wants(ws, Nil)
39}

41// The resource named `key`, loading until the renderer resolves it. Under `suspense`, the
42// boundary shows its fallback until this and its other resources are loaded.
43pub fn resource<K: Shared + NoBorrow, P: Shared + NoBorrow>(
44  t: Tree(K, P),
45  key: Text
46): delim::src::signal::Signal(Load) ! {rt, div} {
47  let s := signal(Loading)
48  let b := match perform ref_get(t.boundaries) {
49    Nil => None
50    Cons(c, rest) => {
51      update(c, (n: u64) => handle { n + 1 } with { overflow() => n })
52      Some(c)
53    }
54  }
55  let r := current()
56  perform ref_set(
57    r.wants,
58    Cons(
59      Want { key: key, state: s, boundary: b, settled: perform ref_new(false) },
60      perform ref_get(r.wants)
61    )
62  )
63  s
64}

66// The resource named `key`, outside any suspense boundary: for what is asked for after the
67// view is built, such as by a handler.
68pub fn request(key: Text): delim::src::signal::Signal(Load) ! {rt, div} {
69  let s := signal(Loading)
70  let r := current()
71  perform ref_set(
72    r.wants,
73    Cons(
74      Want { key: key, state: s, boundary: None, settled: perform ref_new(false) },
75      perform ref_get(r.wants)
76    )
77  )
78  s
79}

81// Only the first of `resolve` and `reject` counts: a second would let the boundary go on
82// while another of its resources still loads.
83fn settle(w: Want, l: Load): Unit ! {rt, div} {
84  if perform ref_get(w.settled) { #() } else {
85    perform ref_set(w.settled, true)
86    batch(() => {
87      set(w.state, l)
88      match w.boundary {
89        None => #()
90        Some(c) => update(c, (n: u64) => handle { n - 1 } with { overflow() => 0 })
91      }
92    })
93  }
94}

96pub fn resolve(w: Want, v: Text): Unit ! {rt, div} { settle(w, Loaded(v)) }

98pub fn reject(w: Want, why: Text): Unit ! {rt, div} { settle(w, LoadFailed(why)) }

100// `body()`, once every resource it asked for while being built has loaded, and `fallback()`
101// until then, and again while one loads again. The body is built at once, so its resources are asked for at once, but put in
102// place only when they are loaded.
103pub fn suspense_in<K: Shared + NoBorrow, P: Shared + NoBorrow>(
104  t: Tree(K, P),
105  fallback: () -> Node ! {rt, div},
106  body: () -> Node ! {rt, div}
107): Node ! {rt, div, overflow} {
108  let g := node_of(t)
109  emit(t, CreateGroup(g))
110  let waiting := signal(0)
111  let outer := perform ref_get(t.boundaries)
112  perform ref_set(t.boundaries, Cons(waiting, outer))
113  let b := body()
114  perform ref_set(t.boundaries, outer)
115  let shown := perform ref_new(false)
116  // The fallback is shown again whenever a resource of the boundary loads again (a
117  // `reloadable_in`'s `reload`), above the body, which stays as it was until then.
118  let f := show(t, () => get(waiting) > 0, fallback)
119  emit(t, Append(g, f))
120  watch(() => if get(waiting) == 0 && not(perform ref_get(shown)) {
121    perform ref_set(shown, true)
122    emit(t, Append(g, b))
123  } else { #() })
124  g
125}

127// What a typed resource has: nothing yet, its value, or why it has none.
128pub(open) enum Fetched(A: Type) {
129  Fetching
130  FetchedOk(A)
131  FetchFailed(Text)
132}
133
134// `s` read as `A` by `decode`. A text `decode` refuses is `Failed`, not a wrong value.
135pub fn decoded<A: Shared + NoBorrow>(
136  s: delim::src::signal::Signal(Load),
137  decode: (Text) -> Option(A) ! {div}
138): delim::src::signal::Signal(Fetched(A)) ! {rt, div} {
139  memo(() => match get(s) {
140    Loading => Fetching
141    LoadFailed(why) => FetchFailed(why)
142    Loaded(v) => match decode(v) {
143      Some(a) => FetchedOk(a)
144      None => FetchFailed("the resource did not read back")
145    }
146  })
147}

149// The resource named `key`, read as `A` by `decode` (a codec's `decode` fits). It counts for
150// the suspense boundary like `resource`.
151pub fn resource_as<K: Shared + NoBorrow, P: Shared + NoBorrow, A: Shared + NoBorrow>(
152  t: Tree(K, P),
153  key: Text,
154  decode: (Text) -> Option(A) ! {div}
155): delim::src::signal::Signal(Fetched(A)) ! {rt, div} {
156  decoded(resource(t, key), decode)
157}

159// Like `request`, read as `A`: for a handler to ask for what it needs.
160pub fn request_as<A: Shared + NoBorrow>(
161  key: Text,
162  decode: (Text) -> Option(A) ! {div}
163): delim::src::signal::Signal(Fetched(A)) ! {rt, div} {
164  decoded(request(key), decode)
165}

167// Runs `f` once, with the value, when `s` becomes `Ready`; nothing runs if it fails. This is
168// how a handler waits: it asks with `request_as` and hands the rest of its work to
169// `on_ready`, so the handler itself never suspends and its effects stay `rt` and `Div`.
170pub fn on_ready<A: Shared + NoBorrow>(
171  s: delim::src::signal::Signal(Fetched(A)),
172  f: (A) -> Unit ! {rt, div}
173): Unit ! {rt, div, overflow} {
174  let done := perform ref_new(false)
175  watch(() => match get(s) {
176    FetchedOk(a) => if perform ref_get(done) { #() } else {
177      perform ref_set(done, true)
178      untracked(() => f(a))
179    }
180    Fetching => #()
181    FetchFailed(why) => #()
182  })
183}

185// A repeatable request. Every load uses a fresh Want but writes the same signal. Settling
186// the previous Want before replacing it prevents late responses from overwriting the new one.
187// One made by `reloadable_in` counts for the suspense boundary it was made under while a load
188// is pending, so that boundary shows its fallback again on `reload`.
189pub struct(shared) Reloadable {
190  state: delim::src::signal::Signal(Load),
191  pending: Ref(Option(Ref(Bool))),
192  active: Ref(Bool),
193  boundary: Option(delim::src::signal::Signal(u64))
194}
195
196// Ends the pending load, if any, without its result: its response is dropped when it comes,
197// and the boundary stops waiting for it. Whether it ended unsettled.
198fn drop_pending(r: Reloadable): Bool ! {rt, div} {
199  match perform ref_get(r.pending) {
200    None => false
201    Some(settled) => {
202      perform ref_set(r.pending, None)
203      if perform ref_get(settled) { false } else {
204        perform ref_set(settled, true)
205        match r.boundary {
206          None => #()
207          Some(c) => update(c, (n: u64) => handle { n - 1 } with { overflow() => 0 })
208        }
209        true
210      }
211    }
212  }
213}
214
215fn reloadable_with(boundary: Option(delim::src::signal::Signal(u64))): Reloadable ! {rt, div} {
216  let r := Reloadable {
217    state: signal(Loading),
218    pending: perform ref_new(None),
219    active: perform ref_new(true),
220    boundary: boundary
221  }
222  on_cleanup(() => {
223    perform ref_set(r.active, false)
224    let dropped := drop_pending(r)
225    #()
226  })
227  r
228}
229
230pub fn reloadable(): Reloadable ! {rt, div} { reloadable_with(None) }
231
232// A reloadable that counts for the innermost suspense boundary `t` is building, if any.
233pub fn reloadable_in<K: Shared + NoBorrow, P: Shared + NoBorrow>(
234  t: Tree(K, P)
235): Reloadable ! {rt, div} {
236  reloadable_with(match perform ref_get(t.boundaries) {
237    Nil => None
238    Cons(c, rest) => Some(c)
239  })
240}

242pub fn resource_state(r: Reloadable): delim::src::signal::Signal(Load) { r.state }

244pub fn cancel_resource(r: Reloadable): Unit ! {rt, div} {
245  if drop_pending(r) { set(r.state, LoadFailed("resource cancelled")) } else { #() }
246}

248pub fn reload(r: Reloadable, key: Text): Unit ! {rt, div} {
249  if not(perform ref_get(r.active)) { #() } else {
250    batch(() => {
251      // Retain only the response token, not the Want and its signal.
252      let dropped := drop_pending(r)
253      let settled := perform ref_new(false)
254      let w := Want { key: key, state: r.state, boundary: r.boundary, settled: settled }
255      perform ref_set(r.pending, Some(settled))
256      match r.boundary {
257        None => #()
258        Some(c) => update(c, (n: u64) => handle { n + 1 } with { overflow() => n })
259      }
260      let runtime := current()
261      perform ref_set(runtime.wants, Cons(w, perform ref_get(runtime.wants)))
262      set(r.state, Loading)
263    })
264  }
