3import std::builtin::par { Ref, ref_get, ref_new, ref_set }

5// Keeping a part of a view to run later (a handler when its event fires, a part built again
6// when the signals it reads change) under the handlers that stood where it was kept: the
7// contexts `with_context` gave, the tree `ui` gave. `keep` takes the rest of the build as a
8// continuation, so a part run later sees what the build saw, without each handler being
9// lifted by hand.
10
11pub(open) enum Kept { KNode(Node), KUnit, KNone }
12// What `fork` answers: in the part that goes on building, how to run the kept part again.
13pub(open) enum Forked { Child, Parent(() -> Kept ! {rt, div}) }
14enum Out(T: Type) { Built(T), RanNode(Node), RanUnit }
15
16pub effect fork(): Forked resume ω
17pub effect done_node(n: Node): Never resume 0
18pub effect done_unit(): Never resume 0
19
20// What keeping performs; `rooted` handles it.
21pub effect keeping { fork, done_node, done_unit }
22
23// `f`, to be run later with what stands where it was kept.
24pub fn keep<X: Shared + NoBorrow, !r>(
25  f: (x: X) -> Node ! {rt, div} ∪ r
26): ((x: X) -> Node ! {rt, div}) ! {keeping, rt, div} ∪ r {
27  let arg: Ref(Option(X)) := perform ref_new(None)
28  match perform fork() {
29    Child => match perform ref_get(arg) {
30      Some(x) => perform done_node(f(x))
31      None => perform done_node(Node { id: 0 })
32    }
33    Parent(run) => (x: X) => {
34      perform ref_set(arg, Some(x))
35      match run() { KNode(n) => n, KUnit => Node { id: 0 }, KNone => Node { id: 0 } }
36    }
37  }
38}

40// A handler `f`, to be run later with what stands where it was kept.
41pub fn keep_handler<X: Shared + NoBorrow, !r>(
42  f: (x: X) -> Unit ! {rt, div} ∪ r
43): ((x: X) -> Unit ! {rt, div}) ! {keeping, rt, div} ∪ r {
44  let arg: Ref(Option(X)) := perform ref_new(None)
45  match perform fork() {
46    Child => {
47      match perform ref_get(arg) { Some(x) => f(x), None => #() }
48      perform done_unit()
49    }
50    Parent(run) => (x: X) => {
51      perform ref_set(arg, Some(x))
52      match run() { KNode(_) => #(), KUnit => #(), KNone => #() }
53    }
54  }
55}

57// `f`, to be run later with what stands where it was kept.
58pub fn keep_run<!r>(
59  f: () -> Unit ! {rt, div} ∪ r
60): (() -> Unit ! {rt, div}) ! {keeping, rt, div} ∪ r {
61  match perform fork() {
62    Child => {
63      f()
64      perform done_unit()
65    }
66    Parent(run) => () => match run() { KNode(_) => #(), KUnit => #(), KNone => #() }
67  }
68}

70// A value `f` gives, built later with what stands where it was kept.
71pub fn keep_value<A: Shared + NoBorrow, !r>(
72  f: () -> A ! {rt, div} ∪ r
73): (() -> A ! {rt, div}) ! {keeping, rt, div} ∪ r {
74  let out: Ref(Option(A)) := perform ref_new(None)
75  let h := keep_handler((u: Unit) => perform ref_set(out, Some(f())))
76  () => {
77    h(#())
78    match perform ref_get(out) {
79      Some(v) => v
80      // `h` always sets it; this is not reached.
81      None => f_unreached(out)
82    }
83  }
84}
85fn f_unreached<A: Shared + NoBorrow>(out: Ref(Option(A))): A ! {div, ref_get} {
86  match perform ref_get(out) { Some(v) => v, None => f_unreached(out) }
87}

89// The root of a view: what `body` keeps runs later, after this returns. Every handler that
90// should stand around a kept part (contexts, the tree) is inside `body`; nothing outside.
91pub fn rooted<T>(body: () -> T ! {keeping, rt, div}): Option(T) ! {rt, div} {
92  let out := handle { Built(body()) } with {
93    fork(), k => k(Parent(() => match k(Child) {
94      RanNode(n) => KNode(n)
95      RanUnit => KUnit
96      Built(x) => KNone
97    }))
98    done_node(n) => RanNode(n)
99    done_unit() => RanUnit
100    return x => x
101  }
102  match out { Built(x) => Some(x), RanNode(_) => None, RanUnit => None }
103}

105// The tree a view builds on. `with_tree` answers it; a part kept with `keep` gets the same tree
106// when it runs later.
107pub effect ui<K: Shared + NoBorrow, P: Shared + NoBorrow>(): Tree(K, P) direct

109// `body` on the tree `t`.
110pub fn with_tree<K: Shared + NoBorrow, P: Shared + NoBorrow, A, !f>(
111  t: Tree(K, P),
112  body: () -> A ! {ui(K, P)} ∪ f
113): A ! f {
114  handle { body() } with {
115    ui<K, P>(), k => k(t)
116    return value => value
117  }
