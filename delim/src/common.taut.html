1// What components shared by every renderer may use: a few element kinds and properties that
2// each renderer gives its own form, in a `Vocab`. The vocabulary is kept apart from the
3// `Tree`: the tree holds handlers, and a handler may hold the vocabulary.
4
5// What an element is to assistive technology. An element a person can act on carries its
6// name, so an unnamed button or link cannot be written.
7pub(open) enum Role {
8  Button(Text)
9  Link(Text)
10  Checkbox(Text)
11  TextField(Text)
12  Tab(Text)
13  MenuItem(Text)
14  Heading(u64)
15  ListBox
16  ListEntry
17  Region(Text)
18  Presentation
19}
20
21pub(open) struct Vocab(K: Type, P: Type) {
22  // Boxes laying out their children in a column and in a row, something to press, and an
23  // input field.
24  column: K,
25  row: K,
26  pressable: K,
27  input: K,
28  role: (r: Role) -> P,
29  // A further name for an element nobody acts on, such as an image.
30  name: (t: Text) -> P,
31  disabled: (b: Bool) -> P,
32  // The text an input field shows.
33  value: (t: Text) -> P
34}
