4// Conventions for components. A component is a function from its props and its children
5// (`List(Node)`) to a `Node`; what it needs from around it shows in its effect row:
6// `rt` for signals, `context(A)` for a context (`context.taut`), `fail(Text)` for an error
7// an `error_boundary` above it can catch.
8
9// `body()`, or `fallback(why)` if `body` fails with `fail(why)` before it returns. The
10// nodes `body` made before failing were never put anywhere, so none of them is shown.
11pub fn error_boundary_in<K: Shared + NoBorrow, P: Shared + NoBorrow, !e>(
12  t: Tree(K, P),
13  body: () -> Node ! {rt, div, fail(Text)} ∪ e,
14  fallback: (Text) -> Node ! {rt, div} ∪ e
15): Node ! {rt, div} ∪ e {
16  handle { body() } with {
17    fail<Text>(why) => fallback(why)
18    return n => n
19  }
20}

22// Puts what `body` builds under `target`, another node (a modal layer, say), instead of
23// where the portal stands, and takes it away when the portal's owner ends. The portal itself
24// leaves an empty group in its place. Events of the moved nodes reach their handlers as for
25// any node: handlers belong to the tree, not to the place.
26pub fn portal<K: Shared + NoBorrow, P: Shared + NoBorrow>(
27  t: Tree(K, P),
28  target: Node,
29  body: () -> Node ! {rt, div}
30): Node ! {rt, div, overflow} {
31  let g := node_of(t)
32  emit(t, CreateGroup(g))
33  let n := body()
34  emit(t, Append(target, n))
35  on_cleanup(() => emit(t, Remove(n)))
36  g
37}
