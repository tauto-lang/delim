1import std::builtin::par { Ref, ref_get, ref_new, ref_set }
2import "signal.taut" { Signal, get, memo_eq, peek, rt, set, signal }
3
4// An Elm-style store: the state that matters lives in one model, changed only by a pure
5// `update`, so its invariants can be proved about `update` alone. Views read slices of the
6// model with `select`, which runs its readers only when the slice changes.
7//
8// Work that waits (a request, a timer) is not done here: `update` returns it as a job, and
9// the renderer's loop runs the jobs and sends their results back with `dispatch`.
10
11pub(open) struct Next(M: Type, J: Type) { model: M, jobs: List(J) }
12
13pub fn just<M, J>(m: M): Next(M, J) { Next { model: m, jobs: Nil } }
14
15struct Update(M: Type, Msg: Type, J: Type) { run: (M, Msg) -> Next(M, J) ! {div, overflow} }
16
17pub struct(shared) Store(M: Shared + NoBorrow, Msg: Shared + NoBorrow, J: Shared + NoBorrow) {
18  model: delim::src::signal::Signal(M),
19  update: Ref(Update(M, Msg, J)),
20  // The jobs `update` gave and the loop has not taken yet, newest first.
21  jobs: Ref(List(J))
22}
23
24pub fn store<M: Shared + NoBorrow, Msg: Shared + NoBorrow, J: Shared + NoBorrow>(
25  init: M,
26  update: (M, Msg) -> Next(M, J) ! {div, overflow}
27): Store(M, Msg, J) ! {rt} {
28  Store {
29    model: signal(init),
30    update: perform ref_new(Update { run: (m, msg) => update(m, msg) }),
31    jobs: perform ref_new(Nil)
32  }
33}
34
35fn push_jobs<J: Shared + NoBorrow>(l: List(J), acc: List(J)): List(J) {
36  match l {
37    Nil => acc
38    Cons(j, rest) => push_jobs(rest, Cons(j, acc))
39  }
40}
41
42// Sends `msg` to `update` and writes the new model.
43pub fn dispatch<M: Shared + NoBorrow, Msg: Shared + NoBorrow, J: Shared + NoBorrow>(
44  s: Store(M, Msg, J),
45  msg: Msg
46): Unit ! {rt, div} {
47  let u := perform ref_get(s.update)
48  let next := (u.run)(peek(s.model), msg)
49  perform ref_set(s.jobs, push_jobs(next.jobs, perform ref_get(s.jobs)))
50  set(s.model, next.model)
51}
52
53// The jobs given since the last call, oldest first.
54pub fn take_jobs<M: Shared + NoBorrow, Msg: Shared + NoBorrow, J: Shared + NoBorrow>(
55  s: Store(M, Msg, J)
56): List(J) ! {ref_get, ref_set} {
57  let js := perform ref_get(s.jobs)
58  perform ref_set(s.jobs, Nil)
59  push_jobs(js, Nil)
60}
61
62pub fn model<M: Shared + NoBorrow, Msg: Shared + NoBorrow, J: Shared + NoBorrow>(
63  s: Store(M, Msg, J)
64): M ! {rt, overflow} { get(s.model) }
65
66// The slice `f` of the model, followed: its readers run only when the slice changes.
67pub fn select<M: Shared + NoBorrow, Msg: Shared + NoBorrow, J: Shared + NoBorrow, A: Shared + NoBorrow>(
68  s: Store(M, Msg, J),
69  f: (M) -> A ! {div, overflow}
70): delim::src::signal::Signal(A) ! {rt, div} where DecEq(A) {
71  memo_eq(() => f(get(s.model)))
72}
