38pub(open) enum ActionPolicy { IgnoreWhilePending, LatestWins }
39pub(open) effect action_live(): Bool direct
40pub fn commit_current(body: () -> Unit ! {rt, div}): Unit ! {rt, div, action_live} {
41  if perform action_live() { body() } else { #() }
42}
43// `on_cancel(f)` in a run: `f()` runs if the run ends without settling (stopped, invalidated,
44// replaced by a newer run, or its owner disposed), and not once it has settled.
45pub(open) effect action_on_cancel(): Ref(Bool) direct
46pub fn on_cancel(f: () -> Unit ! {rt, div}): Unit ! {rt, action_on_cancel} {
47  // Whether the run has settled, read when its owner ends.
48  let settled := perform action_on_cancel()
49  on_cleanup(() => if perform ref_get(settled) { #() } else { f() })
50}
51pub(open) struct ActionOptions(
52  A: Shared + NoBorrow,
53  B: Shared + NoBorrow,
54  E: Shared + NoBorrow,
55  I: Shared + NoBorrow,
56  R: Shared + NoBorrow
57) {
58  policy: ActionPolicy,
59  // What the action's input depends on: when what it reads changes, the current run is
60  // cancelled and its result cleared. It never starts a run.
61  invalidate_on: () -> I ! {rt, div},
62  run: (input: A) -> B ! {
63    rt,
64    div,
65    resource_wait(R, E),
66    resource_cancelled,
67    fail(E),
68    action_live,
69    action_on_cancel
70  },
71  // Runs once when a run succeeds, before its `Ready` is published; never after it is cancelled.
72  on_success: (value: B) -> Unit ! {rt, div}
73}

75// Starts loading the resource named `key` now; reading it is independent of where it started.
76pub fn use_res(key: Text): Resource(Text, Text) ! {rt, div} {
77  let load := request(key)
78  resource_handle(memo(() => match get(load) {
79    Loading => Pending
80    Loaded(value) => Ready(value)
81    LoadFailed(why) => Failed(why)
82  }))
83}
84pub(open) struct(shared) Action(A: Shared + NoBorrow, B: Shared + NoBorrow, E: Shared + NoBorrow) {
85  start: (input: A) -> Unit ! {rt, div},
86  cancel: () -> Unit ! {rt, div},
87  clear: () -> Unit ! {rt, div},
88  pending: () -> Bool ! {rt},
89  result: () -> Option(Resource(B, E)) ! {rt}
90}
91// Whether the current run still counts is read in place: this frame holds no suspended run.
92fn with_live<B: Shared + NoBorrow, E: Shared + NoBorrow, R: Shared + NoBorrow>(
93  live: Ref(Bool),
94  settled: Ref(Bool),
95  result: delim::src::signal::Signal(ResourceState(B, E)),
96  body: affine () -> Unit ! {
97    rt,
98    div,
99    resource_wait(R, E),
100    resource_cancelled,
101    fail(E),
102    action_live,
103    action_on_cancel
104  }
105): Unit ! {rt, div, resource_wait(R, E), resource_cancelled, fail(E)} {
106  handle { body() } with {
107    action_live(), k => k(perform ref_get(live))
108    action_on_cancel(), k => k(settled)
109    return value => value
110  }
111}
112fn resume_action<B: Shared + NoBorrow, E: Shared + NoBorrow, R: Shared + NoBorrow>(
113  body: affine () -> Unit ! {rt, div, resource_wait(R, E), resource_cancelled, fail(E)},
114  live: Ref(Bool),
115  settled: Ref(Bool),
116  result: delim::src::signal::Signal(ResourceState(B, E))
117): Unit ! {rt, div} {
118  handle {
119    body()
120  } with {
121    resource_wait<R, E>(resource), k => {
122      // The affine continuation is consumed once, even if the resource changes again.
123      // The owner's cleanup holds the continuation; the watcher only looks at it.
124      let continuation := perform ref_new(perform mutex_new(Some(k)))
125      on_cleanup(() => { let held := perform ref_get(continuation); #() })
126      let waiting := perform weak_of(continuation)
127      watch(() => match match state(resource) { Pending => false, Ready(_) => true, Failed(_) => true, Cancelled => true } {
128        false => #()
129        true => match perform weak_get(waiting) {
130          None => #()
131          Some(slot) => {
132            var guard := Mutex::lock(perform ref_get(slot))
133            let next := replace(MutexGuard::get_mut(&mut guard), None)
134            match next {
135              Some(continue_once) => if perform ref_get(live) {
136                // The run waits on `R` again: the written type says so where nothing else does.
137                let again: affine () -> Unit ! {rt, div, resource_wait(R, E), resource_cancelled, fail(E)} := () => continue_once(#())
138                resume_action(again, live, settled, result)
139              } else { #() }
140              None => #()
141            }
142          }
143        }
144      })
145    }
146    fail<E>(error) => if perform ref_get(live) {
147      perform ref_set(settled, true); set(result, Failed(error)); perform ref_set(live, false)
148    } else { #() }
149    resource_cancelled() => if perform ref_get(live) {
150      set(result, Cancelled); perform ref_set(live, false)
151    } else { #() }
152    return value => value
153  }
154}
155pub fn use_action<A: Shared + NoBorrow, B: Shared + NoBorrow, E: Shared + NoBorrow, I: Shared + NoBorrow, R: Shared + NoBorrow>(
156  options: ActionOptions(A, B, E, I, R)
157): Action(A, B, E) ! {rt, div} {
158  let latest: delim::src::signal::Signal(
159    Option(delim::src::signal::Signal(ResourceState(B, E)))
160  ) := signal(None)
161  let live := perform ref_new(perform ref_new(false))
162  let child := perform ref_new(None)
163  let cancel := () => {
164    perform ref_set(perform ref_get(live), false)
165    match peek(latest) {
166      Some(value) => match peek(value) {
167        Pending => set(value, Cancelled),
168        Ready(_) => #(),
169        Failed(_) => #(),
170        Cancelled => #()
171      }
172      None => #()
173    }
174    match perform ref_get(child) {
175      Some(owner) => { perform ref_set(child, None); dispose(owner) },
176      None => #()
177    }
178  }
179  on_cleanup(cancel)
180  let watched := perform ref_new(false)
181  watch(() => {
182    let input := (options.invalidate_on)()
183    if perform ref_get(watched) { untracked(() => { cancel(); set(latest, None) }) } else { perform ref_set(watched, true) }
184  })
185  let owner := perform ref_get(current().owner)
186  Action {
187    start: input => {
188      let busy := match peek(latest) {
189        Some(value) => match peek(value) {
190          Pending => true,
191          Ready(_) => false,
192          Failed(_) => false,
193          Cancelled => false
194        },
195        None => false
196      }
197      if busy && options.policy == IgnoreWhilePending { #() } else {
198        cancel()
199        let active := perform ref_new(true)
200        perform ref_set(live, active)
201        let value: delim::src::signal::Signal(ResourceState(B, E)) := signal(Pending)
202        set(latest, Some(value))
203        let runtime := current()
204        let previous := perform ref_get(runtime.owner)
205        perform ref_set(runtime.owner, owner)
206        let settled := perform ref_new(false)
207        let invocation := scope(() => resume_action(() => with_live(active, settled, value, () => {
208          let answer := (options.run)(input)
209          if perform ref_get(active) { perform ref_set(settled, true); (options.on_success)(answer); set(value, Ready(answer)); perform ref_set(active, false) } else { #() }
210        }), active, settled, value))
211        perform ref_set(runtime.owner, previous)
212        perform ref_set(child, Some(invocation.owner))
213      }
214    },
215    cancel: cancel,
216    clear: () => { cancel(); set(latest, None) },
217    pending: () => match get(latest) {
218      Some(value) => match get(value) {
219        Pending => true,
220        Ready(_) => false,
221        Failed(_) => false,
222        Cancelled => false
223      },
224      None => false
225    },
226    result: () => match get(latest) { Some(value) => Some(resource_handle(value)), None => None }
227  }
228}
