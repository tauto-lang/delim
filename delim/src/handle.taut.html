3// A read-only shared view of a single operation. Cancellation authority stays with its owner.
4pub(open) enum ResourceState(A: Type, E: Type) {
5  Pending
6  Ready(A)
7  Failed(E)
8  Cancelled
9}
10pub struct(shared) Resource(
11  A: Shared + NoBorrow,
12  E: Shared + NoBorrow
13) { value: Signal(ResourceState(A, E)) }
14pub fn resource_handle<A: Shared + NoBorrow, E: Shared + NoBorrow>(
15  value: Signal(ResourceState(A, E))
16): Resource(A, E) {
17  Resource { value: value }
18}
19pub fn state<A: Shared + NoBorrow, E: Shared + NoBorrow>(
20  r: Resource(A, E)
21): ResourceState(A, E) ! {rt} { get(r.value) }
22pub fn resolved_resource<A: Shared + NoBorrow, E: Shared + NoBorrow>(value: A): Resource(
23  A,
24  E
25) ! {rt} {
26  resource_handle(signal(Ready(value)))
27}
28pub fn failed_resource<A: Shared + NoBorrow, E: Shared + NoBorrow>(error: E): Resource(A, E) ! {rt} {
29  resource_handle(signal(Failed(error)))
30}
31// A host/task/display handler decides how to wait. Resuming means the state changed;
32// read checks it again and never mistakes a spurious wakeup for a result.
33pub(open) effect resource_wait<A: Shared + NoBorrow, E: Shared + NoBorrow>(
34  resource: Resource(A, E)
35): Unit
36pub(open) effect resource_cancelled(): Never resume 0
37pub fn read<A: Shared + NoBorrow, E: Shared + NoBorrow>(
38  r: Resource(A, E)
39): A ! {rt, div, resource_wait(A, E), resource_cancelled, fail(E), overflow} {
40  match state(r) {
41    Ready(value) => value
42    Failed(error) => perform fail(error)
43    Cancelled => perform resource_cancelled()
44    Pending => { perform resource_wait(r); read(r) }
45  }
46}
47pub fn join<A: Shared + NoBorrow, B: Shared + NoBorrow, E: Shared + NoBorrow>(
48  a: Resource(A, E),
49  b: Resource(B, E)
50): Resource(A × B, E) ! {rt, div} {
51  let result := signal(Pending)
52  watch(() => match peek(result) {
53    Pending => {
54      let next := match state(a) {
55        Failed(e) => Failed(e)
56        Cancelled => Cancelled
57        Ready(x) => match state(b) {
58          Failed(e) => Failed(e)
59          Cancelled => Cancelled
60          Ready(y) => Ready(#(x, y))
61          Pending => Pending
62        }
63        Pending => match state(b) {
64          Failed(e) => Failed(e)
65          Cancelled => Cancelled
66          Ready(_) => Pending
67          Pending => Pending
68        }
69      }
70      match next {
71        Pending => #()
72        Ready(v) => set(result, Ready(v))
73        Failed(e) => set(result, Failed(e))
74        Cancelled => set(result, Cancelled)
75      }
76    }
77    Ready(_) => #()
78    Failed(_) => #()
79    Cancelled => #()
80  })
81  resource_handle(result)
82}
83// Winner is latched at the first observed terminal outcome; shared losers are untouched.
84pub(open) enum Selected(A: Type, B: Type) { First(A), Second(B) }
85pub fn select<A: Shared + NoBorrow, B: Shared + NoBorrow, E: Shared + NoBorrow>(
86  a: Resource(A, E),
87  b: Resource(B, E)
88): Resource(Selected(A, B), E) ! {rt, div} {
89  let result := signal(Pending)
90  watch(() => match peek(result) {
91    Pending => match state(a) {
92      Ready(x) => set(result, Ready(First(x)))
93      Failed(e) => set(result, Failed(e))
94      Cancelled => set(result, Cancelled)
95      Pending => match state(b) {
96        Ready(y) => set(result, Ready(Second(y)))
97        Failed(e) => set(result, Failed(e))
98        Cancelled => set(result, Cancelled)
99        Pending => #()
100      }
101    }
102    Ready(_) => #()
103    Failed(_) => #()
104    Cancelled => #()
105  })
106  resource_handle(result)
107}
