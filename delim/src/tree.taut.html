24// The element layer: elements, their properties and events, and the nodes whose shape
25// changes (`show`, `switch_on`, `keyed_each`). It does not draw. It keeps a retained tree of
26// numbered nodes and appends each change to a list of patches, which a renderer (the DOM,
27// a native toolkit, or the headless one in this file) applies. `K` is the type of element
28// kinds and `P` the type of properties of that renderer.
29
30// A node of the retained tree. Only `Tree` hands out numbers, and a number is not reused.
31pub(open) struct(shared) Node { id: u32 }
32
33// The kind of an event a handler is registered for.
34pub(open) enum EventKind {
35  Press
36  Focus
37  Blur
38  Key
39  // The text of an input field was committed.
40  Edit
41  PointerEnter
42  PointerLeave
43  PointerDown
44  PointerMove
45  PointerUp
46  Scroll
47  // A form was submitted; the page does not navigate.
48  Submit
49  // A named action, such as one an assistive technology asks for.
50  NamedAction(Text)
51  // A drag of an element the renderer was told is a drag source: it has moved far enough to
52  // be a drag (`DragStart`, with where it started), it moves (`DragMove`), and it ends
53  // (`DragEnd`), each delivered to the source with the pointer's position. Where it was let
54  // go, the node under the pointer gets `Drop` first.
55  DragStart
56  DragMove
57  DragEnd
58  Drop
59  // A key was let go, with the key it names.
60  KeyUp
61  // Two presses in quick succession.
62  DoubleClick
63  // The context menu was asked for, as by a right press; the browser's own menu still opens.
64  ContextMenu
65  // The wheel turned, with how far across and down (`ScrollData`).
66  Wheel
67  // Text was pasted, with the plain text pasted (`TextData`).
68  Paste
69}

74  NoData
75  KeyData(Text)
76  TextData(Text)
77  Pointer(i64, i64, i64)
78  ScrollData(i64, i64)

83  match d {
84    TextData(s) => s,
85    KeyData(s) => s,
86    NoData => "",

88    ScrollData(_, _) => ""

92// A property of an element: fixed, following the signals its function reads, or a handler.
93// A handler only reads and writes signals; work that waits goes through a store's jobs.
94pub(open) enum Prop(P: Type) {
95  Static(P)
96  Dyn(() -> P ! {rt, div})
97  On(EventKind, (d: EventData) -> Unit ! {rt, div})
98  // A handler the signal runtime keeps, by the id `hold` gave; it reads the event's data
99  // with `current_event`. The node takes it over: it is released when the node's owner ends.
100  Held(EventKind, u64)
101}

103// The changes to the retained tree, in the order a renderer applies them.
104//
105// - A node is created detached, by `Create`, `CreateText` or `CreateGroup`, and numbered once.
106// - `Append` and `InsertBefore` on a node that already has a parent move it.
107// - `Remove` detaches a node with all of its subtree, and every node in it is dead: no patch
108//   names it again. Only the root of what was removed gets a `Remove`, not its descendants.
109// - A group has no box of its own; its children stand in its place in its parent.
110pub(open) enum Patch(K: Type, P: Type) {
111  Create(Node, K)
112  CreateText(Node, Text)
113  // A node with no box of its own, standing for the nodes `show`, `switch_on` and `keyed_each`
114  // put in it.
115  CreateGroup(Node)
116  SetProp(Node, P)
117  SetText(Node, Text)
118  // Puts `child` last among the children of `parent`.
119  Append(Node, Node)
120  // Puts `child` right before `before`, a child of `parent`.
121  InsertBefore(Node, Node, Node)
122  Remove(Node)
123  // The subtree under the node, made by `template`, has the shape every other instance of
124  // the template has: a renderer may build the first one and copy it for the others.
125  Template(u64, Node)
126}

128// A handler of the tree: its function is kept by the signal runtime under `held`, and reads
129// the event's data from the tree's `event`, so that the tree holds no function.

132pub(open) struct(shared) Tree(K: Shared + NoBorrow, P: Shared + NoBorrow) {
133  next: Ref(u32),
134  patches: Ref(List(Patch(K, P))),
135  handlers: Ref(List(Handler)),
136  // The data of the event being delivered.
137  event: Ref(EventData),
138  // The resources still loading under each suspense boundary being built, innermost first.
139  boundaries: Ref(List(delim::src::signal::Signal(u64))),

141}

246fn run_handlers(l: List(Handler), n: u32, k: EventKind): Unit ! {rt, div} {
247  match l {
248    Nil => #()
249    Cons(h, rest) => {
250      if h.node == n && same_kind(h.kind, k) { run_held(h.held) } else { #() }
251      run_handlers(rest, n, k)
252    }
253  }

265pub fn fire<K: Shared + NoBorrow, P: Shared + NoBorrow>(
266  t: Tree(K, P),
267  n: Node,
268  k: EventKind,
269  d: EventData
270): Unit ! {rt, div} {

276// Whether an event of the kind goes from the node it happened on up to the nearest node that
277// has a handler for it, as the DOM's events bubble: a press, a key, an edit, a submit, a
278// pointer going down, moving or up, a named action, a drop, a key let go, a double press, a
279// context menu, a wheel turn and a paste. A renderer that knows the
280// parents of its nodes finds that node with `handler_nodes`; other events go to the node.
281pub fn bubbles(k: EventKind): Bool ! {div} {
282  let c := kind_code(k)
283  c == 0 || c == 3 || c == 4 || c == 7 || c == 8 || c == 9 || c == 11 || c == 12 || c >= 16
284}

308fn drop_handler(l: List(Handler), held: u64): List(Handler) {
309  match l {
310    Nil => Nil
311    Cons(h, rest) => if h.held == held { rest } else {
312      Cons(h, drop_handler(rest, held))
313    }
314  }
315}

317// Removes the subtree root when the current owner ends. Each handler is separately
318// owned by the scope that registered it, including handlers on descendant nodes.
319fn remove_on_cleanup<K: Shared + NoBorrow, P: Shared + NoBorrow>(
320  t: Tree(K, P),
321  n: Node
322): Unit ! {rt} {
323  on_cleanup(() => {
324    emit(t, Remove(n))
325  })
326}

328fn add_handler<K: Shared + NoBorrow, P: Shared + NoBorrow>(
329  t: Tree(K, P),
330  n: Node,
331  e: EventKind,
332  held: u64
333): Unit ! {rt} {
334  perform ref_set(
335    t.handlers,
336    Cons(Handler { node: n.id, kind: e, held: held }, perform ref_get(t.handlers))
337  )
338  // A node inside a part that ends is not removed itself (only the part's root is),
339  // so each handler is forgotten when the owner it was made under ends.
340  let handlers := t.handlers
341  on_cleanup(() => {
342    release(held)
343    perform ref_set(handlers, drop_handler(perform ref_get(handlers), held))
344  })
345}

347fn apply_props<K: Shared + NoBorrow, P: Shared + NoBorrow>(
348  t: Tree(K, P),
349  n: Node,
350  props: List(Prop(P))
351): Unit ! {rt, div, overflow} where DecEq(P) {
352  match props {
353    Nil => #()
354    Cons(p, rest) => {
355      match p {
356        Static(v) => emit(t, SetProp(n, v))
357        Dyn(f) => {
358          let v := memo_eq(() => f())
359          watch(() => emit(t, SetProp(n, get(v))))
360        }
361        On(e, h) => {
362          let event := t.event
363          add_handler(t, n, e, hold(() => h(perform ref_get(event))))
364        }
365        Held(e, held) => add_handler(t, n, e, held)
366      }
367      apply_props(t, n, rest)
368    }
369  }
370}

421// Text that follows the signals `f` reads.
422pub fn text_dyn<K: Shared + NoBorrow, P: Shared + NoBorrow>(
423  t: Tree(K, P),
424  f: () -> Text ! {rt, div}
425): Node ! {rt, div, overflow} {
426  let n := node_of(t)
427  let v := memo_eq(() => f())
428  emit(t, CreateText(n, peek(v)))
429  // The first run only subscribes: the text was created with the value.
430  let first := perform ref_new(true)
431  watch(() => {
432    let s := get(v)
433    if perform ref_get(first) { perform ref_set(first, false) } else { emit(t, SetText(n, s)) }
434  })
435  n
436}

438// `build()` while `cond()` holds, nothing otherwise. What `build` made ends when the
439// condition turns false.
440pub fn show<K: Shared + NoBorrow, P: Shared + NoBorrow>(
441  t: Tree(K, P),
442  cond: () -> Bool ! {rt, div},
443  build: () -> Node ! {rt, div}
444): Node ! {rt, div, overflow} {
445  let g := node_of(t)
446  emit(t, CreateGroup(g))
447  let c := memo_eq(() => cond())
448  watch(() => {
449    if get(c) {
450      let n := untracked(() => build())
451      emit(t, Append(g, n))
452      remove_on_cleanup(t, n)
453    } else { #() }
454  })
455  g
456}

458// `build(v)` for the value `v` of `f()`, made again only when the value changes.
459pub fn switch_on<K: Shared + NoBorrow, P: Shared + NoBorrow, V: Shared + NoBorrow>(
460  t: Tree(K, P),
461  f: () -> V ! {rt, div},
462  build: (V) -> Node ! {rt, div}
463): Node ! {rt, div, overflow} where DecEq(V) {
464  let g := node_of(t)
465  emit(t, CreateGroup(g))
466  let v := memo_eq(() => f())
467  watch(() => {
468    let x := get(v)
469    let n := untracked(() => build(x))
470    emit(t, Append(g, n))
471    remove_on_cleanup(t, n)
472  })
473  g
474}

476// Like `switch_on`, without comparing values: what `f` reads changing builds again, even to the
477// same value. It is what `@match` makes, so a handle (whose value cannot be compared) picks its
478// branch too; wrap `f` in `memo_eq` to build again only for a different value.
479pub fn switch_on_change<K: Shared + NoBorrow, P: Shared + NoBorrow, V: Shared + NoBorrow>(
480  t: Tree(K, P),
481  f: () -> V ! {rt, div},
482  build: (V) -> Node ! {rt, div}
483): Node ! {rt, div, overflow} {
484  let g := node_of(t)
485  emit(t, CreateGroup(g))
486  watch(() => {
487    let x := f()
488    let n := untracked(() => build(x))
489    emit(t, Append(g, n))
490    remove_on_cleanup(t, n)
491  })
492  g
493}

500// The rows by key, so finding the row of a key does not walk the rows.
501fn rows_by_key(m: &mut HashMap(u64, KeyedRow), rows: List(KeyedRow)): Unit {
502  match rows {
503    Nil => #()
504    Cons(r, rest) => {
505      let old := HashMap::insert(&mut m, r.key, r)
506      rows_by_key(&mut m, rest)
507    }
508  }
509}

511fn find_row(m: &HashMap(u64, KeyedRow), k: u64): Option(KeyedRow) {
512  match HashMap::get(&m, &k) {
513    None => None
514    Some(r) => Some(own(&r))
515  }

518// The keys of `items`, so a row asks whether its key stays without walking the items.
519fn keys_of<A: Shared + NoBorrow>(
520  m: &mut HashMap(u64, u64),

523): Unit ! {div} {
524  match l {
525    Nil => #()
526    Cons(x, rest) => {

528      keys_of(&mut m, rest, key)

530  }

540// Ends the rows whose key is not among `keys`.
541fn end_missing<K: Shared + NoBorrow, P: Shared + NoBorrow>(
542  t: Tree(K, P),
543  rows: List(KeyedRow),
544  keys: &HashMap(u64, u64)
545): Unit ! {rt, div, overflow} {
546  match rows {
547    Nil => #()
548    Cons(r, rest) => {
549      if has_seen(&keys, r.key) { #() } else {
550        dispose(r.owner)
551        emit(t, Remove(r.node))
552      }
553      end_missing(t, rest, &keys)
554    }
555  }
556}

558// The rows for `items` in order, keeping the row of a key that was there before and making
559// the others; newest first.
560fn has_seen(m: &HashMap(u64, u64), k: u64): Bool {
561  match at(&m, k) {
562    None => false
563    Some(v) => true
564  }
565}

567fn key_reported(keys: List(u64), k: u64): Bool {
568  match keys {
569    Nil => false
570    Cons(x, rest) => x == k || key_reported(rest, k)
571  }
572}

580  perform ref_set(t.duplicate_keys, Nil)
581  keys

584// An item whose key an earlier item already has is left out: one key, one row.
585fn rows_for<A: Shared + NoBorrow>(
586  old: &HashMap(u64, KeyedRow),
587  items: List(A),
588  key: (A) -> u64 ! {div},
589  row: (A) -> Node ! {rt, div},
590  duplicates: Ref(List(u64)),
591  seen: &mut HashMap(u64, u64),
592  acc: List(KeyedRow)
593): List(KeyedRow) ! {rt, div} {
594  match items {
595    Nil => acc
596    Cons(x, rest) => {
597      let k := key(x)
598      if has_seen(&seen, k) {
599        let reported := perform ref_get(duplicates)
600        if key_reported(reported, k) { #() } else { perform ref_set(duplicates, Cons(k, reported)) }
601        rows_for(&old, rest, key, row, duplicates, &mut seen, acc)
602      } else {
603        put(&mut seen, k, 0)
604        let r := match find_row(&old, k) {
605          Some(r) => r
606          None => {
607            let made := scope(() => untracked(() => row(x)))
608            KeyedRow { key: k, owner: made.owner, node: made.value }
609          }
610        }
611        rows_for(&old, rest, key, row, duplicates, &mut seen, Cons(r, acc))
612      }
613    }
614  }
615}

652// The positions in `rows` that stay where they are: a longest run of rows whose old
653// positions increase. Every other row is moved, so the number of moves is the least.
654fn staying(
655  old: List(KeyedRow),
656  rows: List(KeyedRow)
657): HashMap(u64, u64) ! {div, overflow, hash_seed} {
658  var old_at: HashMap(u64, u64) := HashMap::new()
659  index_rows(&mut old_at, old, 0)
660  // seq: the new position → the old position, for the rows that were there before.
661  var seq: HashMap(u64, u64) := HashMap::new()
662  var n: u64 := 0
663  var rest := rows
664  var going := true
665  while going {
666    match rest {
667      Nil => { going := false }
668      Cons(r, more) => {
669        match at(&old_at, r.key) {
670          None => #()
671          Some(o) => put(&mut seq, n, o)
672        }
673        n := n + 1
674        rest := more
675      }
676    }
677  }
678  // Patience sorting: tails(l) is the new position ending the best run of length l + 1.
679  var tails: HashMap(u64, u64) := HashMap::new()
680  var prev: HashMap(u64, u64) := HashMap::new()
681  var len: u64 := 0
682  var i: u64 := 0
683  while i < n {
684    match at(&seq, i) {
685      None => #()
686      Some(v) => {
687        var lo: u64 := 0
688        var hi: u64 := len
689        while lo < hi {
690          let mid := div(lo + hi, 2)
691          let tv := match at(&tails, mid) {
692            None => 0
693            Some(p) => match at(&seq, p) {
694              None => 0
695              Some(x) => x
696            }
697          }
698          if tv < v { lo := mid + 1 } else { hi := mid }
699        }
700        if lo > 0 {
701          match at(&tails, lo - 1) {
702            None => #()
703            Some(p) => put(&mut prev, i, p)
704          }
705        } else { #() }
706        put(&mut tails, lo, i)
707        if lo == len { len := len + 1 } else { #() }
708      }
709    }
710    i := i + 1
711  }
712  var keep: HashMap(u64, u64) := HashMap::new()
713  if len > 0 {
714    var cur := at(&tails, len - 1)
715    var walking := true
716    while walking {
717      match cur {
718        None => { walking := false }
719        Some(p) => {
720          put(&mut keep, p, 1)
721          cur := at(&prev, p)
722        }
723      }
724    }
725  } else { #() }
726  keep
727}

729fn reverse_indexed(
730  l: List(KeyedRow),
731  i: u64,
732  acc: List(u64 × KeyedRow)
733): List(u64 × KeyedRow) ! {overflow} {
734  match l {
735    Nil => acc
736    Cons(r, rest) => reverse_indexed(rest, i + 1, Cons(#(i, r), acc))
737  }
738}

740// Puts the rows of `g` in the order of `rows`, from the last one back, moving only the rows
741// that are not staying.
742fn place_rows<K: Shared + NoBorrow, P: Shared + NoBorrow>(
743  t: Tree(K, P),
744  g: Node,
745  back: List(u64 × KeyedRow),
746  keep: &HashMap(u64, u64),
747  anchor: Option(Node)
748): Unit ! {div, ref_get, ref_set} {
749  match back {
750    Nil => #()
751    Cons(p, rest) => {
752      match at(&keep, (p: u64)) {
753        Some(x) => #()
754        None => match anchor {
755          None => emit(t, Append(g, (p: KeyedRow).node))
756          Some(a) => emit(t, InsertBefore(g, (p: KeyedRow).node, a))
757        }
758      }
759      place_rows(t, g, rest, &keep, Some((p: KeyedRow).node))
760    }
761  }
762}

764// A node per item of `items()`, matched by `key`: an item whose key was there before keeps
765// its node and what it made; a new key gets `row(item)`; a key that left ends with its node.
766// Of items with the same key, only the first gets a row. After a change only the rows
767// outside a longest run that kept its order are moved.
768pub fn keyed_each<K: Shared + NoBorrow, P: Shared + NoBorrow, A: Shared + NoBorrow>(
769  t: Tree(K, P),
770  items: () -> List(A) ! {rt, div},
771  key: (A) -> u64 ! {div},
772  row: (A) -> Node ! {rt, div}
773): Node ! {rt, div, overflow} {
774  let g := node_of(t)
775  emit(t, CreateGroup(g))
776  let rows := perform ref_new(Nil)
777  // The rows belong to the owner `keyed_each` was made in, not to the watcher, so a change of
778  // `items` keeps the rows it keeps.
779  let home := scope(() => #())
780  watch(() => {
781    let xs := items()
782    let old := perform ref_get(rows)
783    var keys: HashMap(u64, u64) := HashMap::new()
784    keys_of(&mut keys, xs, key)
785    end_missing(t, old, &keys)
786    let fresh := reverse_rows(in_owner(home.owner, () => {
787      var by_key: HashMap(u64, KeyedRow) := HashMap::new()
788      rows_by_key(&mut by_key, old)
789      var seen: HashMap(u64, u64) := HashMap::new()
790      rows_for(&by_key, xs, key, row, t.duplicate_keys, &mut seen, Nil)
791    }), Nil)
792    perform ref_set(rows, fresh)
793    match old {
794      Nil => append_rows(t, g, fresh)
795      Cons(o, more) => {
796        let keep := staying(old, fresh)
797        place_rows(t, g, reverse_indexed(fresh, 0, Nil), &keep, None)
798      }
799    }
800  })
801  on_cleanup(() => dispose(home.owner))
802  g
803}

805// `build()`, marked as an instance of the template `id`: every call with the same `id` must
806// make the same elements with the same static properties, differing only in text and in
807// followed properties.
808pub fn template<K: Shared + NoBorrow, P: Shared + NoBorrow>(
809  t: Tree(K, P),
810  id: u64,
811  build: () -> Node ! {rt, div}
812): Node ! {rt, div} {
813  let n := build()
814  emit(t, Template(id, n))
815  n
816}

818// What a row of `keyed_each_signal` may be keyed by: a `Nat` or a `u64`.
819pub class RowKey(R: Type) {
820  fn row_key(r: R): Nat
821}
822instance RowKey(Nat) {
823  fn row_key(r: Nat): Nat { r }
824}
825instance RowKey(u64) {
826  fn row_key(r: u64): Nat { Nat::of_u64(r) }
827}

829// Like keyed_each, with a live value per retained key. The row is built once; use get(item)
830// in text_dyn or Dyn to follow changes without losing focus, handlers or row-local state.
831pub fn keyed_each_signal<K: Shared + NoBorrow, P: Shared + NoBorrow, A: Shared + NoBorrow, R>(
832  t: Tree(K, P),
833  items: () -> List(A) ! {rt, div},
834  key_by: (A) -> R ! {div},
835  row: (delim::src::signal::Signal(A)) -> Node ! {rt, div}
836): Node ! {rt, div, overflow} where RowKey(R) {
837  let key := (a: A) => U64::of_nat(RowKey::row_key(key_by(a)))
838  let values := memo(items)
839  keyed_each(t, () => get(values), key, (initial: A) => {
840    let k := key(initial)
841    let item := memo(() => item_for_key(get(values), key, k, initial))
842    row(item)
843  })
844}
845
846fn item_for_key<A: Shared + NoBorrow>(
847  xs: List(A),
848  key: (A) -> u64 ! {div},
849  k: u64,
850  fallback: A
851): A ! {div} {
852  match xs {
853    Nil => fallback
854    Cons(x, rest) => if key(x) == k { x } else { item_for_key(rest, key, k, fallback) }
855  }
