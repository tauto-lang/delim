1// Context: a value the components below a point can read without being handed it. It is the
2// effect `context(A)`, one instance for each type `A`: `with_context(v, body)` handles it for
3// `body` (`provide` in `ui.taut` for a view), and `use_context()` performs it. A component that reads a context has
4// `context(A)` in its type, so one used outside any `provide` does not type-check. To give
5// one type two contexts, wrap the values in different structs.
6
7pub effect context<A>(): A direct
8
9pub fn use_context<A>(): A ! {context(A)} { perform context() }
10
11pub fn with_context<A, T, !e>(value: A, body: () -> T ! {context(A)} ∪ e): T ! e {
12  handle { body() } with {
13    context<A>(), k => k(value)
14    return v => v
15  }
16}
